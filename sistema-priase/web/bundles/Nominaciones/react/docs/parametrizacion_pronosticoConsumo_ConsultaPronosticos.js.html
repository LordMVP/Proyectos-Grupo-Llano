import React, { Component } from 'react';
import connect from 'react-redux/es/connect/connect';
import { bindActionCreators } from 'redux';
import PropTypes from 'prop-types';
import { Input, Botonera, Fecha, Combo, Util, Tabla } from 'appfuture-react';
import RUTAS_API from '../../../global/rutas_api';
import RUTAS_VISTA from '../../../global/rutas_vista';
import { TECLAS } from '../../../global/constantes';
import axios from 'axios';

class ConsultaPronosticos extends Component {

  state = {
    criterio: '',
    fechaInicial: '',
    fechaFinal: '',
    puntosConsumo: this.props.puntosConsumo,
    puntoConsumo: '',
    listadoEntidad: [],
  };

  /**
   * Método encargado de generar las columnas para el componente tabla.
   * @returns {Array}
   */
  obtenerColumnas = () => {
    const contexto = this;
    return [
      {
        Header: 'Consulta Pronosticos',
        columns: [
          {
            Header: 'Punto de Consumo',
            accessor: 'ptcIdepuntoconsumo.ptcIderegistro',
            Cell: contexto.obtenerPuntoConsumo
          },
          {
            Header: 'Valor',
            accessor: 'prnValor'
          },
          {
            Header: 'Fecha ',
            accessor: 'prnFechapronostico',
            Cell: contexto.obtenerFecha,
          },

        ]
      }
    ];
  };


  /**
   * Método encargado de obtener el nombre del punto de consumo.
   * @param {Object} props Propiedades del componente tabla.
   * @returns {String}
   */
  obtenerPuntoConsumo = (props) => {
    const { puntosConsumo } = this.state;
    const idPunto = parseInt(props.row._original.ptcIdepuntoconsumo.ptcIderegistro);
    const punto = puntosConsumo.find(p => p.ptcIderegistro === idPunto);
    return (punto) ? punto.ptcoNombre : '';
  };

  /**
  * Método encargado de obtener el formato string de la fecha
  * @param {Object} props Propiedades del componente Tabla
  */
  obtenerFecha = (props) => {
    const fechaNumero = parseInt(props.row._original.prnFechapronostico);
    let fecha = new Date(fechaNumero);
    const anio = fecha.getFullYear();
    const dia = fecha.getDate();
    const mes = fecha.getMonth();
    fecha = `${anio}/${(mes < 10) ? '0' : ''}${mes + 1}/${(dia < 10) ? '0' : ''}${dia}`;
    return (fechaNumero) ? fecha : '';
  };

  /**
   * Método encargado de generar los botones del formulario
	 * @returns {Object}
   */
  obtenerFunciones = () => {
    let funciones = [{ texto: 'Consultar', callback: this.onBuscar }];
    funciones.push({ texto: 'Limpiar', callback: this.limpiarFormulario });
    return funciones;
  };

  /**
   * Metodo encargado de realizar la consulta
   * @returns {bool}
   */
  onBuscar = () => {
    const { fechaInicial, fechaFinal, puntoConsumo, puntosConsumo } = this.state;
    let punto = '';
    if (fechaInicial.trim() === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar la fecha final inicial');
      return false;
    }

    if (fechaFinal.trim() === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar la fecha final');
      return false;
    }
    if (puntoConsumo != '') {
      punto = puntosConsumo.find(p => p.ptcIderegistro == puntoConsumo);
    }
    const parametros = {
      'fechainicio': fechaInicial,
      'fechafinal': fechaFinal,
      'criterio': (punto != '') ? punto.ptcoNombre : punto,
    };

    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_PRONOSTICO_CONSUMO.CONSULTAR_PRONOSTICOS, parametros)
      .then(respuesta => {
        this.setState({ listadoEntidad: respuesta.data.datos });
      });
  };

  /**
   * Método encargado de limpiar el formulario
   */
  limpiarFormulario = () => {
    this.setState({
      criterio: '',
      puntoConsumo: '',
      fechaFinal: '',
      fechaInicial: '',
      listadoEntidad: [],
    });
  };

  /**
   * Método encargado de controlar el cambio del criterio
   * @param {Event} event El evento que se ejecuta en el control de usuario.
   */
  onCriterioChange = (event) => {
    this.setState({ criterio: event.target.value });
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    this.setState(change);
  };

  /**
   * Método encargado de generar el componente Tabla
   * @returns {Component}
   */
  renderTabla = () => {
    if (!Util.validarArreglo(this.state.listadoEntidad)) {
      return <div className='text-center'>Sin resultados</div>;
    }

    return (
      <Tabla
        datos={this.state.listadoEntidad}
        columnas={this.obtenerColumnas()}
      />
    );
  };

  /**
   * Metodo encargado de realizar la consulta cuando se preciona la tecla enter
   * @returns {bool}
   */
  onKeyPress = (evento) => {
    if (evento.charCode === TECLAS.ENTER) {
      this.onBuscar()
    }
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <div className='consulta-tramos'>
        <Botonera funciones={this.obtenerFunciones()} />
        <div className='row'>
          <Combo
            opciones={this.state.puntosConsumo}
            propTexto='ptcoNombre'
            propValor='ptcIderegistro'
            label='Punto de Consumo:'
            name='puntoConsumo'
            value={this.state.puntoConsumo}
            onChange={this.controlarCambio}
          />
          <Fecha
            label='Fecha Inicial:'
            name='fechaInicial'
            fecha={this.state.fechaInicial}
            onChange={this.controlarCambio}
          />
          <Fecha
            label='Fecha Final:'
            name='fechaFinal'
            fecha={this.state.fechaFinal}
            onChange={this.controlarCambio}
          />
          <div className='mt-5 col-12'>
            {this.renderTabla()}
          </div>
        </div>
      </div>
    );
  };
}

ConsultaPronosticos.propTypes = {
  history: PropTypes.object,
  esModal: PropTypes.bool,
  seleccionMultiple: PropTypes.bool,
  seleccionarEntidad: PropTypes.func,
  seleccionarEntidades: PropTypes.func,
  entidadesSeleccionadas: PropTypes.array,
  mostrarAlerta: PropTypes.func,
  puntosConsumo: PropTypes.array,
};

ConsultaPronosticos.defaultProps = {
  esModal: false
};
