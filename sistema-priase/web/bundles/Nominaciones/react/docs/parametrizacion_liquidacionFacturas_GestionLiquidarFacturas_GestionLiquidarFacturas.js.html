import React, { Component, Fragment } from 'react';
import connect from 'react-redux/es/connect/connect';
import { bindActionCreators } from 'redux';
import PropTypes from 'prop-types';
import { Input, Botonera, Combo, Tabla, VentanaModal, Util, Fecha, TextoNumerico } from 'appfuture-react';
import axios from 'axios';
import RUTAS_API from '../../../../global/rutas_api';
import { mostrarAlerta } from '../../../../store/actions/AplicacionAcciones';
import { limpiarHistorico } from '../../../../global/util_nominaciones';
import RUTAS_VISTA from '../../../../global/rutas_vista';
import './GestionLiquidarFacturas.scss';
import { toast } from 'react-toastify';

const listaSiNo = [
  { texto: 'Aprobado', valor: 'A' },
  { texto: 'Pendiente', valor: 'P' },
];

const EVENTO_EXIMENTE = 'Evento Eximente';
const NORMAL = 'Normal';
const PENDIENTE = 'P';
const APROBADO = 'A';

class GestionLiquidarFacturas extends Component {

  state = {
    // Datos de la entidad
    fechaInicio: '',
    fechaFin: '',
    aprobado: '',
    tercero: '',
    subtotal: '',
    contratoDetalle: '',
    listaContratos: [],
    objetoOriginal: {},
    listaCargos: [],
    // Estado de la aplicacion
    mostrarModalConsulta: false,
    estado: false,
  };

  /**
   * Método encargado de comprobar si el formulario ya cargo.
   */
  componentDidMount() {
    const { state } = this.props.history && this.props.history.location;
    if (state && state.entidadEditar) {
      const tercero = state.entidadEditar;
      this.setState({ tercero: tercero });
    }
  };

  /**
   * Método encargado ejecutar una acción cuando se elimina el componente.
   */
  componentWillUnmount() {
    this.props.history.replace({ entidadEditar: null });
  };

  /**
   * Método encargado de limpiar los campos del formulario.
   * @param {Event} evento El evento que se ejecuta en el control de usuario.
   */
  limpiarFormulario = (evento) => {
    this.setState({
      // Datos de la entidad
      fechaInicio: '',
      fechaFin: '',
      aprobado: '',
      tercero: '',
      subtotal: '',
      contratoDetalle: '',
      listaContratos: [],
      objetoOriginal: '',
      listaCargos: [],
      // Estado de la aplicacion
      mostrarModalConsulta: false,
      estado: false,
    });
    limpiarHistorico(this.props);
  };

  /**
   * Método encargado de limpiar el formulario al momento de salir.
   */
  componentWillUnmount() {
    this.limpiarFormulario();
  };

  /**
   * Método encargado de generar los botones del formulario.
   * @returns {Object}
   */
  obtenerFunciones = () => {
    return [
      { texto: 'Guardar', callback: this.guardarEntidad },
      { texto: 'Consultar Detalle', callback: this.consultarDetalle },
      { texto: 'Consultar', callback: this.consultarEntidad },
      { texto: 'Limpiar', callback: this.limpiarFormulario }
    ];
  };

  /**
   * Método encargado de validar las variables del formulario.
   * @returns {Object}
   */
  validarFormulario = () => {
    const { aprobado, listaContratos, fechaInicio, fechaFin } = this.state;
    const listaFiltrada = listaContratos.filter(p => p.seleccionado == true);
    // Ejemplo Validacion
    if (aprobado == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe de especificar si se ha aprobado o no el cruce' } };
    }
    if (fechaInicio == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar la fecha inicial.' } };
    }

    if (fechaFin == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar la fecha final.' } };
    }
    if (!Util.validarArreglo(listaFiltrada)) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar al menos un contrato' } };
    }

    return { respuesta: true };
  };

  /**
   * Método encargado de consultar la liquidacion.
   * @returns {bool}
   */
  consultarEntidad = async () => {
    const tercero = { ...this.state.tercero };
    const verificar = (Object.keys(tercero).length === 0) ? false : true;
    if (!verificar) {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar un tercero');
      return false;
    }
    const respuesta = await axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_TRANSPORTE_KPC.CONSULTAR_LIQUIDACION, { idtercero: tercero.terIderegistro });
    if (respuesta.data.codigo > 0) {
      const datos = JSON.parse(respuesta.data.datos);
      const contratos = datos.detalleFactura;
      this.setState({
        listaContratos: contratos,
        objetoOriginal: datos,
        estado: true,
        fechaInicio: this.obtenerFecha(datos.detalleFactura[0].fechaInicio),
        fechaFin: this.obtenerFecha(datos.detalleFactura[0].fechaFinal)
      }, this.consultarCargos);
    }
  };

  /**
   * Método encargado de procesar el detalle para validar las fechas.
   * @returns {bool}
   */
  validarValorEventoEximente = (contrato) => {
    if (!contrato.listaFecha) {
      return true;
    }
    for (let index = 0; index < contrato.listaFecha.length; index++) {
      const fecha = contrato.listaFecha[index];
      if (!fecha.eveCantidad || fecha.eveCantidad === '') {
        this.props.mostrarAlerta('Datos incompletos', 'Debe ingresar un valor para todos los eventos eximentes');
        return false;
      }
    }
  };

  /**
   * Método encargado de validar los valores de la tabla detalle.
   * @return {bool}
   */
  validarTablaDetalle = () => {
    const listaCompuestos = this.state.listaContratos.filter(c => c.seleccionado);
    for (let i = 0; i < listaCompuestos.length; i++) {
      const contrato = listaCompuestos[i];
      this.validarValorEventoEximente(contrato);
    };
    return true;
  };

  /**
   * Método encargado de validar los datos necesarios para consultar el detalle.
   * @returns {Object}
   */
  validarDetalle = () => {
    const { tercero, fechaFin, fechaInicio } = this.state;
    if (typeof tercero == 'undefined' || tercero == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe consultar un tercero.' } };
    }

    if (fechaInicio == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar la fecha inicial.' } };
    }

    if (fechaFin == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar la fecha final.' } };
    }

    return { respuesta: true };
  };

  /**
   * Método encargado de consultar el detalle.
   * @returns {boolean}
   */
  consultarDetalle = async () => {
    const { fechaFin, fechaInicio, tercero } = this.state;
    const validacion = this.validarDetalle();
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return;
    }
    const parametros = {
      fechainicio: fechaInicio,
      fechafinal: fechaFin,
      idtercero: tercero.terIderegistro
    };
    const respuesta = await axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_TRANSPORTE_KPC.CONSULTAR_DETALLE_LIQUIDACION, parametros);
    if (respuesta.data.codigo > 0) {
      const datos = JSON.parse(respuesta.data.datos);
      const contratos = datos[0].detalleFactura;
      if (contratos == null) {
        toast.error('No se encontrarón datos del detalle');
        return;
      }
      this.setState({
        listaContratos: contratos,
        objetoOriginal: datos[0]
      }, this.consultarCargos);
    }
  };

  /**
   * Método encargado de obtener el objeto para guardar.
   * @returns {Array}
   */
  obtenerObjetoGuardar = () => {
    const { listaContratos, aprobado, fechaFin, fechaInicio, tercero } = this.state;
    const objeto = {
      fechaInicio: fechaInicio,
      fechaFinal: fechaFin,
      terIderegistro: tercero.terIderegistro,
      terNomcompleto: tercero.terNomcompleto,
      verificarAprobacion: aprobado,
    }
    const lista = listaContratos.filter(dato => dato.seleccionado == true).map((dato) => {
      return {
        contrato: dato.contrato,
        ultimotrm: dato.ultimotrm,
        nominacionMbtu: dato.nominacionMbtu,
        nominacionKpc: dato.nominacionKpc,
        listaFecha: (!dato.listaFecha) ? [] : dato.listaFecha,
        cantidadNominadambtu: dato.cantidadNominadambtu,
        cantidadNominadakpc: dato.cantidadNominadakpc,
        podercalorifico: dato.podercalorifico,
        cantidadSuperior: dato.cantidadSuperior,
        cargoContratos: {
          contrato: dato.cargo.contrato,
          cargoFjio: dato.cargo.cargoFjio,
          cargoVarible: dato.cargo.cargoVarible,
          cargoAom: dato.cargo.cargoAom,
          impuestoFomento: dato.cargo.impuestoFomento,
          subtotal: dato.cargo.subtotal,
          totalCruce: (dato.cruce) ? dato.cruce[0].total : null
        },
      }
    });
    objeto.detalleFactura = lista;
    return objeto;
  };

  /**
   * Método encargado de guardar los datos de la entidad.
   * @returns {bool}
   */
  guardarEntidad = () => {
    const validacion = this.validarFormulario();
    const validarTablaDetalle = this.validarTablaDetalle();
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return false;
    }
    if (!validarTablaDetalle) {
      return false;
    }
    const entidadGuardar = this.obtenerObjetoGuardar();

    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_TRANSPORTE_KPC.GUARDAR, entidadGuardar)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.limpiarFormulario();
        }
      });
  };

  /**
   * Método encargado de abrir la interfaz de consulta de terceros.
   */
  consultarTercero = () => {
    this.props.history.push({
      pathname: RUTAS_VISTA.CONSULTA_TERCEROS.url,
      state: {
        interfazGestion: RUTAS_VISTA.GESTION_LIQUIDACION_FACTURAS_TRANSPORTE_KPC.url,
      }
    });
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario.
   * @param {Event} evento El evento que se ejecuta en el control de usuario.
   */
  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    this.setState(change);
  };

  /**
   * Controla el evento change de los checks de la tabla usuarios.
   */
  controlarCambioCheck = (evento) => {
    const contratos = [...this.state.listaContratos];
    const value = evento.target.value;
    const index = contratos.findIndex(c => c.contrato.cntNumero == value);
    const contrato = { ...contratos[index] };
    contrato.seleccionado = evento.target.checked;
    contratos[index] = contrato;
    this.setState({ listaContratos: contratos });
  };

  /**
   * Método encargado de consultas las listas del contrato seleccionado.
   */
  consultarCargos = async () => {
    let { subtotal } = this.state;
    const listaContratos = [...this.state.listaContratos];
    const lista = { ...this.state.objetoOriginal };
    const respuesta = await axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_TRANSPORTE_KPC.CONSULTAR_CARGOS_CONTRATO, lista);
    if (respuesta.data.codigo > 0) {
      const cargos = respuesta.data.datos;
      subtotal = cargos[0].subtotal;
      cargos.forEach((cargo, index) => {
        listaContratos[index].cargo = cargo;
        listaContratos[index].cruce = (cargo.info) ? JSON.parse(cargo.info) : '';
      });
    }
    this.setState({
      listaContratos: listaContratos,
      subtotal: subtotal
    });
  };

  /**
   * Método encargado de controlar los cambios en los valores de la tabla.
   */
  controlarCambioTabla = (evento) => {
    const control = evento.target;
    const nombrePropiedad = control.name;
    const indexContrato = control.attributes['data-indexcontrato'].value;
    const indexFecha = control.attributes['data-indexfecha'].value;
    if (nombrePropiedad === 'eveCantidad') {
      const listaContratos = [...this.state.listaContratos];
      listaContratos[indexContrato].listaFecha[indexFecha].eveCantidad = control.value;
      this.setState({ listaContratos: listaContratos }, () => {
        control.focus();
      });
    }
  };

  /**
   * Método encargado de mostrar la lista detalle de la cantidad contratada del contrato seleccionado
   * @returns {JSX}
   */
  renderListaDetalle = () => {
    const { contratoDetalle, listaContratos } = this.state;
    const detalleRender = listaContratos.find(c => c.contrato.cntNumero == contratoDetalle);
    return (
      <table className='table-normaliced table table-condensed table-bordered'>
        <thead>
          <tr className='bg-dark text-white'>
            <th>Fecha</th>
            <th>Cantidad Contratada</th>
            <th>Unidad Medida</th>
          </tr>
        </thead>
        <tbody>
          {
            detalleRender.listaDetalle.map((dato, index) => {
              return (
                <tr key={dato.fecha}>
                  <td>{dato.fecha}</td>
                  <td>
                    <TextoNumerico
                      aceptaDecimales={false}
                      aceptaNegativos={false}
                      cols={12}
                      value={dato.cantidadContratada}
                      onChange={this.controlarCambioCantidadContratada}
                      name='cantidadContratada'
                      extra={{ 'data-index': index, disabled: true }}
                    />
                  </td>
                  <td>{dato.unidad.uniNombre1}</td>
                </tr>
              )
            })
          }
        </tbody>
      </table>
    )
  };

  /**
   * Método encargado de consultar el detalle día a día del contrato seleccionado
   * @param {Event} evento
   */
  verDetalle = (evento) => {
    const { fechaFin, fechaInicio, listaContratos } = this.state;
    const idContrato = evento.target.attributes['data-id'].value;
    const obj = {
      fechaInicial: fechaInicio,
      fechaFinal: fechaFin,
      idContrato: idContrato
    };
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_SUMINISTRO_NEGOCIACION_DIRECTA.CONSULTAR_DIA_DIA, obj)
      .then(respuesta => {
        const index = listaContratos.findIndex(c => c.contrato.cntNumero == idContrato);
        const data = respuesta.data.datos;
        listaContratos[index].listaDetalle = data;
        this.setState({
          contratoDetalle: idContrato,
          listaContratos: listaContratos
        });
      });
  };

  /**
   * Método encargado de generar la tabla de los contratos.
   * @returns {Object}
   */
  renderTablaTercero = () => {
    const lista = this.state.listaContratos;
    if (!Util.validarArreglo(lista)) {
      return null;
    }
    return (
      <table className='table-normaliced table table-condensed table-bordered'>
        <thead className='bg-dark text-white'>
          <tr>
            <th className='text-center'>Acciones</th>
            <th className='text-center'>Seleccion</th>
            <th className='text-center'>Contrato</th>
            <th className='text-center'>Modalidad de Contrato</th>
            <th className='text-center'>Fecha de Liquidación de Contrato</th>
            <th className='text-center'>TRM</th>
            <th className='text-center'>Nominación en MBTU</th>
            <th className='text-center'>Nominación en KPC</th>
            <th className='text-center'>Capacidad Contratada MBTU</th>
            <th className='text-center'>Capacidad Contratada KPC</th>
            <th className='text-center'>Poder Calorifico</th>
            <th className='text-center'>Cantidad Superior a la Contratada</th>
            <th className='text-center'>Novedad</th>
            <th colSpan={2} className='text-center'>Modificación Firmeza por Día</th>
          </tr>
        </thead>
        <tbody>
          {Util.validarArreglo(lista) && (
            lista.map((dato, index) => {
              let tamanioFilas = 1;
              if (Util.validarArreglo(dato.listaFecha)) {
                dato.listaFecha.forEach(fecha => {
                  tamanioFilas++;
                });
              }
              return (
                <Fragment key={(dato.contrato.cntNumero)}>
                  <tr key={(dato.contrato.cntNumero)}>
                    <td rowSpan={tamanioFilas}>
                      <a href='javascript:;' onClick={this.verDetalle} data-id={dato.contrato.cntNumero}>Ver Detalle</a>
                    </td>
                    {(!dato.verificarAprobacion || dato.verificarAprobacion == PENDIENTE) &&
                      <td rowSpan={tamanioFilas}>
                        <label>
                          <input
                            type="checkbox"
                            name={Util.generarIdControl('check_' + index)} checked={dato.seleccionado || false}
                            value={dato.contrato.cntNumero}
                            onChange={this.controlarCambioCheck} />
                          Seleccionar</label>
                      </td>
                    }
                    {dato.verificarAprobacion == APROBADO &&
                      <td rowSpan={tamanioFilas}><i className='fa fa-fw fa-warning'></i>La liquidación se encuentra aprobada</td>
                    }
                    <td rowSpan={tamanioFilas}>{dato.contrato.cntNumero}</td>
                    <td rowSpan={tamanioFilas}>{dato.contrato.uniIdemodalidadcontrato.uniNombre1}</td>
                    <td rowSpan={tamanioFilas}>{`2019-08-20`}</td>
                    <td rowSpan={tamanioFilas}>{dato.ultimotrm}</td>
                    <td rowSpan={tamanioFilas}>{dato.nominacionMbtu}</td>
                    <td rowSpan={tamanioFilas}>{dato.nominacionKpc}</td>
                    <td rowSpan={tamanioFilas}>{dato.cantidadNominadambtu}</td>
                    <td rowSpan={tamanioFilas}>{dato.cantidadNominadakpc}</td>
                    <td rowSpan={tamanioFilas}>{dato.podercalorifico}</td>
                    <td rowSpan={tamanioFilas}>{dato.cantidadSuperior}</td>
                    <td rowSpan={tamanioFilas}>{(dato.listaFecha && dato.listaFecha.length > 0) ? EVENTO_EXIMENTE : NORMAL}</td>
                    {(!dato.listaFecha) &&
                      <td colSpan='2'><i className='fa fa-fw fa-warning'></i>No presenta evento eximente</td>
                    }
                    {(dato.listaFecha && dato.listaFecha.length > 0) &&
                      (
                        <Fragment>
                          <td className='th-sub bg-dark text-white'>Fecha</td>
                          <td className='th-sub bg-dark text-white'>Valor</td>
                        </Fragment>
                      )
                    }
                  </tr>
                  {
                    Util.validarArreglo(dato.listaFecha) && (
                      dato.listaFecha.map((fecha, indexFecha) => {
                        return (
                          <Fragment key={(fecha.eveFechainicio + indexFecha)}>
                            <tr key={(fecha.eveFechainicio + indexFecha + 'TR')}>
                              <td>{fecha.eveFechainicio}</td>
                              <td>
                                <div className='form-group col-12 no-label input-table'>
                                  <TextoNumerico
                                    aceptaDecimales={false}
                                    aceptaNegativos={false}
                                    cols={12}
                                    value={fecha.eveCantidad}
                                    onChange={this.controlarCambioTabla}
                                    name='eveCantidad'
                                    extra={{ 'data-indexfecha': indexFecha, 'data-indexcontrato': index }}
                                  />
                                </div>
                              </td>
                            </tr>
                          </Fragment>
                        );
                      })
                    )
                  }
                </Fragment>
              );
            })
          )}
        </tbody>
      </table>
    );
  };

  /**
   * Método encargado de generar la tabla de cargos.
   * @returns {Object}
   */
  renderTablaCargos = () => {
    let lista = [...this.state.listaContratos];
    lista = lista.filter(p => p.seleccionado == true);
    if (lista.length <= 0) {
      return;
    }
    return (
      <table className='table table-striped'>
        <thead>
          <tr>
            <th>Contrato</th>
            <th>Cargo Fijo</th>
            <th>Cargo Variable</th>
            <th>Cargo AOM</th>
            <th>Impuesto Fomento Y transporte</th>
          </tr>
        </thead>
        <tbody>
          {lista.map((dato, index) => {
            return (
              <tr>
                <td>{dato.cargo.contrato.cntNumero}</td>
                <td>{dato.cargo.cargoFjio}</td>
                <td>{dato.cargo.cargoVarible}</td>
                <td>{dato.cargo.cargoAom}</td>
                <td>{dato.cargo.impuestoFomento}</td>
              </tr>
            )
          })}
        </tbody>
      </table>
    );
  };

  /**
   * Método encargado de para obtener la fecha y retornar en formato String.
   * @param {String} fecha fecha a convertir.
   * @returns {String}
   */
  obtenerFecha = (fecha) => {
    if (!fecha || fecha == null) {
      return '';
    }
    fecha = new Date(fecha);
    fecha = (fecha.getFullYear() + "-" + (fecha.getMonth() + 1) + "-" + fecha.getDate());
    return fecha;
  };

  /**
   * Método encargado de generar la tabla de los contratos para cruce.
   * @returns {Object}
   */
  renderTablaSeleccionarContrato = () => {
    let lista = [...this.state.listaContratos];
    lista = lista.filter(p => p.seleccionado == true && p.cruce && p.cruce.length > 0);
    if (lista.length <= 0) {
      return;
    }
    return (
      <table className='table table-striped'>
        <thead>
          <tr>
            <th className='text-center'>Contrato</th>
            <th colSpan={3} className='text-center'>Datos Cruce</th>
            <th className='text-center'>Total</th>
          </tr>
        </thead>
        <tbody>
          {lista.map((dato, index) => {
            return (
              dato.cruce.map((listaFecha, indexFecha) => {
                return (
                  <Fragment key={Util.generarIdControl(indexFecha)}>
                    <tr key={Util.generarIdControl(indexFecha + '_tr')}>
                      <td rowSpan={listaFecha.fecha.length + 1}>{dato.contrato.cntNumero}</td>
                      <td>Fecha Inicio Cobertura</td>
                      <td>Fecha Fin Cobertura</td>
                      <td>Saldo</td>
                      <td rowSpan={listaFecha.fecha.length + 1}>{((dato.cargo.totalCruce && dato.cargo.totalCruce != '') ? dato.cargo.totalCruce : listaFecha.total)}</td>
                    </tr>
                    {
                      listaFecha.fecha.map((cruce) => {
                        return (
                          <Fragment key={cruce.cntgIderegistro}>
                            <tr key={'tr_' + cruce.cntgIderegistro}>
                              <td>{this.obtenerFecha(cruce.cntgFechainicio)}</td>
                              <td>{this.obtenerFecha(cruce.cntgFechafin)}</td>
                              <td>{cruce.cntgValorcontrato}</td>
                            </tr>
                          </Fragment>
                        );
                      })
                    }
                  </Fragment>
                );
              })
            );
          })}
        </tbody>
      </table>
    );
  };

  /**
   * Método encargado de mostrar el formulario.
   * @returns {Object}
   */
  render() {
    return (
      <Fragment>
        <Botonera funciones={this.obtenerFunciones()} />
        <div className='group-section'>
          <label className='legend-section'>Periodo a Liquidar</label>
          <div className='body-section'>

                  <input

                  />

              <Fecha
                label='Fecha Inicial:'

                extra={{ disabled: this.state.estado, readOnly: this.state.estado }}
              />
              <Fecha
                label='Fecha Final:'
                name='fechaFin'
                fecha={this.state.fechaFin}
                onChange={this.controlarCambio}
                extra={{ disabled: this.state.estado, readOnly: this.state.estado }}
              />
            </div>
          </div>
        </div>
        <div className='conf-general row mt-5'>
          {this.state.listaContratos.length > 0 &&
            this.renderTablaTercero()
          }
          {this.state.contratoDetalle != '' &&
            this.renderListaDetalle()
          }
          <div className='col-6'>
            {this.state.listaContratos.length > 0 &&
              this.renderTablaCargos()
            }
          </div>
          <div className='col-6'>
            {this.state.listaContratos.length > 0 &&
              this.renderTablaSeleccionarContrato()
            }
          </div>
          <Combo
            opciones={listaSiNo}
            propTexto='texto'
            propValor='valor'
            label='Aprobar:'
            name='aprobado'
            value={this.state.aprobado}
            onChange={this.controlarCambio}
          />
          <TextoNumerico
            aceptaDecimales={false}
            aceptaNegativos={false}
            label='Subtotal en Pesos:'
            value={this.state.subtotal}
            onChange={this.controlarCambio}
            name='subtotal'
            extra={{ disable: true, readOnly: true }}
          />
        </div>
      </Fragment>
    );
  };
}
