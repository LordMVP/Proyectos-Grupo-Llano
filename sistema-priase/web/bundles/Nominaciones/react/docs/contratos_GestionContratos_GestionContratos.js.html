  /**
   * Verifica si se está consultando un contrato y ejecuta la consulta correspondiente para obtener los datos.
   * @returns {Boolean}
   */
  consultarContrato = () => {
    const { idContrato } = this.state;
    if (!idContrato) {
      return;
    }
    axios.post(RUTAS_API.CONTRATOS.CONSULTAR_DETALLE_CONTRATO, { idContrato: idContrato })
      .then((respuesta) => {
        if (respuesta.data.codigo > 0) {
          this.mostrarObjeto(respuesta.data.datos);
        } else {
          this.props.mostrarAlerta('Error', 'No se encontró el contrato');
        }
      });
  };

  /**
   * Valdia el formulario.
   * @return {object}

  validarFormulario = () => {
    if (!this.props.cabecera.agenteTercero) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el tercero.' } };

      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Complete la subida de los documentos.' } };;

    const tiposContrato = this.props.listas.tiposContrato.filter(tipoContrato => (tipoContrato.seleccionado && tipoContrato.uniPropiedad && tipoContrato.uniPropiedad.tipocontrato == 'CNX') || (tipoContrato.seleccionado && tipoContrato.uniPropiedad && tipoContrato.uniPropiedad.tipocontrato == 'GNC'));

      const rutasSeleccionadas = this.props.listas.rutasGNC.filter(ruta => ruta.seleccionado);

  /**
   * Obtiene el objeto garantia teniendo en cuenta el tipo...

    //Tener en cuenta que hay tres tipos de garantia.
    let { garantia, cabecera } = this.props;
    const tipoGarantia = garantia.tipoGarantia;

    switch (tipoGarantia) {
      case TIPOS_GARANTIA.BANCARIA:

        break;
      case TIPOS_GARANTIA.PREPAGO:

      case TIPOS_GARANTIA.POLIZA:
        garantia = garantia.poliza;
        break;
    }

    const objetoGarantia = {
      cntgTipo: tipoGarantia,
      cntgCrucecuenta: ((garantia.cruzaCuentas) ? 'S' : 'N'),
      cntgValorcontrato: cabecera.precioContrato,
      cntgPrecio: garantia.precioGarantia, //Renombrar los precios calculados.
      cntgValordm: garantia.dm,
      cntgFechainicio: garantia.fechaInicial,
      cntgFechafin: garantia.fechaFin,
      cntgPorcentaje: garantia.porcentajeComercializacion,
      cntgNumerodias: garantia.numeroDiasPrepago,
      cntgCalculodiario: garantia.calculoDiario,

  /**
   * Obtiene el objeto de garantía usando el objeto contrato que viene del registro.
   * @return {Object}
   */
  obtenerGarantiaContrato = (contrato) => {
    if (!contrato.garantia) {
      return { tipoGarantia: 'NA' };
    }
    let garantia = {
      tipoGarantia: contrato.cntTipogarantia,
    };
    let tipoGarantia = null;
    switch (garantia.tipoGarantia) {
      case TIPOS_GARANTIA.BANCARIA:
        tipoGarantia = 'bancaria';
        break;
      case TIPOS_GARANTIA.PREPAGO:
        tipoGarantia = 'prepago';
        break;
      case TIPOS_GARANTIA.POLIZA:
        tipoGarantia = 'poliza';
        break;
    };
    let fechaIni = contrato.garantia.cntgFechainicio;
    fechaIni = moment(fechaIni);
    fechaIni = moment(fechaIni).format("YYYY-MM-DD");
    let fechaFin = contrato.garantia.cntgFechafin;
    fechaFin = moment(fechaFin);
    fechaFin = moment(fechaFin).format("YYYY-MM-DD");
    garantia[tipoGarantia] = {
      cruzaCuentas: ('S' === contrato.garantia.cntgCrucecuenta),
      precioContrato: contrato.garantia.cntgValorcontrato,
      precioGarantia: contrato.garantia.cntgPrecio,
      dm: contrato.garantia.cntgValordm,
      fechaInicial: fechaIni,
      fechaFin: fechaFin,
      porcentajeComercializacion: contrato.garantia.cntgPorcentaje,
      numeroDiasPrepago: contrato.garantia.cntgNumerodias,
      calculoDiario: contrato.cntCalculoDiario,
      expidePoliza: contrato.garantia.cntgEmpresaexpide,
      tipoContrato: contrato.garantia.cntgTipocontrato,
      valorCruceCuentas: contrato.garantia.cruce
    };
    return garantia;
  };

  obtenerUnidadMedidaCargo = (tramo, tipo) => {

    } else if (canasta.tarifaHoraria) {

  /**
   * Obtiene el valor por defecto de los rangos que no tienen tramos ni rutas.

   * @return {Array}

  obtenerValorPorDefecto = (rango) => {
    return [{
      trmIdetramo: rango.tramo ? {
        trmIderegistro: typeof rango.tramo === 'object' ? rango.tramo.trmIderegistro : rango.tramo
      } : null,
      cncvValor: (rango.valorRango != undefined) ? rango.valorRango : rango.valor,
      uniIdemedida: (rango.unidadMedida) ? {
        uniIderegistro: rango.unidadMedida
      } : null,
      cncvPorcentaje: (rango.porcentajeRango !== undefined) ? rango.porcentajeRango : rango.porcentaje,
      uniIderuta: {
        uniIderegistro: null
      }
    }];
  };


  /**
   * Obtiene la lista de valores por rutas...
   * @param {Object} rango Datos del rango
   * @return {Array}
   */
  obtenerValoresPorRutas = (rango) => {
    return rango.rutas.map((ruta) => {
      return {
        trmIdetramo: rango.tramo ? {
          trmIderegistro: rango.tramo
        } : null,
        cncvValor: ruta.valor,
        uniIdemedida: {
          uniIderegistro: ruta.unidadMedida
        },
        cncvPorcentaje: (ruta.porcentajeRango !== undefined) ? ruta.porcentajeRango : ruta.porcentaje,

          uniIderegistro: ruta.uniIderegistro
        }
      }

  /**
   * Obtiene la lista de valores por tramos...
   * @param {Object} rango Datos del rango
   * @return {Array}
   */
  obtenerValoresPorTramos = (rango) => {
    return rango.tramos.map((tramo) => {
      return {
        trcaIdetramocargo: {
          trcaIderegistro: (tramo.listaCargos.find(cargo => cargo.seleccionado)).trcaIderegistro
        },
        cncvValor: this.obtenerValorCargo(tramo, 'V'),
        uniIdemedida: {
          uniIderegistro: tramo.unidadMedida
        },
        cncvPorcentaje: (tramo.porcentajeRango !== undefined) ? tramo.porcentajeRango : ((tramo.porcentaje) ? tramo.porcentaje : tramo.valorPorcentaje),
        uniIderuta: {
          uniIderegistro: null
        }
      }
    });
  };

  /**
   * Obtiene la lista de valores por rutas con tramos...
   * @param {Object} rango Datos del rango
   * @return {Array}
   */
  obtenerValoresPorRutasConTramos = (rango) => {
    let listaValores = [];
    for (let i = 0, ruta; ruta = rango.rutas[i], i < rango.rutas.length; i++) {
      for (let i = 0, tramo; tramo = ruta.tramos[i], i < ruta.tramos.length; i++) {
        const valor = {
          trmIdetramo: {
            trmIderegistro: tramo.trmIderegistro
          },
          trcaIdetramocargo: {
            trcaIderegistro: (tramo.listaCargos.find(cargo => cargo.seleccionado)).trcaIderegistro
          },
          cncvValor: null,
          uniIdemedida: {
            uniIderegistro: (tramo.unidadMedida && tramo.unidadMedida != -1) ? tramo.unidadMedida : null
          },
          cncvPorcentaje: (tramo.valorPorcentaje != undefined) ? tramo.valorPorcentaje : tramo.trmPorcentajevariable,
          uniIderuta: {
            uniIderegistro: ruta.uniIderegistro
          }
        };
        listaValores.push(valor);
      }
    }
    return listaValores;
  };

  /**
   * Obtiene los valores por tramo.
   * @param {Object} rango Datos del rango
   * @return {Array}
   */
  obtenerValoresPorTramo = (rango) => {
    return [{
      trmIdetramo: {
        trmIderegistro: (typeof rango.tramo === 'string') ? rango.tramo : rango.tramo.trmIderegistro
      },
      trcaIdetramocargo: {
        trcaIderegistro: (tramo.listaCargos.find(cargo => cargo.seleccionado)).trcaIderegistro
      },
      cncvValor: null,
      uniIdemedida: {
        uniIderegistro: rango.tramo.unidadMedida
      },
      cncvPorcentaje: rango.porcentaje,
      uniIderuta: {
        uniIderegistro: null
      }
    }];
  };

  /**
   * Recibe la lista de valores(rutas) y retorna el arreglo en el formato que se debe enviar para guardar.
   * @param {Object} rango Datos del rango
   * @returns {Array}
   */
  obtenerListaValores = (rango) => {
    //Retorna un solo valor con los datos extra del rango...
    if (!rango.tramos && !rango.rutas) {
      return this.obtenerValorPorDefecto(rango);
    }
    //Retorna los valores por rutas...
    if (!rango.tramos && rango.rutas && rango.rutas.length > 0 && (!rango.rutas[0].tramos)) {
      return this.obtenerValoresPorRutas(rango);
    }
    //Retorna los valores por tramos...
    if (!rango.rutas && rango.tramos) {
      return this.obtenerValoresPorTramos(rango);
    }
    //Retorna los valores por rutas con tramos...
    if (!rango.tramos && rango.rutas && rango.rutas.length > 0 && (rango.rutas[0].tramos)) {
      return this.obtenerValoresPorRutasConTramos(rango);
    }
    //Retorna los valores por tramo.
    if (!rango.rutas && rango.tramo) {
      return this.obtenerValoresPorTramo(rango);
    }
  };

  obtenerCanastaTarifaria = () => {

    //Valida si el tipo de calculo seleccionado no es canasta para retornar nulo.
    if (tipoCalculo.tipoCalculo !== TIPOS_CACULO_PRECIO.CANASTA) {
      return null;
    }
    //Procesamos los rangos agregados en Canasta Horaria Suministro.
    if (tipoCalculo.tipoServicio === TIPOS_SERVICIO.SUMINISTRO) {
      return this.obtenerListaCanastaSuministro(tipoCalculo);
    }
    if (tipoCalculo.tipoServicio === TIPOS_SERVICIO.TRANSPORTE) {
      return this.obtenerListaCanastaTransporte(tipoCalculo);
    }
  };

   * @return {Number}

    const tiposContratoSeleccionados = this.obtenerTiposContratoSeleccionados();
    if (tiposContratoSeleccionados.findIndex(tc => (tc.uniPropiedad) && ((tc.uniPropiedad.tipocontrato === 'T') || tc.uniPropiedad.tipocontrato === 'S')) >= 0) {
      const claseContrato = getProp(this.props, 'cabecera.claseContrato', null);
      if (!claseContrato) {
        return null;
      }
      return {
        uniIderegistro: claseContrato
      };
    } else {
      return null;
    }
  };

   * @return {Object}

  obtenerTipoCalculoContrato = (contrato) => {
    let tipoCalculo = {
      tipoCalculo: contrato.cntTipocalculoprecio,
      tipoServicio: contrato.cntTiposervicio,
    };
    const listaCanasta = contrato.listaCanastaTarifaria;
    //Obtenemos los tipos de tarifa.
    tipoCalculo.tarifaHoraria = (!Util.validarArreglo(listaCanasta)) ? false : (((listaCanasta.findIndex(item => (item.cncTipo === TIPOS_CANASTA.HORARIA) || (item.cncTipo === TIPOS_CANASTA.DUPLEX))) >= 0) ? true : false);
    tipoCalculo.tarifaConsumo = (!Util.validarArreglo(listaCanasta)) ? false : (((listaCanasta.findIndex(item => (item.cncTipo === TIPOS_CANASTA.CONSUMO) || (item.cncTipo === TIPOS_CANASTA.DUPLEX))) >= 0) ? true : false);
    //Obtenemos la información de canastas horarias...
    tipoCalculo.canastaHorariaSuministro = (tipoCalculo.tarifaHoraria && tipoCalculo.tipoServicio === TIPOS_SERVICIO.SUMINISTRO) ? this.obtenerTarifaHorarias(listaCanasta) : {};
    tipoCalculo.canastaHorariaTransporte = (tipoCalculo.tarifaHoraria && tipoCalculo.tipoServicio === TIPOS_SERVICIO.TRANSPORTE) ? this.obtenerTarifaHorariasTransporte(listaCanasta) : {};
    //Obtenemos la información de canastas consumo...
    tipoCalculo.canastaConsumoSuministro = (tipoCalculo.tarifaConsumo && tipoCalculo.tipoServicio === TIPOS_SERVICIO.SUMINISTRO) ? this.obtenerTarifaConsumo(listaCanasta) : {};
    tipoCalculo.canastaConsumoTransporte = (tipoCalculo.tarifaConsumo && tipoCalculo.tipoServicio === TIPOS_SERVICIO.TRANSPORTE) ? this.obtenerTarifaConsumoTransporte(listaCanasta) : {};

    if (tipoCalculo.tarifaHoraria && tipoCalculo.tarifaConsumo) {
      tipoCalculo.canastaDuplexTransporte = (tipoCalculo.tipoServicio === TIPOS_SERVICIO.TRANSPORTE) ? this.obtenerDuplexTransporte(contrato.listaCanastaFusion) : {};
      tipoCalculo.canastaDuplexSuministro = (tipoCalculo.tipoServicio === TIPOS_SERVICIO.SUMINISTRO) ? this.obtenerDuplexSuministro(contrato.listaCanastaFusion) : {};

    return tipoCalculo;

      return this.obtenerRuta(valor);

                const tr = this.props.listas.listaTramos.find(t => t.trmIderegistro === tramo.trmIderegistro);

            horaFin: subRango.cncHorafin,

        cargoFijo: this.obtenerValorCargo(idTramo, 'F'),
        cargoVariable: this.obtenerValorCargo(idTramo, 'V'),
        horaFin: item.cncHoraFin,
        idRangoSinRuta: Util.generarIdControl('idRuta'),
        porcentajeRango: valores.cncvPorcentaje,
        rango: {
          horaInicio: item.cncHorainicio,
          horaFin: item.cncHorafin
        },
        tramos: item.listaValores.map(tramo => {
          //Buscamos el tramo...
          const tr = this.props.listas.listaTramos.find(t => t.trmIderegistro === tramo.trmIderegistro);
          const nuevoTramo = { ...tr };
          // const indexCargo = nuevoTramo.listaCargos.findIndex(cargo => cargo.trcaIderegistro == tramo.trcaIderegistro);
          // nuevoTramo.listaCargos[indexCargo].trcaCargovariable =
          nuevoTramo.cargoVariable = tramo.trcaCargovariable;
          nuevoTramo.valorPorcentaje = tramo.cncvPorcentaje;
          nuevoTramo.unidadMedida = tramo.uniIdemedida;
          return nuevoTramo;
        }),
        unidadMedida: valores.uniIdemedida
      };
    });

  /**
   * Obtiene un tramo...
   * @param {Number} idTramo Identificador del tramo
   * @return {object}
   */
  obtenerTramo = (idTramo) => {
    if ((typeof idTramo === 'string') || (typeof idTramo === 'number')) {
      const i = this.props.listas.listaTramos.findIndex(t => t.trmIderegistro == idTramo);
      const tramo = (i >= 0) ? this.props.listas.listaTramos[i] : null;

  /**
  * Obtiene el valor cargo de un tramo.
  * @param {Object} tramo Datos del tramo
  * @param {String} tipo Tipo del cargo
  * @return {number}
  */
  obtenerValorCargo = (tramo, tipo) => {
    if (!tramo) {
      return -1;
    }

    //Verificamos el cargo seleccionado...
    if (typeof tramo === 'string' || typeof tramo === 'number') {
      tramo = this.props.listas.listaTramos.find(t => t.trmIderegistro == tramo);
    }

    if (!Util.validarArreglo(tramo.listaCargos)) {
      return -1;
    }

    var cargoSeleccionado = tramo.listaCargos.find(cargo => cargo.seleccionado);
    switch (tipo) {
      case 'V':
        return cargoSeleccionado.trcaCargovariable;
      case 'F':
        return cargoSeleccionado.trcaCargofijo;
      case 'A':
        return cargoSeleccionado.trcaCargoaoym;
    }
  };

   * @return {array}

  obtenerTramosSeleccionados = () => {

  /**
   * Calcula el precio del contrato entre otros valores necesarios para garantias.
   * @return {Object}
   */
  calcularPrecioContrato = () => {
    const tramosSeleccionados = this.obtenerTramosSeleccionados();
    if (tramosSeleccionados.length == 0) {
      return;
    }
    let valores = {
      precioContrato: 0,
      valorCargos: {
        variable: 0,
        fijo: 0,
        aom: 0
      }
    };
    for (let i = 0; i < tramosSeleccionados.length; i++) {
      const tramo = tramosSeleccionados[i];
      let valorVariable = tramo.cargoVariable ? tramo.cargoVariable : this.obtenerValorCargo(tramo, 'V');
      let ValorFijo = this.obtenerValorCargo(tramo, 'F');
      let valorAOM = this.obtenerValorCargo(tramo, 'A');
      valores.valorCargos.variable += valorVariable;
      valores.valorCargos.fijo += ValorFijo;
      valores.valorCargos.aom += valorAOM;
      valores.precioContrato += (valorVariable + ValorFijo + valorAOM);
    }

    this.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, valorCargos: valores.valorCargos });
  };
