  /**
   * Método encargado de ejecutar acciones al momento de cargar el componente
   */
  componentDidMount() {
    const { state } = this.props.history && this.props.history.location;
    if (state && state.entidadEditar) {
      this.cargarDatos(state.entidadEditar);
    }

    this.consultarListas();

  /**
   * Método encargado de ejecutar las consultas iniciales del componente
   */
  consultarListas = () => {
    this.consultarTiposContrato();
    this.consultarTiposNegociacion();
  };

  /**
   * Consulta los tipos de negociación.
   */
  consultarTiposNegociacion = () => {
    axios.post(RUTAS_API.CONFIGURACION.CONSULTAR_UNIDAD, { criterio: '', idClase: CLASES_UNIDADES.TIPOS_NEGOCIACION })
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.setState({ listaTiposNegociacion: respuesta.data.datos });
        }
      })
  };

  /**
   * Consulta los tipos de contrato...
   */
  consultarTiposContrato = () => {
    axios.post(RUTAS_API.CONFIGURACION.CONSULTAR_UNIDAD, { criterio: '', idClase: CLASES_UNIDADES.TIPO_CONTRATO })
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.setState({ listaTiposContrato: respuesta.data.datos });
        }
      });
  };

  /**
   * Ejecuta la petición correspondiente al tipo del cálculo seleccionado.
   */
  ejecutarConsulta = () => {
    switch (this.state.tipoCalculo) {
      case TIPOS_CALCULO.RENOMINACION:
        this.consultarPuntosConsumo();
        break;
      case TIPOS_CALCULO.EXCEDENTES:
        this.calcularExcedentes();
        break;
      case TIPOS_CALCULO.PRECIO:
        this.consultarReserva();
        break;
    }
  };

  /**
   * Consulta los puntos de consumo...
   */
  consultarPuntosConsumo = () => {
    let { tipoContrato, tipoNegociacion, listaTiposContrato, listaTiposNegociacion, fechaNominacion, negociacion } = this.state;

    const indexTipoContrato = listaTiposContrato.findIndex(tc => tc.uniIderegistro == tipoContrato);
    const tipoContratoObj = (indexTipoContrato >= 0) ? listaTiposContrato[indexTipoContrato] : null;
    let tipoContratoProp = (tipoContratoObj) ? getProp(tipoContratoObj, 'uniPropiedad', null) : null;
    tipoContratoProp = (tipoContratoProp) ? JSON.parse(tipoContratoProp).tipocontrato : null;

    const indexTipoNegociacion = listaTiposNegociacion.findIndex(tn => tn.uniIderegistro == tipoNegociacion);
    const tipoNegociacionObj = (indexTipoNegociacion >= 0) ? listaTiposNegociacion[indexTipoNegociacion] : null;
    let tipoNegociacionProp = (tipoNegociacionObj) ? getProp(tipoNegociacionObj, 'uniPropiedad', null) : null;
    tipoNegociacionProp = (tipoNegociacionProp) ? JSON.parse(tipoNegociacionProp).clasecontrato : null;


    let obj = {
      fecha: fechaNominacion,
      tipoContrato: tipoContratoProp
    };

    if (negociacion === 'S') {
      obj.tipoNegociacion = tipoNegociacionProp;
    }

    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_RENOMINACION.CONSULTAR_PUNTOS_CONSUMO, obj)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.setState({ listaPuntosConsumo: JSON.parse(respuesta.data.datos) });
        }
      });
  };

  /**
   * Consulta la reserva.
   * @returns {Boolean}
   */
  consultarReserva = () => {
    if (this.state.contratoSeleccionado == null) {
      return;
    }
    const { fechaNominacion, idContrato, tipoContrato } = this.state;
    let obj = {
      fecha: fechaNominacion,
      tipoContrato: tipoContrato,
    };
    //Si es de tipo transporte se le envia el id del contrato...
    if (this.obtenerTipoContrato() === 'T') {
      obj.idContrato = this.state.contratoSeleccionado.cntIderegistro;
    }
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_RENOMINACION.CONSULTAR_RESERVA, obj)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          if (this.obtenerTipoContrato() == 'T') {
            this.setState({
              precio: getProp(respuesta.data.datos[0], 'precio', ''),
              unidadMedida: getProp(respuesta.data.datos[0], 'idemedida.uniIderegistro', '')
            });
          }
          this.setState({ listaPreciosRecerva: respuesta.data.datos });
        }
      });
  };

  consultarTipoNegociacion = () => {
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_RENOMINACION.CONSULTAR_TIPOS_NEGOCIACION, {

  /**
   * Métpodo encargado de obtener el tipo de contrato
   * @returns {String}
   */
  obtenerTipoContrato = () => {
    const { tipoContrato, listaTiposContrato } = this.state;
    const indexTipoContrato = listaTiposContrato.findIndex(tc => tc.uniIderegistro == tipoContrato);
    const tipoContratoObj = (indexTipoContrato >= 0) ? listaTiposContrato[indexTipoContrato] : null;
    let tipoContratoProp = (tipoContratoObj) ? getProp(tipoContratoObj, 'uniPropiedad', null) : null;
    tipoContratoProp = (tipoContratoProp) ? JSON.parse(tipoContratoProp).tipocontrato : null;
    return tipoContratoProp;

  /**
   * Método encargado de ejecutar el calculo de los excedentes
   */
  calcularExcedentes = () => {
    const tipoContrato = this.obtenerTipoContrato();
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_RENOMINACION.CALCULAR_EXCEDENTES, {
      fecha: this.state.fechaNominacion,
      tipoContrato: tipoContrato
    }).then(respuesta => {
      if (respuesta.data.codigo > 0) {
        this.setState({ listaCalculosExcedentes: respuesta.data.datos });
      }
    });
  };

  /**
   * Método encargado de objeter el objeto guardar
   * @returns {Array}
   */
  obtenerEntidadGuardar = () => {
    const { listaPuntosConsumo, listaCalculosExcedentes, tipoCalculo, fechaNominacion } = this.state;
    if (tipoCalculo == TIPOS_CALCULO.RENOMINACION) {
      const listaPuntos = listaPuntosConsumo.filter(punto => punto.seleccionado)
        .map(punto => {
          return {
            fecha: fechaNominacion,
            puntoConsumo: punto,
            valorReNominado: punto.valorRenominacion
          };
        });
      return listaPuntos;
    } else if (tipoCalculo == TIPOS_CALCULO.EXCEDENTES) {
      const listaPuntos = listaCalculosExcedentes.map(calculo => {
        return {
          fecha: fechaNominacion,
          puntoConsumo: calculo.puntoConsumo,
          excentes: calculo.excentes
        };
      });
      return listaPuntos;
    }
  };

  /**
   * Método encargado de guardar la entidad
   * @returns {Boolean}
   */
  guardarEntidad = () => {
    const validacion = this.validarFormulario();
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return false;
    }

    const obj = this.obtenerEntidadGuardar();

    // Reemplazar con ruta del Endpoint para guardar
    let url = RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_RENOMINACION.GUARDAR_RENOMINACION;
    if (this.state.tipoCalculo == TIPOS_CALCULO.EXCEDENTES) {
      url = RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_RENOMINACION.GUARDAR_EXCEDENTES;
    }
    axios.post(url, obj)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.limpiarFormulario();
        }
      });
  };

  /**

  consultarEntidad = () => {
    this.setState({ mostrarModalConsulta: true });

  /**
   * Método encargado de controlar el cambio de los componentes
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    this.setState(change, this.verificarConsultaPuntoConsumo);
  };

  /**
   * Método encargado de verificar la consulta de puntos de consumo
   * @returns {Boolean}
   */
  verificarConsultaPuntoConsumo = () => {
    const { tipoCalculo, tipoNegociacion, negociacion, fechaNominacion, tipoContrato } = this.state;
    if (!fechaNominacion || typeof fechaNominacion != 'string' || fechaNominacion.trim() == '') {
      return;
    }
    if (!tipoContrato || tipoContrato < 0) {
      return;
    }
    if (tipoCalculo == TIPOS_CALCULO.PRECIO && this.obtenerTipoContrato() === 'S') {
      this.ejecutarConsulta();
      return;
    }
    if ((tipoCalculo == TIPOS_CALCULO.PRECIO) && (this.obtenerTipoContrato() == 'T') && (this.state.contratoSeleccionado != null)) {
      this.ejecutarConsulta();
      return;
    }
    if (tipoCalculo == TIPOS_CALCULO.EXCEDENTES) {
      this.ejecutarConsulta();
      return;
    }
    if (negociacion === 'N') {
      this.ejecutarConsulta();
      return;
    }
    if (!tipoNegociacion || tipoNegociacion.trim() === '') {
      return;
    }
    if (!negociacion || negociacion.trim() === '') {
      return;
    }
    this.ejecutarConsulta();
  }

  /**
   * Método encargado de controlar los cambios hechos en la tabla
   * @param {Event} evento Evento ejecutado en el control de usuario
   * @returns {Boolean}
   */
  controlarCambioTabla = (evento) => {
    const control = evento.target;
    const idPunto = (control.attributes['data-punto']) ? control.attributes['data-punto'].value : null;
    if (idPunto) {
      const { listaPuntosConsumo } = this.state;
      const indexPunto = listaPuntosConsumo.findIndex(punto => punto.ptcIderegistro == idPunto);
      const nuevaListaPuntos = [...listaPuntosConsumo];
      nuevaListaPuntos[indexPunto][control.name] = control.value;
      this.setState({ listaPuntosConsumo: nuevaListaPuntos });
      return;
    }
    const idCalculo = (control.attributes['data-idcalculo']) ? control.attributes['data-idcalculo'].value : null;
    if (idCalculo) {
      const { listaCalculosExcedentes } = this.state;
      const indexPunto = listaCalculosExcedentes.findIndex(calculo => calculo.puntoConsumo.ptcIderegistro == idCalculo);
      const nuevaListaPuntos = [...listaCalculosExcedentes];
      nuevaListaPuntos[indexPunto][control.name] = control.value;
      this.setState({ listaCalculosExcedentes: nuevaListaPuntos });
    }
  };

  /**
   * Renderiza la tabla puntos consumo...
   * @returns {JSX}
   */

            <th>Tipo Venta</th>

        </thead>

            <th>Punto de Consumo</th>

    const value = evento.target.value;

  /**
 * Obtiene la fecha en string
 * @return {string}
 */
  convertirFecha = (timestamp) => {
    if (timestamp == null) {
      return 'Indefinido';
    }
    const fecha = new Date(timestamp);
    const anio = fecha.getFullYear();
    const dia = fecha.getDate();
    const mes = fecha.getMonth() + 1;
    return `${((dia < 9) ? '0' : '')}${dia}/${(mes < 9) ? '0' : ''}${mes}/${anio}`;
  };

  /**
   * Render Precios Recerva Suministro.
   * @returns {JSX}
   */
  renderPreciosRecerva = () => {
    if (!Util.validarArreglo(this.state.listaPreciosRecerva)) {
      return null;
    }
    return (
      <table className='table table-condensed table-striped table-bordered table-hover'>
        <thead className='bg-dark text-white'>
          <tr>
            <th>Productor</th>
            <th>Fecha a Nominar</th>
            <th>$ medidor reg</th>
            <th>Cantidad Excedente Medidor Reg</th>
            <th>$ medidor gnv</th>
            <th>$ medidor ind</th>
            <th>Cantidad excedente medidor ind</th>
            <th>Cálculo Precio</th>
          </tr>
        </thead>
        <tbody>
          {
            Util.validarArreglo(this.state.listaPreciosRecerva) && this.state.listaPreciosRecerva.map(
              (precio, index) => {
                return (
                  <tr key={index + '_ps'}>
                    <td>{getProp(precio, 'nombreProductor', 'Indefinido')}</td>
                    <td>{this.convertirFecha(precio.fechaNominar)}</td>
                    <td>{getProp(precio, 'medidorReg', 0)}</td>
                    <td>{getProp(precio, 'excdMedidorReg', 0)}</td>
                    <td>{getProp(precio, 'medidorGNV', 0)}</td>
                    <td>{getProp(precio, 'medidorInd', 0)}</td>
                    <td>{getProp(precio, 'excdMedidorInd', 0)}</td>
                    <td>{getProp(precio, 'calculoPrecio', 0)}</td>
                  </tr>
                );
              }
            )
          }
        </tbody>
      </table>
    )
  };

  /**
   * Método encargado de mostrar el formulario de calculo de la renominación
   * @returns {JSX}
   */
  renderCalculoRenominacion = () => {
    return (
      <Fragment>
        <div className='col-md-12'>
          <h2>Cálculo Renominación Transporte, ATR y Suministro</h2>
        </div>
        <Combo
          opciones={this.state.listaOpcionesNegociacion}
          propTexto='texto'
          propValor='valor'
          label='Negociación:'
          name='negociacion'
          value={this.state.negociacion}
          onChange={this.controlarCambio}
        />
        {this.state.negociacion === 'S' && (
          <Combo
            opciones={this.state.listaTiposNegociacion}
            propTexto='uniNombre1'
            propValor='uniIderegistro'
            label='Tipos Negociación:'
            name='tipoNegociacion'
            value={this.state.tipoNegociacion}
            onChange={this.controlarCambio}
          />
        )}
        <SelectorMultiple
          lista={this.state.listaPuntosConsumo}
          propValor='ptcIderegistro'
          propTexto='ptcoNombre'
          titulo='Puntos Consumo:'
          seleccionarItem={this.seleccionarPunto}
          cols={4}
        />
      </Fragment>
    );
  }

  /**
   * Método encargado de mostrar el selector de contratos
   * @returns {JSX}
   */
  renderSelectorContratos = () => {
    return (
      <Fragment>
        <div className='form-group col-4'>
          <label>Contrato:</label>
          <div className='input-group'>
            <input type='text' className='form-control' placeholder='Seleccionar Contrato' disabled={true} value={(this.state.contratoSeleccionado) ? this.state.contratoSeleccionado.cntNumero : 'Ningun contrato seleccionado'} />
            <div className='input-group-btn'>
              <button className='btn btn-primary' onClick={() => { this.setState({ mostrarModalConsulta: true }) }}><i className='fa fa-fw fa-check-square-o'></i></button>
            </div>
          </div>
        </div>
        <VentanaModal
          mostrar={this.state.mostrarModalConsulta}
          titulo={'Consultar Contratos'}
          cerrarModal={this.abrirCerrarModal}>
          <RConsultaContratos
            esModal
            seleccionMultiple={false}
            seleccionarEntidad={this.seleccionarContrato}
            tiposContrato={[this.obtenerTipoContrato()]}
            autoconsultar={true}
            ocultarFiltros={true}
          />
        </VentanaModal>
      </Fragment>
    );
  }

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <Fragment>
        <div className='d-flex justify-content-center'>
          <Botonera funciones={this.obtenerFunciones()} />
        </div>

        <div className='conf-general row mt-5'>
          <Combo
            opciones={this.state.listaTiposCalculo}
            propTexto='texto'
            propValor='valor'
            label='Tipo Cálculo:'
            name='tipoCalculo'
            value={this.state.tipoCalculo}
            onChange={this.controlarCambio}
          />
          <Fecha
            label='Fecha a nominar:'
            name='fechaNominacion'
            fecha={this.state.fechaNominacion}
            onChange={this.controlarCambio}
          />
          <Combo
            opciones={this.state.listaTiposContrato}
            propTexto='uniNombre1'
            propValor='uniIderegistro'
            label='Tipo Contrato:'
            name='tipoContrato'
            value={this.state.tipoContrato}
            onChange={this.controlarCambio}
          />
          {
            this.state.tipoCalculo == TIPOS_CALCULO.RENOMINACION && (this.renderCalculoRenominacion())
          }
          {this.state.tipoCalculo === TIPOS_CALCULO.RENOMINACION && (
            <div className='col-12'>
              {this.renderTablaPuntosConsumo()}
            </div>
          )}
          {
            this.state.tipoCalculo === TIPOS_CALCULO.EXCEDENTES && Util.validarArreglo(this.state.listaCalculosExcedentes) && (
              <div className='col-12'>
                <h2>Cálculo Excedentes</h2>
                {this.renderCalculoExcedentes()}
              </div>
            )
          }
          {
            this.state.tipoCalculo === TIPOS_CALCULO.PRECIO && (
              <Fragment>
                {
                  this.obtenerTipoContrato() === 'S' && (
                    <div className='col-12'>
                      <h2>Precio de Reserva Suministro</h2>
                      {this.renderPreciosRecerva()}
                    </div>
                  )
                }
                {
                  this.obtenerTipoContrato() === 'T' && (
                    <div className='col-12 row'>
                      <div className='col-12'>
                        <h2>Precio de Reserva Transporte</h2>
                      </div>
                      {this.renderSelectorContratos()}
                      <TextoNumerico
                        aceptaDecimales={false}
                        aceptaNegativos={false}
                        label='Precio:'
                        cols={4}
                        value={this.state.precio}
                        onChange={this.controlarCambio}
                        name='precio'
                        extra={{ disabled: true }}
                      />
                      <Combo
                        opciones={this.state.listaUnidadesMedida}
                        propTexto='texto'
                        propValor='valor'
                        label='Unidades Medida:'
                        name='unidadesMedida'
                        value={this.state.unidadesMedida}
                        onChange={this.controlarCambio}
                        extra={{ disabled: true }}
                      />
                      <div className='col-12'>
                        {this.renderPreciosRecerva()}
                      </div>
                    </div>
                  )
                }
                <div className='col-md-3 form-group'>
                  <button className='btn btn-primary' onClick={this.verificarConsultaPuntoConsumo}><i className='fa fa-fw fa-calculator'></i> Calcular</button>
                </div>
              </Fragment>
            )
          }
        </div>
      </Fragment>
    );
  }
}

GestionCalculoRenominacion.propTypes = {
  history: PropTypes.object,
  mostrarAlerta: PropTypes.func
};

    mostrarAlerta,
  }, dispatch);
