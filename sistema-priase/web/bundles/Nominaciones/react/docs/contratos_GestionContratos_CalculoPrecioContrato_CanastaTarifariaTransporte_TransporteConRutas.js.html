import React, { Component } from 'react';
import PropTypes from 'prop-types';
import { ComboHoras } from '../Componentes/ComboHoras';
import { Util, TextoNumerico, Combo, Input } from 'appfuture-react';
import Modal from 'react-bootstrap4-modal';

import './TransporteConRutas.scss';

class TransporteConRutas extends Component {

  state = {
    mostrarRutas: false,
    mostrarTramos: false,
    horaFin: '',
    ultimaHora: '00:00'
  };

  /**
   * Método encargado de controlar el cambio de los componentes
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  actualizarState = (evento) => {
    let cambio = {};
    const target = evento.target;
    cambio[target.name] = target.value;
    this.setState(cambio);
  };

   /**
   * Método encargado de actualizar el objeto redux
   * @param {Event} cambio Cambio a realizar
   */
  actualizarEstadoRedux = (cambio) => {
    if (this.props.intervalos) {
      this.props.actualizarTipoCalculoContrato({
        canastaConsumoTransporte: { ...this.props.canastaConsumoTransporte, ...cambio }
      });
      return;
    }
    this.props.actualizarTipoCalculoContrato({
      canastaHorariaTransporte: { ...this.props.canastaHorariaTransporte, ...cambio }
    });
  };

  /**
   * Método encargado de obtener las rutas seleccionadas
   * @returns {Array}
   */
  obtenerRutasSeleccionadas = () => {
    return this.props.rutasGNC.filter(r => (r.seleccionado));
  };

  /**
   * Método encargado de obtener los tramos seleccionados
   * @returns {Array}
   */
  obtenerTramosSeleccionados = () => {
    return this.props.listaTramos.filter(t => (t.seleccionado));
  };

  /**
   * Método encargado de mostrar el buscador de rutas
   * @returns {JSX}
   */
  renderBuscadorRutas = () => {
    const rutasSeleccionadas = this.obtenerRutasSeleccionadas().length;
    const textoRutas = '(' + rutasSeleccionadas + ') seleccionadas';
    return (
      <div className='col-3 form-group'>
        <label>Rutas:</label>
        <div className="input-group mb-3">
          <input value={textoRutas} disabled={true} type="text" className="form-control" readOnly />
          <div className="input-group-prepend">
            <button
              className="btn-primary btn-buscador"
              title='Seleccionar Rutas'
              onClick={() => this.setState({ mostrarRutas: true })}><i className="fa fa-fw fa-check-square-o"></i></button>
          </div>
        </div>
      </div>
    );
  };

   /**
   * Método encargado de mostrar el buscador de tramos
   * @returns {JSX}
   */
  renderBuscadorTramos = () => {
    const tramosSeleccionados = this.obtenerTramosSeleccionados().length;
    const textoTramos = '(' + tramosSeleccionados + ') seleccionadas';
    return (
      <div className='col-3 form-group'>
        <label>Tramos:</label>
        <div className="input-group mb-3">
          <input value={textoTramos} disabled={true} type="text" className="form-control" readOnly />
          <div className="input-group-prepend">
            <button
              className="btn-primary btn-buscador"
              title='Seleccionar Tramos'
              onClick={() => this.setState({ mostrarTramos: true })}><i className="fa fa-fw fa-check-square-o"></i></button>
          </div>
        </div>
      </div>
    );
  };

  /**
   * Método encargado de mostrar el modal de rutas
   * @returns {JSX}
   */
  renderModalRutas = () => {
    return (
      <Modal visible={this.state.mostrarRutas}>
        <div className="modal-header">
          <h4 className="modal-title"><b>Rutas</b></h4>
        </div>
        <div className="modal-body">
          <div>
            <p>Seleccione las Rutas</p>
            {
              this.props.rutasGNC.map(r => {
                return (
                  <div key={`ruta_${r.uniIderegistro}`}>
                    <label>
                      <input type="checkbox" value={r.uniIderegistro} checked={r.seleccionado || false} onChange={this.seleccionarRuta} />
                      <span> {r.uniNombre1}</span>
                    </label>
                  </div>
                );
              })
            }
          </div>
        </div>
        <div className="modal-footer">
          <button className='btn btn-primary' onClick={() => { this.setState({ mostrarRutas: false }) }}>Aceptar</button>
        </div>
      </Modal>
    );
  };

  /**
   * Método encargado de mostrar el modal de tramos
   * @returns {JSX}
   */
  renderModalTramos = () => {
    return (
      <Modal visible={this.state.mostrarTramos}>
        <div className="modal-header">
          <h4 className="modal-title"><b>Tramos</b></h4>
        </div>
        <div className="modal-body">
          <div>
            <p>Seleccione los Tramos</p>
            {
              this.props.listaTramos.map(t => {
                return (
                  <div key={`tramo_${t.trmIderegistro}`}>
                    <label>
                      <input type="checkbox" value={t.trmIderegistro} checked={t.seleccionado || false} onChange={this.seleccionarTramo} />
                      <span> {t.trmNombre}</span>
                    </label>
                  </div>
                );
              })
            }
          </div>
        </div>
        <div className="modal-footer">
          <button className='btn btn-primary' onClick={() => { this.setState({ mostrarTramos: false }) }}>Aceptar</button>
        </div>
      </Modal>
    );
  };

  /**
   * Método encargado de controlar el cambio en el componente selector de rutas
   * @param {Event} event Evento ejecutado en el control de usuario
   */
  seleccionarRuta = (event) => {
    const rutas = [...this.props.rutasGNC];
    const idRuta = parseInt(event.target.value);
    const index = rutas.findIndex(t => t.uniIderegistro === idRuta);
    rutas[index].seleccionado = event.target.checked;
    this.props.actualizarListaContratos({ rutasGNC: [...rutas] });
    this.actualizarRutasEnRangos();
  };

  /**
   * Método encargado de realizar la seleccion de los tramos
   * @param {Array} event Lista de tramos
   */
  seleccionarTramo = (evento) => {
    const tramos = [...this.props.listaTramos];
    const idTramo = parseInt(evento.target.value);
    const index = tramos.findIndex(t => t.trmIderegistro === idTramo);
    const listaCargos = tramos[index].listaCargos;
    if (!Array.isArray(listaCargos) || listaCargos.length == 0) {
      this.props.mostrarAlerta('No hay cargos', 'Lo sentimos el tramo seleccionado no tiene cargos configurados.');
      return;
    }
    tramos[index].seleccionado = evento.target.checked;
    this.props.actualizarListaContratos({ listaTramos: [...tramos] });
  };

  /**
   * Método encargado de actualizar las rutas en los rangos en el objeto redux
   * @returns {Boolean}
   */
  actualizarRutasEnRangos = () => {
    const { canastaHorariaTransporte } = this.props;
    if (!canastaHorariaTransporte || !Util.validarArreglo(canastaHorariaTransporte.canastasConRutas)) {
      return;
    }

    const rutasSeleccionadas = [...this.obtenerRutasSeleccionadas()];
    let nuevaCanasta = [...canastaHorariaTransporte.canastasConRutas];

    nuevaCanasta.forEach(canasta => {

      if (!Util.validarArreglo(canasta.rutas)) {
        return;
      }

      let nuevasRutas = [...canasta.rutas];

      // Validar que las rutas del rango coincidan con las rutas seleccionadas
      rutasSeleccionadas.forEach(ruta => {

        // Si hay más rutas seleccionadas que rutas en el rango, se agregan las nuevas rutas
        if (rutasSeleccionadas.length > nuevasRutas.length) {
          const rutaExiste = nuevasRutas.findIndex(rr => rr.uniIderegistro === ruta.uniIderegistro) > -1;
          if (!rutaExiste) {
            nuevasRutas.push({ ...ruta, valor: 0, unidadMedida: -1, porcentaje: 0, idRutaRango: Util.generarIdControl('rangoRuta') });
          }
        }

        // Si el rango tiene más rutas de las que están seleccionadas, remover la ruta sobrante
        if (nuevasRutas.length > rutasSeleccionadas.length) {
          nuevasRutas = nuevasRutas.filter(rangoRuta => rutasSeleccionadas.findIndex(r => r.uniIderegistro === rangoRuta.uniIderegistro) > -1);
        }

      });
      canasta.rutas = nuevasRutas;
    });


    this.actualizarEstadoRedux({ canastasConRutas: nuevaCanasta });
  };

  /**
   * Método encargado de agregar el rango parametrizado a la tabla
   * @returns {Boolean}
   */
  agregarRango = () => {
    // Validar si hay rutas y la hora es valida
    const validacion = this.validarFormulario();
    if (!validacion.respuesta) {
      const { titulo, mensaje } = validacion.mensaje;
      this.props.mostrarAlerta(titulo, mensaje);
      return;
    }

    if (this.props.intervalos) {
      const { canastaConsumoTransporte } = this.props;
      if (Util.validarArreglo(canastaConsumoTransporte.canastasConRutas)) {
        rangos = [...canastaConsumoTransporte.canastasConRutas];
      }
      return;
    }

    const { canastaHorariaTransporte } = this.props;
    const { horaFin, ultimaHora } = this.state;
    let rangos = [];

    if (Util.validarArreglo(canastaHorariaTransporte.canastasConRutas)) {
      rangos = [...canastaHorariaTransporte.canastasConRutas];
    }

    // Determina cuál es la hora de inicio del rango que se va a crear
    const horaInicio = (ultimaHora !== '00:00')
      ? this.convertirValorAHora(parseInt(ultimaHora) - 1).substring(0, 4) + '1'
      : ultimaHora;

    const nuevoRango = {
      rango: {
        horaInicio: horaInicio,
        horaFin: horaFin
      },
      rutas: [...this.obtenerRutasConTramos()],
    };

    rangos.push({ idRangoConRuta: Util.generarIdControl('canastaConRuta_'), ...nuevoRango });
    this.setState({ ultimaHora: this.obtenerNuevaHoraFin(horaFin) });
    const rango = rangos[0];
    const valores = this.obtenerPrecioContrato(rango);
    const valorPorcentaje = valores.valorPorcentaje;
    this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, precioCanasta: valores.precioContrato, valorCargos: valores.valorCargos, porcentajeComercializacion: valorPorcentaje });
    this.actualizarEstadoRedux({ canastasConRutas: rangos });
  };

  /**
   * Método encargado de obtener las rutas que tienen tramos
   * @returns {Array}
   */
  obtenerRutasConTramos = () => {
    let rutas = [...this.configurarRangoPorRutas()];
    for (let i = 0; i < rutas.length; i++) {
      rutas[i].tramos = [...this.configurarRangoPorTramos()];
    }
    return rutas;
  };

  /**
   * Método encargado de obtener el precio del contrato dependiendo del rango
   * @param {Object} rango Datos del rango
   */
  obtenerPrecioContrato = (rango) => {
    let valores = {
      precioContrato: 0,
      valorCargos: {
        variable: 0,
        fijo: 0,
        aom: 0
      },
      valorPorcentaje: 0,
    };
    const ruta = rango.rutas[0];
    if (!Array.isArray(ruta.tramos) || ruta.tramos.length == 0) {
      return valores;
    }
    //Recorremos los tramos y sumamos los valores.
    const tramos = ruta.tramos;
    for (let i = 0; i < tramos.length; i++) {
      const tramo = tramos[i];
      const valorCargoVariable = this.obtenerValorCargo(tramo, 'V');
      const valorCargoFijo = this.obtenerValorCargo(tramo, 'F');
      const valorCargoAOM = this.obtenerValorCargo(tramo, 'A');
      valores.valorCargos.variable += valorCargoVariable;
      valores.valorCargos.fijo += valorCargoFijo;
      valores.valorCargos.aom += valorCargoAOM;
      console.log('TRAMITO', tramo);
      valores.valorPorcentaje += parseFloat(tramo.valorPorcentaje);
      valores.precioContrato += (valorCargoVariable + valorCargoFijo + valorCargoAOM);
    }
    return valores;
  };

  /**
   * Método encargado de configurar los rangos dependiendo de la ruta
   * @returns {Array}
   */
  configurarRangoPorRutas = () => {
    const rutas = [...this.obtenerRutasSeleccionadas()];
    return rutas.map(r => (
      { ...r, idRutaRango: Util.generarIdControl('rangoRuta') }
    ));
  };

  /**
   * Método encargado de configurar los rangos dependiendo del tramo
   * @returns {Array}
   */
  configurarRangoPorTramos = () => {
    const tramos = [...this.obtenerTramosSeleccionados()];
    return tramos.map(t => (
      { ...t, cargoVariable: 0, cargoFijo: 0, cargoAOM: 0, unidadMedida: -1, valorPorcentaje: 0, idTramoRango: Util.generarIdControl('rangoTramo') }
    ));
  };

  /**
   * Método encargado de obtener una nueva hora fin
   * @param {Number} horaFin Hora fin actual
   */
  obtenerNuevaHoraFin = (horaFin) => {
    let nuevaHoraFin = horaFin === '23:59' ? 25 : parseInt(horaFin) + 1;
    return nuevaHoraFin;
  };

  /**
   * Método encargado de pasar el valor seleccionado a hora
   * @param {String} valor Valor seleccionado
   */
  convertirValorAHora = (valor) => {
    valor = parseInt(valor);
    if (valor.toString().length === 1) {
      return `0${valor}:00`;
    }
    return `${valor}:00`;
  };

  /**
   * Método encargado de validar los campos del formulario
   * @returns {Object}
   */
  validarFormulario = () => {
    // Valida que haya rutas seleccionadas.
    if (!Util.validarArreglo(this.obtenerRutasSeleccionadas())) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar al menos una ruta.' } };
    }

    //Validamos que se haya seleccionado un rango..
    const { horaFin } = this.state;
    if (horaFin === '' || horaFin === '-1') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar una hora fin.' } };
    }

    //Valida que haya tramos seleccioandos.
    if (!Util.validarArreglo(this.obtenerTramosSeleccionados())) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar al menos un tramo.' } };
    }

    return { respuesta: true };
  };

  /**
   * Método encargado de controlar el cambio en los rangos del tramo
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambioRangoTramo = (evento) => {
    // Controlar cambio
    const control = evento.target;
    const valor = control.value;
    const nombrePropiedad = control.name;
    const idRango = control.attributes['data-rango'].value;
    const idRuta = control.attributes['data-ruta'].value;
    const idTramo = control.attributes['data-tramo'].value;
    const canastaTransporte = (!this.props.intervalos) ? this.props.canastaHorariaTransporte : this.props.canastaConsumoTransporte;
    const nuevosRangos = canastaTransporte.canastasConRutas.map((r) => {
      const rango = { ...r };
      //Buscamos el rango...
      if (rango.idRangoConRuta === idRango) {
        //Buscamos las rutas del rango...
        rango.rutas = rango.rutas.map((r) => {
          const ruta = { ...r };
          if (ruta.idRutaRango === idRuta) {
            //Buscamos los tramos de la ruta.
            ruta.tramos = ruta.tramos.map((t) => {
              const tramo = { ...t };
              if (tramo.idTramoRango === idTramo) {
                tramo[nombrePropiedad] = valor;
              }
              return tramo;
            });
          }
          return ruta;
        });
      }
      return rango;
    });

    if (nombrePropiedad === 'valorPorcentaje') {
      const rango = { ...nuevosRangos[0] };
      const valores = this.obtenerPrecioContrato(rango);
      const valorPorcentaje = valores.valorPorcentaje;
      this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, precioCanasta: valores.precioContrato, valorCargos: valores.valorCargos, porcentajeComercializacion: valorPorcentaje });
    }
    this.actualizarEstadoRedux({ canastasConRutas: nuevosRangos });
  };

  /**
   * Método encargado de eliminar el ultimo rango agregado
   */
  borrarUltimoRango = () => {
    // TODO: Confirmar si desea borrar el rango
    const canastaTransporte = (!this.props.intervalos) ? this.props.canastaHorariaTransporte : this.props.canastaConsumoTransporte;
    const rangos = canastaTransporte.canastasConRutas.slice(0, canastaTransporte.canastasConRutas.length - 1);

    if (Util.validarArreglo(rangos)) {
      const nuevaHoraFin = rangos[rangos.length - 1].rango.horaFin;
      this.setState({ ultimaHora: parseInt(nuevaHoraFin) + 1, horaFin: '' });
    } else {
      this.setState({ ultimaHora: '0', horaFin: '' });
    }
    this.actualizarEstadoRedux({ canastasConRutas: rangos });
  };

  /**
   * Obtiene el valor cargo de un tramo.
   * @param {Object} tramo Datos del tramo
   * @param {String} tipo Tipo de cargo
   * @return {Array}
   */
  obtenerValorCargo = (tramo, tipo) => {
    //Verificamos el cargo seleccionado...
    if (!Util.validarArreglo(tramo.listaCargos)) {
      return -1;
    }
    var cargoSeleccionado = tramo.listaCargos.find(cargo => cargo.seleccionado);
    switch (tipo) {
      case 'V':
        return cargoSeleccionado.trcaCargovariable;
      case 'F':
        return cargoSeleccionado.trcaCargofijo;
      case 'A':
        return cargoSeleccionado.trcaCargoaoym;
    }
  };

  /**
   * Método encargado de mostrar la tabla de rangos
   * @return {JSX}
   */
  renderTablaRangos = () => {
    // TODO: Si se agregan rutas después de que la tabla se ha renderizado, actualizar la tabla
    const canastaTransporte = (!this.props.intervalos) ? this.props.canastaHorariaTransporte : this.props.canastaConsumoTransporte;
    if (!canastaTransporte || !canastaTransporte.canastasConRutas || !Util.validarArreglo(canastaTransporte.canastasConRutas)) {
      return null;
    }
    // TODO: Agregar cabecera a esta tabla
    return (
      <div className='mt-5'>
        <div className="lst-canasta-rangos">
          <div className="lst-canasta-rangos__rango header">
            <div className="text-center horas-rango">Rango</div>
          </div>
          <div className="lst-canasta-rangos__rutas header no-border">
            <div className="lst-canasta-rangos__rutas__fila header">
              <div className="lst-canasta-rangos__rutas__fila__ruta header">Ruta</div>
              <div className="lst-canasta-rangos__rutas__fila__columna header">Tramo</div>
              <div className="lst-canasta-rangos__rutas__fila__columna header">C. Variable</div>
              <div className="lst-canasta-rangos__rutas__fila__columna header">Unidad Medida</div>
              <div className="lst-canasta-rangos__rutas__fila__columna header">C. Fijo</div>
              <div className="lst-canasta-rangos__rutas__fila__columna header">C. AO&M</div>
              {/* <div className="lst-canasta-rangos__rutas__fila__columna header">U.Medida</div> */}
              <div className="lst-canasta-rangos__rutas__fila__columna header">%</div>
            </div>
          </div>
        </div>
        {
          canastaTransporte.canastasConRutas.map((rango, index) => {
            return (
              <div className='lst-canasta-rangos' key={rango.idRangoConRuta}>
                <div className="lst-canasta-rangos__rango">
                  <div className='text-center horas-rango'>
                    {rango.rango.horaInicio + ' - ' + rango.rango.horaFin}
                  </div>
                  {
                    ((index === canastaTransporte.canastasConRutas.length - 1) && !this.props.intervalos) &&
                    <div className='text-center mt-3'>
                      <button className='btn btn-danger btn-eliminar-rango btn-sm' title="Eliminar" onClick={this.borrarUltimoRango}><i className="fa fa-fw fa-trash"></i></button>
                    </div>
                  }
                </div>
                <div className="lst-canasta-rangos__rutas">
                  {
                    rango.rutas.map(ruta => {
                      return (
                        <div key={rango.idRangoConRuta + ruta.idRutaRango}>
                          <div className="lst-canasta-rangos__rutas__fila no-padding">
                            <div className="lst-canasta-rangos__rutas__fila__ruta">
                              {ruta.uniNombre1}
                            </div>
                            <div className="lst-canasta-rangos__rutas__fila__tramo">
                              {
                                ruta.tramos.map(tramo => {
                                  return (
                                    <div key={tramo.idTramoRango} className='lst-canasta-rangos__rutas__item'>
                                      <div className='lst-canasta-rangos__rutas__item__columna'>
                                        {tramo.trmNombre}
                                      </div>
                                      <div className='lst-canasta-rangos__rutas__item__columna'>
                                        <Input
                                          label='Valor'
                                          cols={12}
                                          value={(tramo.cargoVariable) ? tramo.cargoVariable : this.obtenerValorCargo(tramo, 'V')}
                                          onChange={this.controlarCambioRangoTramo}
                                          name='cargoVariable'
                                          extra={{ 'data-rango': rango.idRangoConRuta, 'data-ruta': ruta.idRutaRango, 'data-tramo': tramo.idTramoRango }}
                                        />
                                      </div>
                                      <div className='lst-canasta-rangos__rutas__item__columna'>
                                        <Combo
                                          opciones={this.props.listaUnidadesMedida}
                                          propTexto='uniNombre1'
                                          propValor='uniIderegistro'
                                          name='unidadMedida'
                                          cols={12}
                                          value={tramo.unidadMedida}
                                          onChange={this.controlarCambioRangoTramo}
                                          extra={{ 'data-rango': rango.idRangoConRuta, 'data-ruta': ruta.idRutaRango, 'data-tramo': tramo.idTramoRango }}
                                        />
                                      </div>
                                      <div className='lst-canasta-rangos__rutas__item__columna'>
                                        <Input
                                          label='Valor'
                                          cols={12}
                                          value={this.obtenerValorCargo(tramo, 'F')}
                                          onChange={this.controlarCambioRangoTramo}
                                          name='cargoFijo'
                                          extra={{ 'data-rango': rango.idRangoConRuta, 'data-ruta': ruta.idRutaRango, 'data-tramo': tramo.idTramoRango, disabled: true }}
                                        />
                                      </div>
                                      <div className='lst-canasta-rangos__rutas__item__columna'>
                                        <Input
                                          label='Valor'
                                          cols={12}
                                          value={this.obtenerValorCargo(tramo, 'A')}
                                          onChange={this.controlarCambioRangoTramo}
                                          name='cargoAOM'
                                          extra={{ 'data-rango': rango.idRangoConRuta, 'data-ruta': ruta.idRutaRango, 'data-tramo': tramo.idTramoRango, disabled: true }}
                                        />
                                      </div>
                                      {/* <div className='lst-canasta-rangos__rutas__item__columna'>
                                        <Combo
                                          opciones={this.props.listaUnidadesMedida}

                                      <div className='lst-canasta-rangos__rutas__item__columna'>
                                        <TextoNumerico
                                          aceptaDecimales={false}
                                          aceptaNegativos={false}
                                          label='Valor'
                                          cols={12}
                                          value={tramo.valorPorcentaje}
                                          onChange={this.controlarCambioRangoTramo}
                                          name='valorPorcentaje'
                                          extra={{ 'data-rango': rango.idRangoConRuta, 'data-ruta': ruta.idRutaRango, 'data-tramo': tramo.idTramoRango }}

                          </div>
                        </div>
                      );
                    })
                  }
                </div>
              </div>
            );
          })
        }
      </div>
    );
  };

  /**
   * Método obtener los porcentajes dado un rango
   * @param {Object} rango Datos del rango
   * @returns {Object}
   */
  obtenerPorcentajeRango = (ruta) => {
    let porcentaje = ruta.tramos.reduce((c, r) => (c + parseFloat(r.valorPorcentaje)), 0);
    if (isNaN(porcentaje)) {
      porcentaje = 0;
    }
    let estilo = porcentaje !== 100 ? 'red' : 'green';
    let icono = porcentaje !== 100 ? <i className='fa fa-fw fa-warning'></i> : <i className='fa fa-fw fa-check'></i>;
    return (<span className="linea-alerta" style={{ color: estilo }}>{icono} Porcentaje de Comercialización del Rango: {porcentaje}%</span>);
  }

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    let deshabilitarComboHoras = (this.state.ultimaHora > 24);
    return (
      <div>
        <div className='row'>
          {!this.props.intervalos && this.renderBuscadorRutas()}
          {!this.props.intervalos && (
            <ComboHoras
              valor={this.state.horaFin}
              onChange={this.actualizarState}
              disabled={deshabilitarComboHoras}
              ultimaHora={this.state.ultimaHora}
            />
          )}
          {!this.props.intervalos && this.renderBuscadorTramos()}
          {!this.props.intervalos && (
            <div className="form-group col-3 m-t-24">
              <button className='ml-3 btn btn-primary' onClick={this.agregarRango}><i className="fa fa-fw fa-plus"></i> Agregar</button>
            </div>
          )}
        </div>
        <div>
          {this.renderTablaRangos()}
        </div>

        {!this.props.intervalos && this.renderModalRutas()}
        {!this.props.intervalos && this.renderModalTramos()}

      </div>
    );
  }
}

TransporteConRutas.propTypes = {

  actualizarCabeceraRedux: PropTypes.func,
