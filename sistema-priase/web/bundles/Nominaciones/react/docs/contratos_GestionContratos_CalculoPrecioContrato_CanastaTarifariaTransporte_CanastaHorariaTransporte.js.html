import React, { Component, Fragment } from 'react';
import { Input, Combo, Util } from 'appfuture-react';
import PropTypes from 'prop-types';
// Util
import { get as getProp } from 'object-path';
import { TransporteSinRutas } from './TransporteSinRutas';
import { TransporteConRutas } from './TransporteConRutas';

const rutasGNC = [
  { texto: 'Sí', valor: 'S' },
  { texto: 'No', valor: 'N' }
];

class CanastaHorariaTransporte extends Component {
  state = {
    tipoTransporte: '',
    horaFinIntervalo: '',
    usaRutasGNC: 'N',
    nombreUnidadMedida: ''
  };

  /**
   * Método encargado de mostrar el formulario sin rutas
   * @returns {Component}
   */
  renderFormularioSinRutas = () => {
    const { canastaHorariaTransporte, actualizarTipoCalculoContrato, listaUnidadesMedida, listaTramos, mostrarAlerta, rutasGNC } = this.props;
    return (
      <TransporteSinRutas
        canastaHorariaTransporte={canastaHorariaTransporte}
        actualizarTipoCalculoContrato={actualizarTipoCalculoContrato}
        listaUnidadesMedida={listaUnidadesMedida}
        listaTramos={listaTramos}
        mostrarAlerta={mostrarAlerta}
        rutasGNC={rutasGNC}
        actualizarCabeceraRedux={this.props.actualizarCabeceraRedux}
        actualizarListaContratos={this.props.actualizarListaContratos}
      />
    )
  };

  /**
   * Método encargado de mostrar el formulario con rutas
   * @returns {Component}
   */
  renderFormularioConRutas = () => {
    const { canastaHorariaTransporte, listaTramos, listaUnidadesMedida, actualizarTipoCalculoContrato, actualizarListaContratos, rutasGNC, mostrarAlerta } = this.props;
    return (
      <TransporteConRutas
        canastaHorariaTransporte={canastaHorariaTransporte}
        listaTramos={listaTramos}
        listaUnidadesMedida={listaUnidadesMedida}
        actualizarTipoCalculoContrato={actualizarTipoCalculoContrato}
        actualizarListaContratos={actualizarListaContratos}
        rutasGNC={rutasGNC}
        mostrarAlerta={mostrarAlerta}
        actualizarCabeceraRedux={this.props.actualizarCabeceraRedux}
      />
    );
  };

  /**
   * Método encargado validar que componente mostrar
   * @returns {Component}
   */
  renderFormularioPorRuta = () => {
    const usaRutasGNC = getProp(this.props, 'canastaHorariaTransporte.usaRutasGNC', 'N');
    if (usaRutasGNC === 'N') {
      return this.renderFormularioSinRutas();
    }
    return this.renderFormularioConRutas();
  };

  /**
   * Método encargado de controlar el cambio de los componentes
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambio = (evento) => {
    let cambio = {};
    const target = evento.target;
    cambio[target.name] = target.value;
    if (target.name === 'unidadMedida') {
      cambio['nombreUnidadMedida'] = target.text;
    }
    this.actualizarEstadoRedux(cambio);
  };

  /**
   * Método encargado de controlar el cambio del objeto redux
   * @param {Object} cambio Cambio a realizar
   */
  actualizarEstadoRedux = (cambio) => {
    this.props.actualizarTipoCalculoContrato({ canastaHorariaTransporte: { ...this.props.canastaHorariaTransporte, ...cambio } });
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <div className='canasta-horaria container'>
        <div className='mt-3 card box-shadow'>
          <div className='card-header'>Canasta Tarifa Horaria de Transporte</div>
          <div className='card-body'>
            <div>
              <h3 className='bold text-center'>Rangos Horarios para contrato Tipo transporte</h3>
            </div>
            {this.renderFormularioPorRuta()}
          </div>
        </div>
      </div>
    );
  }
}

CanastaHorariaTransporte.propTypes = {
  mostrarAlerta: PropTypes.func,
  listaUnidadesMedida: PropTypes.array,
  listaTramos: PropTypes.array,
  rutasGNC: PropTypes.array,
  actualizarTipoCalculoContrato: PropTypes.func,
  actualizarListaContratos: PropTypes.func,
  canastaHorariaTransporte: PropTypes.object,
  actualizarCabeceraRedux: PropTypes.func,
};

CanastaHorariaTransporte.defaultProps = {
  canastaHorariaTransporte: {}
};

export { CanastaHorariaTransporte };
