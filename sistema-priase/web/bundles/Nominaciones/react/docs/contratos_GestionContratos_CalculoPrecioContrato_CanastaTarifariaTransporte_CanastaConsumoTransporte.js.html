import React, { Component } from 'react';

import PropTypes from 'prop-types';
import { Input, TextoNumerico, Combo, Util } from 'appfuture-react';
import Modal from 'react-bootstrap4-modal';
import { SelectorTramos } from '../../../../utils/SelectorTramos';

// Util
import { get as getProp } from 'object-path';

import './CanastaHorariaTransporte.scss';
import { TransporteConRutas } from './TransporteConRutas';
import { TransporteSinRutas } from './TransporteSinRutas';

const usaRutasGNC = [
  { texto: 'Sí', valor: 'S' },
  { texto: 'No', valor: 'N' }
];

const tiposLiquidacion = [
  { texto: 'Volumen Marginal', valor: 'V' },
  { texto: 'Rango Total', valor: 'R' }
];

class CanastaConsumoTransporte extends Component {

  state = {
    horaFin: '',
    valor: '',
    unidadMedida: '',
    nombreUnidadMedida: '',
    porcentaje: '',
    idRangoSinRuta: null,
    idRangoConRuta: null,
    tramo: '',
    mostrarTramos: false,
    mostrarRutas: false,
  };

  /**
   * Método encargado de controlar el cambio de los componentes
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  actualizarState = (evento) => {
    let cambio = {};
    const target = evento.target;
    cambio[target.name] = target.value;
    this.setState(cambio);
  };

  /**
   * Método encargado de controlar el cambio de los componentes
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambio = (evento) => {
    let cambio = {};
    const target = evento.target;
    cambio[target.name] = target.value;
    if (target.name === 'unidadMedida') {
      cambio['nombreUnidadMedida'] = target.text;
    }

    this.actualizarEstadoRedux(cambio);
  };

  /**
   * Métodoe encargado de validar si usa rutas GNC
   * @param {String} nombrePropiedad Nombre de la propiedad
   * @param {String} valorPropiedad Valor de la propiedad
   */
  validarUsaRutasGNCValido = (nombrePropiedad, valorPropiedad) => {
    const tarifaHoraria = getProp(this.props, 'tipoCalculo.tarifaHoraria', false);
    const tarifaHorariaUsaRutasGNC = getProp(this.props, 'tipoCalculo.canastaHorariaTransporte.usaRutasGNC', 'N');
    const tarifaConsumoUsaRutasGNC = valorPropiedad;
    if (nombrePropiedad == 'usaRutasGNC' && tarifaHoraria && tarifaHorariaUsaRutasGNC != tarifaConsumoUsaRutasGNC) {
      return false;
    }
    return true;
  };

  /**
   * Método encargado de actualizar el objeto redux
   * @param {Event} cambio Cambio a realizar
   */
  actualizarEstadoRedux = (cambio) => {
    this.props.actualizarTipoCalculoContrato({ canastaConsumoTransporte: { ...this.props.canastaConsumoTransporte, ...cambio } });
  };

  /**
   * Método encargado de validar los campos del formulario
   * @returns {Object}
   */
  validarFormulario = () => {
    const { unidadMedida, valor, porcentaje, horaFin } = this.state;

    if (unidadMedida === '' || unidadMedida === '-1') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar una unidad de medida.' } };
    }

    if (valor.trim() === '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe ingresar el valor.' } };
    }

    if (this.validarNumero(porcentaje) && porcentaje > 100) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe ingresar un porcentaje válido menor o igual 100%.' } };
    }
    return { respuesta: true };
  };

  /**
   * Método encargado de validar que un valor sea un número
   * @param {Number} numero Valor del número
   * @returns {Boolean}
   */
  validarNumero = (numero) => {
    return !/^([0-9.])*$/.test(numero);
  };

  /**
   * Método encargado de mostrar el formulario con rutas
   * @returns {Component}
   */
  renderFormularioConRutas = () => {
    const { intervalos, canastasConRutas } = this.props.canastaConsumoTransporte;
    if (!intervalos || intervalos === 0 || !Util.validarArreglo(canastasConRutas)) {
      return null;
    }
    const { canastaConsumoTransporte, actualizarTipoCalculoContrato, actualizarListaContratos, listaUnidadesMedida, mostrarAlerta, listaTramos, rutasGNC, actualizarCabeceraRedux } = this.props;
    return (
      <TransporteConRutas
        canastaConsumoTransporte={canastaConsumoTransporte}
        actualizarTipoCalculoContrato={actualizarTipoCalculoContrato}
        actualizarListaContratos={actualizarListaContratos}
        actualizarCabeceraRedux={actualizarCabeceraRedux}
        listaUnidadesMedida={listaUnidadesMedida}
        mostrarAlerta={mostrarAlerta}
        listaTramos={listaTramos}
        rutasGNC={rutasGNC}
        intervalos={parseInt(intervalos)}
      />
    );
  };

  /**
   * Método encargado de mostrar el formulario sin rutas
   * @returns {Component}
   */
  renderFormularioSinRutas = () => {
    const { intervalos, canastasSinRutas } = this.props.canastaConsumoTransporte;
    if (!intervalos || intervalos === 0 || !Util.validarArreglo(canastasSinRutas)) {
      return null;
    }
    const { canastaConsumoTransporte, actualizarTipoCalculoContrato, actualizarListaContratos, listaUnidadesMedida, mostrarAlerta, listaTramos, rutasGNC, actualizarCabeceraRedux } = this.props;
    return (
      <TransporteSinRutas
        canastaConsumoTransporte={canastaConsumoTransporte}
        actualizarTipoCalculoContrato={actualizarTipoCalculoContrato}
        actualizarListaContratos={actualizarListaContratos}
        actualizarCabeceraRedux={actualizarCabeceraRedux}
        listaUnidadesMedida={listaUnidadesMedida}
        mostrarAlerta={mostrarAlerta}
        listaTramos={listaTramos}
        rutasGNC={rutasGNC}
        intervalos={parseInt(intervalos)}
      />
    );
  };

  /**
   * Método encargado de controlar el cambio de los rangos sin ruta
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambioRangoSinRuta = (evento) => {
    // Controlar cambio
    const control = evento.target;
    const valor = control.value;
    const nombrePropiedad = control.name;
    const idRango = control.attributes['data-rango'].value;
    const nuevosRangos = [...this.props.canastaConsumoTransporte.canastasSinRutas];

    const indexRango = nuevosRangos.findIndex(a => a.idIntervaloSinRuta === idRango);
    nuevosRangos[indexRango][nombrePropiedad] = valor;
    // PENDIENTE
    const rango = nuevosRangos[0];
    const valores = this.obtenerPrecioContratoPorTramos(rango);
    this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, valorCargos: valores.valorCargos });
    this.actualizarEstadoRedux({ canastasSinRutas: nuevosRangos });
  };

  /**
   * Obtiene el objeto tramo de la listaTramos por medio del idTramo (trmIderegistro)...
   * @param {Object} tramo Datos del tramo
   */
  obtenerTramo = (tramo) => {
    const idTramo = parseInt(tramo);
    const index = this.props.listaTramos.findIndex(t => t.trmIderegistro === idTramo);
    tramo = (index >= 0) ? this.props.listaTramos[index] : null;
    return tramo;
  };

  /**
   * Método encargado de validar los datos del tramo
   * @param {Object} tramo Datos del tramo
   */
  validarTramo = (tramo) => {
    tramo = this.obtenerTramo(tramo);
    if (!tramo) {
      this.props.mostrarAlerta('Atención', 'No se pudo obtener la información del tramo seleccionado.');
      return false;
    }

    if (!Array.isArray(tramo.listaCargos) || tramo.listaCargos.length === 0) {
      this.props.mostrarAlerta('Atención', 'El tramo seleccionado no tiene cargos configurados.');
      return false;
    }

    return true;
  };

  /**
   * Genera los intervalos...
   * @returns {Boolean}
   */
  generarIntervalos = () => {
    const { intervalos, usaRutasGNC, canastasSinRutas, canastasConRutas, tramo, tipoLiquidacion } = this.props.canastaConsumoTransporte;

    if (!tipoLiquidacion || tipoLiquidacion.trim() === '' || tipoLiquidacion === '-1') {
      this.props.mostrarAlerta('Atención', 'Debe seleccionar un tipo de liquidación.');
      return;
    }

    if (!this.props.cabecera.cantidadContratada) {
      this.props.mostrarAlerta('Atención', 'Debe seleccionar una cantidad contratada en la cabecera del contrato.');
      return;
    }

    if (usaRutasGNC === 'S') {
      if (this.obtenerRutasSeleccionadas().length === 0) {
        this.props.mostrarAlerta('Atención', 'Debe seleccionar una ruta como mínimo.');
        return;
      }
    }

    if (usaRutasGNC === 'S') {
      if (this.obtenerTramosSeleccionados().length === 0) {
        this.props.mostrarAlerta('Atención', 'Debe seleccionar una tramo como mínimo.');
        return;
      }
    }

    if (intervalos === '' || intervalos <= 0) {
      this.props.mostrarAlerta('Atención', 'Debe agregar una cantidad de intervalos mayor a 0');
    }

    if ((!usaRutasGNC && Util.validarArreglo(canastasSinRutas)) || (usaRutasGNC && Util.validarArreglo(canastasConRutas))) {
      // Mostrar mensaje indicando que se van a borrar los intervalos existentes
      const botones = [
        { texto: 'Cancelar', clase: 'btn-danger' },
        { texto: 'Aceptar', clase: 'btn-primary', callback: this.crearArregloIntervalos }
      ];
      this.props.mostrarAlerta('Atención', 'Se eliminarán los intervalos actuales, ¿Desea continuar?', botones);
      return;
    }

    this.crearArregloIntervalos();
  };

  /**
   * Obtiene el valor cargo de un tramo.
   * @param {Object} tramo Datos del tramo
   * @param {String} tipo Tipo de cargo
   * @return {Number}
   */
  obtenerValorCargo = (tramo, tipo) => {
    //Verificamos el cargo seleccionado...
    if (!Util.validarArreglo(tramo.listaCargos)) {
      return -1;
    }
    var cargoSeleccionado = tramo.listaCargos.find(cargo => cargo.seleccionado);
    switch (tipo) {
      case 'V':
        return cargoSeleccionado.trcaCargovariable;
      case 'F':
        return cargoSeleccionado.trcaCargofijo;
      case 'A':
        return cargoSeleccionado.trcaCargoaoym;
    }
  };

  /**
   * Método encargado de obtener las rutas que tienen tramos
   * @returns {Array}
   */
  obtenerRutasConTramos = () => {
    let rutas = this.obtenerRutasSeleccionadas();
    for (let i = 0; i < rutas.length; i++) {
      rutas[i].tramos = this.obtenerTramosSeleccionados();
    }
    return rutas;
  };

  /**
   * Crea el arreglo de intervalos que se mostrará en el panel...
   * @returns {Boolean}
   */
  crearArregloIntervalos = () => {
    const { intervalos, usaRutasGNC, tramo } = this.props.canastaConsumoTransporte;
    const { cantidadContratada } = this.props.cabecera;

    const cantidadRango = cantidadContratada / intervalos;

    if (usaRutasGNC === 'S') {
      let nuevaCanastaConRutas = [];
      let rangoInicial = 0;
      for (let i = cantidadRango; i <= cantidadContratada; i += cantidadRango) {
        nuevaCanastaConRutas.push({ idRangoConRuta: Util.generarIdControl('canastaConRuta_'), rango: { horaInicio: rangoInicial, horaFin: i }, rutas: this.obtenerRutasConTramos(), porcentaje: 0 });
        rangoInicial = i + 1;
      }
      const rango = nuevaCanastaConRutas[0];
      const valores = this.obtenerPrecioContratoPorTramos(rango);
      this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, valorCargos: valores.valorCargos });
      this.actualizarEstadoRedux({ canastasConRutas: nuevaCanastaConRutas });
      return;
    }

    let nuevaCanastaSinRutas = [];
    let rangoInicial = 0;
    for (let i = cantidadRango; i <= cantidadContratada; i += cantidadRango) {
      nuevaCanastaSinRutas.push({ idRangoConRuta: Util.generarIdControl('canastaConRuta_'), rango: { horaInicio: rangoInicial, horaFin: i }, tramos: this.obtenerTramosSeleccionados(), porcentaje: 0 });
      rangoInicial = i + 1;
    }
    const rango = nuevaCanastaSinRutas[0];
    const valores = this.obtenerPrecioContratoPorTramos(rango);
    this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, valorCargos: valores.valorCargos });
    this.actualizarEstadoRedux({ canastasSinRutas: nuevaCanastaSinRutas });

  };

  /**
   * Método encargado de obtener el precio del contrato dependiendo del rango
   * @param {Object} rango Datos del rango
   */
  obtenerPrecioContrato = (rango) => {
    if (!rango || !rango.tramo || !Array.isArray(rango.tramo.listaCargos) || rango.tramo.listaCargos.length === 0) {
      return 0;
    }
    const tramo = rango.tramo;
    let valor = 0;
    let valorCargoVariable = this.obtenerValorCargo(tramo, 'V');
    let valorCargoFijo = this.obtenerValorCargo(tramo, 'F');
    let valorCargoAOM = this.obtenerValorCargo(tramo, 'A');
    valor = (valorCargoVariable + valorCargoFijo + valorCargoAOM);
    return valor;
  };

  /**
   * Método encargado de obtener el precio del contrato dependiendo de los tramos
   * @param {Object} rango Datos del rango
   */
  obtenerPrecioContratoPorTramos = (rango) => {
    let valores = {
      precioContrato: 0,
      valorCargos: {
        variable: 0,
        fijo: 0,
        aom: 0,
      },
      valorPorcentaje: 0,
    };
    if (!rango.rutas) {
      //Obtenemos los valores de los tramos directamente...
      const tramos = rango.tramos;
      for (let i = 0; i < tramos.length; i++) {
        const tramo = tramos[i];
        const valorCargoVariable = this.obtenerValorCargo(tramo, 'V');
        const valorCargoFijo = this.obtenerValorCargo(tramo, 'F');
        const valorCargoAOM = this.obtenerValorCargo(tramo, 'A');
        valores.valorCargos.variable += valorCargoVariable;
        valores.valorCargos.fijo += valorCargoFijo;
        valores.valorCargos.aom += valorCargoAOM;
        valores.precioContrato += (valorCargoVariable + valorCargoFijo + valorCargoAOM);
        valores.valorPorcentaje += tramo.porcentaje;
      }
      return valores;
    }
    const ruta = rango.rutas[0];
    if (!Array.isArray(ruta.tramos) || ruta.tramos.length == 0) {
      return valores;
    }
    //Recorremos los tramos y sumamos los valores.
    const tramos = ruta.tramos;
    for (let i = 0; i < tramos.length; i++) {
      const tramo = tramos[i];
      const valorCargoVariable = this.obtenerValorCargo(tramo, 'V');
      const valorCargoFijo = this.obtenerValorCargo(tramo, 'F');
      const valorCargoAOM = this.obtenerValorCargo(tramo, 'A');
      valores.valorCargos.variable += valorCargoVariable;
      valores.valorCargos.fijo += valorCargoFijo;
      valores.valorCargos.aom += valorCargoAOM;
      valores.precioContrato += (valorCargoVariable + valorCargoFijo + valorCargoAOM);
      valores.valorPorcentaje += tramo.porcentaje;
    }
    return valores;
  };

  /**
   * Método encargado de validar que formulario mostrar
   * @returns {JSX}
   */
  renderTablaIntervalos = () => {
    const usaRutasGNC = getProp(this.props, 'canastaConsumoTransporte.usaRutasGNC', 'N');
    if (usaRutasGNC === 'N') {
      return this.renderFormularioSinRutas();
    }
    return this.renderFormularioConRutas();
  };

  /**
   * Método encargado de mostrar el buscador de tramos
   * @returns {JSX}
   */
  renderBuscadorTramos = () => {
    const tramosSeleccionados = this.obtenerTramosSeleccionados().length;
    const textoTramos = `(${tramosSeleccionados}) seleccionadas`;
    return (
      <div className='col-3 form-group'>
        <label>Tramos:</label>
        <div className="input-group mb-3">
          <input value={textoTramos} disabled={true} type="text" className="form-control" readOnly />
          <div className="input-group-prepend">
            <button
              className="btn-primary btn-buscador"
              title='Seleccionar Tramos'
              onClick={() => this.setState({ mostrarTramos: true })}><i className="fa fa-fw fa-check-square-o"></i></button>
          </div>
        </div>
      </div>
    );
  };

  /**
   * Método encargado de mostrar el modal de tramos
   * @returns {JSX}
   */
  renderModalTramos = () => {
    return (
      <Modal visible={this.state.mostrarTramos}>
        <div className="modal-header">
          <h4 className="modal-title"><b>Tramos</b></h4>
        </div>
        <div className="modal-body">
          <div>
            <p>Seleccione los Tramos</p>
            {
              this.props.listaTramos.map(t => {
                return (
                  <div key={`tramo_${t.trmIderegistro}`}>
                    <label>
                      <input type="checkbox" value={t.trmIderegistro} checked={t.seleccionado || false} onChange={this.seleccionarTramo} />
                      <span> {t.trmNombre}</span>
                    </label>
                  </div>
                );
              })
            }
          </div>
        </div>
        <div className="modal-footer">
          <button className='btn btn-primary' onClick={() => { this.setState({ mostrarTramos: false }) }}>Aceptar</button>
        </div>
      </Modal>
    );
  };

  /**
   * Método encargado de realizar la seleccion de los tramos
   * @param {Array} event Lista de tramos
   */
  seleccionarTramo = (evento) => {
    const tramos = [...this.props.listaTramos];
    const idTramo = parseInt(evento.target.value);
    const index = tramos.findIndex(t => t.trmIderegistro === idTramo);
    const listaCargos = tramos[index].listaCargos;
    if (!Array.isArray(listaCargos) || listaCargos.length == 0) {
      this.props.mostrarAlerta('No hay cargos', 'Lo sentimos el tramo seleccionado no tiene cargos configurados.');
      return;
    }
    tramos[index].seleccionado = evento.target.checked;
    this.props.actualizarListaContratos({ listaTramos: [...tramos] });
  };

  /**
   * Método encargado de obtener las rutas seleccionadas
   * @returns {Array}
   */
  obtenerRutasSeleccionadas = () => {
    const rutas = this.props.rutasGNC.filter((r) => {
      if (r.seleccionado) { r.idRutaRango = Util.generarIdControl('rangoRuta'); return r }
    });
    return [...rutas];
  };

  /**
   * Método encargado de obtener los tramos seleccionados
   * @returns {Array}
   */
  obtenerTramosSeleccionados = () => {
    return this.props.listaTramos.filter((t) => {
      if (t.seleccionado) { t.idTramoRango = Util.generarIdControl('rangoTramo'); return t; }
    });
  };

  /**
   * Método encargado de mostrar el buscador de rutas
   * @returns {JSX}
   */
  renderBuscadorRutas = () => {
    const rutasSeleccionadas = this.obtenerRutasSeleccionadas().length;
    const textoRutas = `(${rutasSeleccionadas}) seleccionadas`;
    return (
      <div className='col-3 form-group'>
        <label>Rutas:</label>
        <div className="input-group mb-3">
          <input value={textoRutas} disabled={true} type="text" className="form-control" readOnly />
          <div className="input-group-prepend">
            <button
              className="btn-primary btn-buscador"
              title='Seleccionar Rutas'
              onClick={() => this.setState({ mostrarRutas: true })}><i className="fa fa-fw fa-check-square-o"></i></button>
          </div>
        </div>
      </div>
    );
  };

  /**
   * Método encargado de controlar el cambio en el componente selector de rutas
   * @param {Event} event Evento ejecutado en el control de usuario
   */
  seleccionarRuta = (event) => {
    const rutas = [...this.props.rutasGNC];
    const idRuta = parseInt(event.target.value);
    const index = rutas.findIndex(t => t.uniIderegistro === idRuta);
    rutas[index].seleccionado = event.target.checked;
    this.props.actualizarListaContratos({ rutasGNC: [...rutas] });
  };

  /**
   * Método encargado de actualizar las rutas en los rangos en el objeto redux
   * @returns {Boolean}
   */
  actualizarRutasEnRangos = () => {
    // Si no hay rangos seleccionados, no hacer nada
    const { canastaConsumoTransporte } = this.props;
    if (!canastaConsumoTransporte || !Util.validarArreglo(canastaConsumoTransporte.canastasConRutas)) {
      return;
    }

    const rutasSeleccionadas = [...this.obtenerRutasSeleccionadas()];
    let nuevaCanasta = [...canastaConsumoTransporte.canastasConRutas];

    nuevaCanasta.forEach(canasta => {

      if (!Util.validarArreglo(canasta.rutas)) {
        return;
      }

      let nuevasRutas = [...canasta.rutas];

      // Validar que las rutas del rango coincidan con las rutas seleccionadas
      rutasSeleccionadas.forEach(ruta => {

        // Si hay más rutas seleccionadas que rutas en el rango, se agregan las nuevas rutas
        if (rutasSeleccionadas.length > nuevasRutas.length) {
          const rutaExiste = nuevasRutas.findIndex(rr => rr.uniIderegistro === ruta.uniIderegistro) > -1;
          if (!rutaExiste) {
            nuevasRutas.push({ ...ruta, valor: 0, unidadMedida: -1, porcentaje: 0, idRutaRango: Util.generarIdControl('rangoRuta') });
          }
        }

        // Si el rango tiene más rutas de las que están seleccionadas, remover la ruta sobrante
        if (nuevasRutas.length > rutasSeleccionadas.length) {
          nuevasRutas = nuevasRutas.filter(rangoRuta => rutasSeleccionadas.findIndex(r => r.uniIderegistro === rangoRuta.uniIderegistro) > -1);
        }

      });
      canasta.rutas = nuevasRutas;
    });


    this.actualizarEstadoRedux({ canastasConRutas: nuevaCanasta });
  };

  /**
   * Método encargado de mostrar el modal de rutas
   * @returns {JSX}
   */
  renderModalRutas = () => {
    return (
      <Modal visible={this.state.mostrarRutas}>
        <div className="modal-header">
          <h4 className="modal-title"><b>Rutas</b></h4>
        </div>
        <div className="modal-body">
          <div>
            <p>Seleccione las Rutas</p>
            {
              this.props.rutasGNC.map(r => {
                return (
                  <div key={`ruta_${r.uniIderegistro}`}>
                    <label>
                      <input type="checkbox" value={r.uniIderegistro} checked={r.seleccionado || false} onChange={this.seleccionarRuta} />
                      <span> {r.uniNombre1}</span>
                    </label>
                  </div>
                );
              })
            }
          </div>
        </div>
        <div className="modal-footer">
          <button className='btn btn-primary' onClick={() => { this.setState({ mostrarRutas: false }) }}>Aceptar</button>
        </div>
      </Modal>
    );
  };

  /**
   * Método encargado de realizar la seleccion de los tramos
   * @param {Array} event Lista de tramos
   */
  seleccionarTramo = (evento) => {
    const tramos = [...this.props.listaTramos];
    const idTramo = parseInt(evento.target.value);
    const index = tramos.findIndex(t => t.trmIderegistro === idTramo);
    const listaCargos = tramos[index].listaCargos;
    if (!Array.isArray(listaCargos) || listaCargos.length == 0) {
      this.props.mostrarAlerta('No hay cargos', 'Lo sentimos el tramo seleccionado no tiene cargos configurados.');
      return;
    }
    tramos[index].seleccionado = evento.target.checked;
    this.props.actualizarListaContratos({ listaTramos: [...tramos] });
    // this.actualizarTramosEnRangos();
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <div className='canasta-horaria container'>
        <div className='mt-3 card box-shadow'>
          <div className='card-header'>Canasta Tarifa Consumo</div>
          <div className='card-body'>
            <div>
              <h3 className='font-weight-bold text-center'>Rangos de Consumo para contrato Tipo Trasporte</h3>
            </div>
            <div className='row mt-2'>
              <Combo
                opciones={tiposLiquidacion}
                propTexto='texto'
                propValor='valor'
                label='Tipo de Liquidación:'
                name='tipoLiquidacion'
                value={getProp(this.props, 'canastaConsumoTransporte.tipoLiquidacion', '-1')}
                onChange={this.controlarCambio}
                cols={3}
              />
              {
                (getProp(this.props, 'canastaConsumoTransporte.usaRutasGNC', 'N') === 'S') && (
                  this.renderBuscadorRutas()
                )
              }
              <SelectorTramos
                titulo='Tramos:'
                propTexto='trmNombre'
                propValor='trmIderegistro'
                seleccionarItem={this.seleccionarTramo}
                lista={getProp(this.props, 'listaTramos', [])}
              />
              <TextoNumerico
                aceptaDecimales={false}
                aceptaNegativos={false}
                label='No. Intervalos:'
                cols={2}
                value={getProp(this.props, 'canastaConsumoTransporte.intervalos', '0')}
                onChange={this.controlarCambio}
                name='intervalos'
              />
              <Input
                label='Cant. Contratada:'
                value={this.props.cabecera.cantidadContratada || '0'}
                name='cantidadContratada'
                extra={{ disabled: true, readOnly: true }}
                disabled={true}
                cols={2}
              />
              <div>
                <button className="btn btn-primary mt-4" onClick={this.generarIntervalos}>Generar Intervalos</button>
              </div>
            </div>
            {this.renderTablaIntervalos()}
          </div>
        </div>
        {this.renderModalTramos()}
        {this.renderModalRutas()}
      </div>
    );
  }
}

CanastaConsumoTransporte.propTypes = {
  mostrarAlerta: PropTypes.func,
  listaUnidadesMedida: PropTypes.array,
  listaTramos: PropTypes.array,
  rutasGNC: PropTypes.array,
  actualizarTipoCalculoContrato: PropTypes.func,
  actualizarListaContratos: PropTypes.func,
  canastaConsumoTransporte: PropTypes.object,
  cabecera: PropTypes.object,
  actualizarCabeceraRedux: PropTypes.func,
};

CanastaConsumoTransporte.defaultProps = {
  canastaConsumoTransporte: {}
};

export { CanastaConsumoTransporte };
