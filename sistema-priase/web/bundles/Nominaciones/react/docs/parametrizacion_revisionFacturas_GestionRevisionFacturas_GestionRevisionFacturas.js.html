import { Input, Botonera, Combo, Tabla, VentanaModal, Util, TextoNumerico, Fecha } from 'appfuture-react';

import RUTAS_VISTA from '../../../../global/rutas_vista';
import { CLASES_UNIDADES } from '../../../../global/constantes';
import { mostrarAlerta } from '../../../../store/actions/AplicacionAcciones';
import { limpiarHistorico } from '../../../../global/util_nominaciones';
import './GestionRevisionFacturas.scss';

const listaProcesos = [
  { texto: 'Validación', id: 'V' },
  { texto: 'Disperción', id: 'DP' }
];

const listaPeriodo = [
  { texto: 'Día Anterior', id: 'D' },
  { texto: 'Trimestre', id: 'T' }
];

const listaEDS = [
  { texto: 'Propias', id: 'S' },
  { texto: 'Terceros', id: 'N' }

class GestionRevisionFacturas extends Component {

  state = {
    // Datos de la entidad
    proceso: '',
    puntoSalida: '',
    periodo: '',
    trmTipo: '',
    contratoSeleccionado: '',
    fecha: '',
    valorTRM: '',
    numeroFactura: '',
    valorFactura: '',
    puntoSeleccionado: '',
    eds: '',
    unidadMedida: '',
    umNominado: '',
    nominado: '',
    estado: false,
    poderCalorificoPeriodo: '',
    listaPuntosSalida: [],
    listaPuntosSalidaAgregados: [],
    listaPuntosSalidaCalculados: [],
    listaContratos: [],
    listaPuntosConsumo: [],
    listaUnidadesMedida: [],
  };

  /**
   * Método encargado de comprobar si el formulario ya cargo
   */
  componentDidMount() {
    const { state } = this.props.history && this.props.history.location;
    if (state && state.listaContratos) {
      this.consultarPuntosSalida(state.listaContratos);
    }
    axios.post(RUTAS_API.CONFIGURACION.CONSULTAR_UNIDAD_PROGRAMA, { criterio: '', idClase: CLASES_UNIDADES.UNIDAD_MEDIDA })
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.setState({ listaUnidadesMedida: respuesta.data.datos });
        }
      });
  };

  /**
   * Método encargado de limpiar los campos del formulario
   * @param {Function} callBack Funcion a ejecutar
   */
  limpiarFormulario = (callback = null) => {
    this.setState({
      // Datos de la entidad
      contratoSeleccionado: '',
      proceso: '',
      puntoSalida: '',
      periodo: '',
      trmTipo: '',
      numeroFactura: '',
      valorFactura: '',
      fecha: '',
      eds: '',
      unidadMedida: '',
      umNominado: '',
      nominado: '',
      puntoSeleccionado: '',
      estado: false,
      listaPuntosSalida: [],
      listaPuntosSalidaAgregados: [],
      listaPuntosSalidaCalculados: [],
      listaContratos: [],
      listaPuntosConsumo: [],
    }, () => {
      if (typeof callback === 'function') {
        callback();
      }
    });
    limpiarHistorico(this.props);
  };

  /**
   * Método encargado de generar los botones del formulario
   * @returns {Object}
   */
  obtenerFunciones = () => {
    let funciones = [];
    if (this.state.proceso == 'V') {
      funciones.push({ texto: 'Generar Archivo', callback: this.generarArchivoValidacion });
    }
    if (this.state.proceso == 'DP') {
      funciones.push({ texto: 'Generar Reporte', callback: this.generarReporte });
    }
    funciones.push({ texto: 'Limpiar', callback: this.limpiarFormulario });
    return funciones;
  };

  /**
   * Método encargado de abrir la interfaz de consulta de contratos
   */
  consultarContrato = () => {
    this.props.history.push({
      pathname: RUTAS_VISTA.CONSULTA_CONTRATOS.url,
      state: {
        interfazGestion: RUTAS_VISTA.GESTION_REVISION_FACTURAS.url,
        seleccionMultiple: true,
      }
    });
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    if (evento.target.name == 'proceso') {
      this.limpiarFormulario(() => {
        this.setState(change);
      });
      return;
    }
    if (evento.target.name === 'trmTipo') {
      this.calcularTRM(() => {
        this.setState(change);
      }, evento.target.value);
      return;
    }
    this.setState(change);
  };

  /**
   * Método encargado de obtener la lista de identificadores de la lista de contratos
   * @param {Object} contratos Lista de contratos seleccionados
   * @returns {string}
   */
  obtenerIdContrato = (contratos) => {
    const identificadores = contratos.map((dato, index) => (
      dato.cntIderegistro
    ));
    return identificadores.toString();
  };

  /**
   * Método encargado de armar un objeto con los datos de punto de salida
   * @param {Object} puntosSalida Puntos de salida consultados
   * @returns {Object}
   */
  armarObjeto = (puntosSalida) => {
    const lista = puntosSalida.map((dato, index) => {
      return {
        ptsaNombre: dato.ptsaIdesalida.ptsaNombre,
        ptsaIderegistro: dato.ptsaIdesalida.ptsaIderegistro,
        idContrato: dato.cntIdecontrato.cntIderegistro,
      }
    });
    return lista;
  };

  /**
   * Método encargado de consultar los puntos de salida del contrato seleccionado
   * @param {Object} entidad Datos del contrato seleccionado
   */
  consultarPuntosSalida = (entidad) => {
    let { proceso, estado } = this.state;
    if (entidad.length > 1) {
      estado = true;
      proceso = 'DP';
      axios.post(RUTAS_API.PARAMETRIZACION.GESTION_DESVIOS.CONSULTAR_PUNTOS_SALIDA, { criterio: '', contratos: this.obtenerIdContrato(entidad) })
        .then(respuesta => {
          if (respuesta.data.codigo > 0) {
            this.setState({
              listaPuntosSalida: this.armarObjeto(respuesta.data.datos),
              listaContratos: entidad,
              listaPuntosSalidaAgregados: [],
              estado: estado,
              proceso: proceso,
            });
          }
        });
      return;
    }
    proceso = 'V';
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_DESVIOS.CONSULTAR_PUNTOS_SALIDA, { criterio: '', contratos: entidad[0].cntIderegistro })
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.setState({
            listaPuntosSalida: this.armarObjeto(respuesta.data.datos),
            contratoSeleccionado: entidad,
            listaPuntosSalidaAgregados: [],
            proceso: proceso,
          });
        }
      });
  };

  /**
   * Método encargado de seleccionar el punto que se desea para hacer el calculo
   * @param {number} posicion Posición del punto seleccionado el la lista
   */
  seleccionarPunto = (posicion) => {
    const punto = this.state.listaPuntosSalidaAgregados[posicion];
    this.setState({
      puntoSeleccionado: punto.puntoNombre
    });
  };

  /**
   * Método encargado de generar el archivo de validación
   * @returns {bool}
   */
  generarArchivoValidacion = () => {
    const { fecha, puntoSeleccionado, listaPuntosSalidaAgregados, contratoSeleccionado } = this.state;
    if (fecha === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el periodo');
      return false;
    }
    if (puntoSeleccionado === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar al cual le desea generar el reporte');
      return false;
    }
    const punto = listaPuntosSalidaAgregados.find(p => p.puntoNombre == puntoSeleccionado);
    const parametros = {
      idPuntosalida: punto.idPuntoSalida,
      anio: fecha.substr(0, 4),
      mes: fecha.substr(6, 1),
      idContrato: contratoSeleccionado[0].cntIderegistro
    };
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_REVISION_FACTURAS.GENERAR_DOCUMENTO_VALIDACION, parametros)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          let a = document.createElement('a');
          a.href = 'data:' + { type: "Content-Type: application/vnd.ms-excel" } + ';base64,' + respuesta.data.datos;
          a.download = "Reporte.xls";
          a.target = '_blank';
          a.click();
        }
      });
  };

  /**
   * Método encargado de calcular la TRM del tipo seleccionado
   * @param {Function} callBack Funcion a ejecutar
   * @param {String} tipoTRM tipo de la trm
   * @returns {bool}
   */
  calcularTRM = (callback, tipoTRM) => {
    const { fecha, trmTipo, puntoSeleccionado, listaPuntosSalidaAgregados, contratoSeleccionado } = this.state;
    if (fecha === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el periodo');
      return false;
    }
    if (puntoSeleccionado === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el punto que desea calcular');
      return false;
    }
    if (tipoTRM <= 0) {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el tipo de la TRM');
      return false;
    }
    const punto = listaPuntosSalidaAgregados.find(p => p.puntoNombre == puntoSeleccionado);
    const parametros = {
      puntoSalida:
        { ptsaIderegistro: punto.idPuntoSalida },
      anio: fecha.substr(0, 4),
      mes: fecha.substr(5, 2),
      tipoTRM: tipoTRM
    };
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_REVISION_FACTURAS.CALCULAR_TRM, parametros)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.setState({
            valorTRM: respuesta.data.datos,
            callback
          })
        }
        if (typeof callback === 'function') {
          callback();
        }
      });
  };

  /**
   * Método encargado de validar los campos necesarios para realizar el calculo de la validación
   * @returns {Object}
   */
  validarFormularioCalcular = () => {
    const { fecha, trmTipo, puntoSeleccionado, valorFactura, valorTRM, numeroFactura } = this.state;
    if (fecha === '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el periodo' } };
    }
    if (puntoSeleccionado === '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el punto que desea calcular' } };
    }
    if (trmTipo <= 0) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el tipo de la TRM' } };
    }
    if (valorFactura === '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe ingresar el valor de la factura' } };
    }
    if (numeroFactura === '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe ingresar el número de la factura' } };
    }
    if (valorTRM === '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe calcular el valor de la trm' } };
    }
    return { respuesta: true };
  };

  /**
   * Método encargado de validar los puntos agregados
   * @param {number} idPunto Identificador del punto de salida que se quiere agregar
   */
  validarPuntoCalculado = (idPunto) => {
    const lista = [...this.state.listaPuntosSalidaCalculados];
    const index = lista.findIndex(p => p.idPuntoSalida == idPunto);
    return index >= 0;
  };

  /**
   * Método encargado de realizar el calculo de la validación para el punto seleccionado
   * @return {bool}
   */
  calcular = () => {
    const { fecha, trmTipo, puntoSeleccionado, valorFactura,
      valorTRM, numeroFactura, listaPuntosSalidaAgregados,
      contratoSeleccionado, listaPuntosSalidaCalculados } = this.state;
    const validacion = this.validarFormularioCalcular();
    const puntoCalculado = listaPuntosSalidaCalculados.find(p => p.puntoNombre == puntoSeleccionado);
    if (puntoCalculado) {
      if (this.validarPuntoCalculado(puntoCalculado.idPuntoSalida)) {
        this.props.mostrarAlerta('Atención', 'Ya se realizo el calculo de el punto seleccionado');
        return false;
      }
    }
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return false;
    }
    const punto = listaPuntosSalidaAgregados.find(p => p.puntoNombre == puntoSeleccionado);
    const objetoEnviar = {
      puntoSalida:
        { ptsaIderegistro: punto.idPuntoSalida },
      anio: fecha.substr(0, 4),
      mes: fecha.substr(5, 2),
      numeroFactura: numeroFactura,
      valorFactura: valorFactura,
      contrato:
        { cntIderegistro: contratoSeleccionado[0].cntIderegistro },
      tipoTRM: trmTipo,
      valorTRM: valorTRM,
    }
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_REVISION_FACTURAS.CALCULAR_VALIDACION, objetoEnviar)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          listaPuntosSalidaCalculados.push({
            nomPropio: respuesta.data.datos.nominadoPropioKPC,
            nomDesvio: respuesta.data.datos.nominadoDesvioKPC,
            cantContratada: respuesta.data.datos.cantidadContratadaDiaria,
            uMedida: respuesta.data.datos.idemedida,
            puntoNombre: punto.puntoNombre,
            pComercializacion: respuesta.data.datos.comercializacion,
            trmDia: respuesta.data.datos.ultimoDiaPeriodoTRM,
            trmImpuesto: respuesta.data.datos.impuestoTRM,
            idPuntoSalida: punto.idPuntoSalida,
            nombreUnidad: respuesta.data.datos.idemedida.uniNombre1
          });
          this.setState({
            listaPuntosSalidaCalculados: listaPuntosSalidaCalculados,
          });
        }
      });
  };

  /**
   * Método encargado de mostrar la tabla con los puntos eds
   * @returns {Array}
   */
  renderTablaDispercion = () => {
    return (
      <table className='table table-striped mt-5'>
        <thead>
          <tr>

  controlarCambioUnidadMedida = (evento, index) => {

  /**
   * Método encargado de generar el reporte para el proceso de disperción
   * @returns {bool}
   */
  generarReporte = () => {
    const { fecha, puntoSeleccionado, listaPuntosSalidaAgregados, contratoSeleccionado, listaPuntosConsumo, listaContratos } = this.state;
    if (fecha === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el periodo');
      return false;
    }
    if (puntoSeleccionado === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar al cual le desea generar el reporte');
      return false;
    }
    const punto = listaPuntosSalidaAgregados.find(p => p.puntoNombre == puntoSeleccionado);
    const parametros = {
      idPuntosalida: punto.idPuntoSalida,
      anio: fecha.substr(0, 4),
      mes: fecha.substr(6, 1),
      idContrato: punto.idContrato
    };
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_REVISION_FACTURAS.GENERAR_DOCUMENTO_DISPERCION, parametros)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          let a = document.createElement('a');
          a.href = 'data:' + { type: "Content-Type: application/vnd.ms-excel" } + ';base64,' + respuesta.data.datos;
          a.download = "Reporte.xls";
          a.target = '_blank';
          a.click();
        }
      });
  };

  /**
   * Método encargado de obtener el identificador de los contratos seleccionados
   * @returns {Array}
   */
  obtenerContratos = () => {
    const { contratoSeleccionado, listaContratos } = this.state;
    let lista = contratoSeleccionado;
    if (listaContratos.length > 0) {
      return listaContratos.map((dato, index) => {
        return {
          cntIderegistro: dato.cntIderegistro
        }
      });
      return;
    }
    return lista.map((dato, index) => {
      return {
        cntIderegistro: dato.cntIderegistro
      }
    });
  };

  /**
   * Método encargado de hacer el calculo de la disperción para el punto seleccionado
   * @returns {bool}
   */
  calcularDispercion = () => {
    const { fecha, eds, puntoSeleccionado, listaPuntosSalidaAgregados, contratoSeleccionado, listaPuntosSalidaCalculados, listaPuntosConsumo } = this.state;
    if (eds === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el EDS');
      return false;
    }
    if (fecha === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el periodo');
      return false;
    }
    if (puntoSeleccionado === '') {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el punto que desea calcular');
      return false;
    }
    const punto = listaPuntosSalidaAgregados.find(p => p.puntoNombre == puntoSeleccionado);
    const objetoEnviar = {
      puntoSalida:
        { ptsaIderegistro: punto.idPuntoSalida },
      anio: fecha.substr(0, 4),
      mes: fecha.substr(5, 2),
      propio: eds,
      listacontrato: this.obtenerContratos(),
    }
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_REVISION_FACTURAS.CALCULAR_DISPERCION, objetoEnviar)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.setState({
            listaPuntosConsumo: respuesta.data.datos,
          });
        }
      });
  };

  /**
   * Método encargado de mostrar el formulario para el proceso de disperción
   * @returns {Object}
   */
  renderDispercion = () => {

        </div>
        {this.state.listaPuntosConsumo.length > 0 &&
          this.renderTablaEDS()
        }
        {this.state.listaPuntosConsumo.length > 0 &&
          this.renderTablaEDSCalculos()
        }
      </div>
    )
  };

  /**
   * Método encargado de validar los puntos agregados
   * @param {number} idPunto Identificador del punto de salida que se quiere agregar
   */
  validarPuntoRepetido = (idPunto) => {
    const lista = [...this.state.listaPuntosSalidaAgregados];
    const index = lista.findIndex(p => p.idPuntoSalida == idPunto);
    return index >= 0;
  };

  /**
   * Método encargado de obtener los tipos del contrato y retornar un string
   * @param {Object} contratoSeleccionado Datos del contrato seleccionado
   */
  obtenerTipos = (contratoSeleccionado) => {
    if (contratoSeleccionado === '') {
      return false;
    }
    const listaTipos = contratoSeleccionado.listaTipos;
    if (!Array.isArray(listaTipos) || listaTipos.length == 0) {
      return 'Indefinido';
    }
    return listaTipos.map(tipo => {
      return tipo.uniIdetipocontrato.uniNombre1;
    }).join(',');
  };

  /**
   * Método encargado de agregar el punto de salida seleccionado a la lista
   * @returns {bool}
   */
  agregarSeleccionado = () => {
    const { contratoSeleccionado, puntoSalida, listaPuntosSalidaAgregados, listaPuntosSalida, proceso } = this.state;
    if (puntoSalida <= 0) {
      this.props.mostrarAlerta('Datos Incompletos', 'Debe seleccionar el punto de salida que desea agregar');
      return false;
    }
    if (this.validarPuntoRepetido(puntoSalida)) {
      this.props.mostrarAlerta('Atención', 'El punto que esta intentado agregar ya se encuentra en la lista');
      return false;
    }
    const puntoSeleccionado = listaPuntosSalida.find(p => p.ptsaIderegistro == puntoSalida);
    const contrato = contratoSeleccionado[0];
    listaPuntosSalidaAgregados.push({
      nombreAgente: (contrato) ? contrato.terIdeagente.terNomcompleto : '',
      numeroContrato: (contrato) ? contrato.cntNumero : '',
      fechaIni: (contrato) ? contrato.cntFechainicio : '',
      fechaFin: (contrato) ? contrato.cntFechafin : '',
      puntoNombre: puntoSeleccionado.ptsaNombre,
      tipos: this.obtenerTipos((contrato) ? contrato : ''),
      idPuntoSalida: puntoSeleccionado.ptsaIderegistro,
      idContrato: (contrato) ? contrato.cntIderegistro : puntoSeleccionado.idContrato,
    });
    this.setState({
      listaPuntosSalidaAgregados: listaPuntosSalidaAgregados
    });
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    let numeroContrato = getProp(this.state, 'contratoSeleccionado', null);
    numeroContrato = Util.validarArreglo(numeroContrato) ? numeroContrato[0].cntNumero : null;
    return (
      <Fragment>
        <Botonera funciones={this.obtenerFunciones()} />
        <div className='conf-general row mt-5'>
          <Combo
            opciones={listaProcesos}
            propTexto='texto'
            propValor='id'
            label='Proceso:'
            name='proceso'
            value={this.state.proceso}
            onChange={this.controlarCambio}
            extra={{ disabled: this.state.estado, readOnly: this.state.estado }}
          />
          <div className='form-group col-4'>
            <label>Contrato:</label>
            <div className='input-group'>
              <input type='text' className='form-control' value={numeroContrato ? numeroContrato : 'Seleccionar Contrato'} disabled='true' />
              <button className='btn btn-primary' onClick={this.consultarContrato}>
                <i className='fa fa-fw fa-check-square-o'></i>
              </button>
            </div>
          </div>
          <Fecha
            label='Periodo:'
            name='fecha'
            fecha={this.state.fecha}
            sinDia={true}
            onChange={this.controlarCambio}
          />
          <Combo
            opciones={this.state.listaPuntosSalida}
            propTexto='ptsaNombre'
            propValor='ptsaIderegistro'
            label='Punto de Salida:'
            name='puntoSalida'
            value={this.state.puntoSalida}
            onChange={this.controlarCambio}
          />
          <div className='col-12'>
            <button className='btn btn-primary botonFacturas' onClick={this.agregarSeleccionado}><i className='fa fa-fw fa-plus'></i> Agregar Punto Salida</button>
          </div>
          {this.state.proceso === 'V' &&
            this.renderValidacion()
          }

        </div>
        {this.state.proceso === 'DP' &&
          this.renderTablaDispercion()
        }
        {this.state.proceso === 'DP' &&
          this.renderDispercion()
        }
      </Fragment>
    );
  };
}
