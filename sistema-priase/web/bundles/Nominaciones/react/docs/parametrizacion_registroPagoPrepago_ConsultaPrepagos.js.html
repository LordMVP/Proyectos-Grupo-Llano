import React, { Component } from 'react';
import connect from 'react-redux/es/connect/connect';
import { bindActionCreators } from 'redux';
import PropTypes from 'prop-types';

import { Tabla, Input, Botonera, TextoNumerico, Fecha, Util } from 'appfuture-react';
import RUTAS_API from '../../../global/rutas_api';
import RUTAS_VISTA from '../../../global/rutas_vista';
import { TECLAS } from '../../../global/constantes';
import ConsultaGenerica from '../../../hoc/consultaGenerica/ConsultaGenerica';

class ConsultaPrepago extends Component {

  consultaGenerica = null;

  state = {
    criterio: '',
    listadoEntidad: [],
  };

  obtenerFunciones = () => {
    let funciones = [{ texto: 'Consultar', callback: this.onBuscar }];
    if (this.props.esModal && this.props.seleccionMultiple) {
      funciones.push({ texto: 'Seleccionar Datos', callback: this.onSeleccionarEntidades });
    }
    funciones.push({ texto: 'Limpiar', callback: this.limpiarFormulario });
    return funciones;
  };

  onSeleccionarEntidades = () => {
    this.props.seleccionarEntidades(this.consultaGenerica.getWrappedInstance()._obtenerEntidades());
  };

  /**
   * Método encargado de limpiar el formulario
   */
  limpiarFormulario = () => {
    this.consultaGenerica.getWrappedInstance()._limpiarFormulario();
    this.setState({
      criterio: '',
      idRecaudo: '',
      periodoRecaudo: '',
      agente: '',
      numeroContrato: '',
      fechaRecaudo: '',
      listadoEntidad: []
    });
  };

  onCriterioChange = (event) => {
    this.setState({ criterio: event.target.value });
  };

  onKeyPress = (evento) => {
    if (evento.charCode === TECLAS.ENTER) {
      this.onBuscar();
    }
  };

  cargarDatos = (datos) => {
    console.log(datos);
    this.setState({
      idRecaudo: datos.rccIderegistro,
      cntIdecontrato: datos.cntNumero,
      numeroContrato: datos.cntIdecontrato,
      periodoRecaudo: datos.rccPeriodo,
      fechaRecaudo: datos.rccFecharecaudo,
      agente: datos.cntIdecontrato.terIdeagente.terNomcompleto
    });
  };

  /**
 * Método encargado de generar los botones del formulario
 * @returns {Object}
 */
  obtenerFunciones = () => {
    let funciones = [{ texto: 'Consultar', callback: this.onBuscar }];
    if (this.props.esModal && this.props.seleccionMultiple) {
      funciones.push({ texto: 'Seleccionar Datos', callback: this.onSeleccionarEntidades });
    }
    funciones.push({ texto: 'Limpiar', callback: this.limpiarFormulario });
    return funciones;
  };

  /**
   * Básicamente recibe un string fecha con el formato YYYY-MM y devuelve uno con el formato MM-YYYY.
   * @return {string}
   */
  obtenerPeriodo = (periodoRecaudo) => {
    if (periodoRecaudo && periodoRecaudo.trim() != "") {
      return periodoRecaudo.split('-').reverse().join('/');
    }
  };

  /**
   * Metodo encargado de realizar la consulta
   */
  onBuscar = () => {
    const { criterio, idRecaudo, periodoRecaudo, agente, numeroContrato, fechaRecaudo } = this.state;
    this.consultaGenerica.getWrappedInstance()._buscar({
      criterio: criterio,
      idRecaudo: idRecaudo,
      periodo: this.obtenerPeriodo(periodoRecaudo),
      agente: agente,
      numeroContrato: numeroContrato,
      fechaRecaudo: fechaRecaudo
    });
  };


  /**
  * Método encargado de obtener las columnas del componente Tabla
  * @returns {Object}
  */
  obtenerColumnas = () => {
    const contexto = this;
    return [
      {
        Header: 'Prepagos',
        columns: [
          {
            Header: 'Número Recaudo',
            accessor: 'rccIderegistro'
          },
          {
            Header: 'Número Contrato',
            accessor: 'cntIdecontrato.cntNumero'
          },
          {
            Header: 'Agente',
            accessor: 'cntIdecontrato.terIdeagente.terNomcompleto'
          },
          {
            Header: 'Fecha Recuado',
            accessor: 'rccFecharecaudo',
          },
          {
            Header: 'Valor',
            accessor: 'rccValor'
          },
        ]
      }
    ];
  };

  /**
   * Método encargado de generar el componente Tabla
   * @returns {Component}
   */
  renderTabla = () => {
    if (!Util.validarArreglo(this.state.listadoEntidad)) {
      return <div className='text-center '>Sin resultados</div>;
    }

    return (
      <Tabla
        datos={this.state.listadoEntidad}
        columnas={this.obtenerColumnas()}
      />
    );
  };

  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    this.setState(change);
  };

  /**
  * Método encargado de obtener el nombre de la actividad
  * @param {Object} props Propiedades del componente Tabla
  */
  obtenerFecha = (props) => {
    const fechaNumero = parseInt(props.row._original.rccFecharecaudo);
    let fecha = new Date(fechaNumero);
    fecha = (fecha.getFullYear() + "/" + (fecha.getMonth() + 1) + "/" + fecha.getDate());
    return (fechaNumero) ? fecha : '';
  };

  render() {
    return (
      <div className='consulta-tramos'>
        <Botonera funciones={this.obtenerFunciones()} />

        <div className='row mt-5'>
          <TextoNumerico
            aceptaDecimales={false}
            aceptaNegativos={false}
            label='Número Recaudo:'
            cols={4}
            value={this.state.idRecaudo}
            onChange={this.controlarCambio}
            name='idRecaudo'
          />
          <TextoNumerico
            aceptaDecimales={false}
            aceptaNegativos={false}
            label='Número de contrato:'
            cols={4}
            value={this.state.numeroContrato}
            onChange={this.controlarCambio}
            name='numeroContrato'
          />
          <Fecha
            label='Periodo Recaudo:'
            name='periodoRecaudo'
            sinDia={true}
            fecha={this.state.periodoRecaudo}
            onChange={this.controlarCambio}
          />
          <Fecha
            label='Fecha Recaudo:'
            name='fechaRecaudo'
            fecha={this.state.fechaRecaudo}
            onChange={this.controlarCambio}
          />
          <Input
            label='Agente:'
            value={this.state.agente}
            onChange={this.controlarCambio}
            name='agente'
          />
        </div>
        <div className='mt-5'>
          <ConsultaGenerica
            {...this.props}
            idEntidad='idRegistro'
            columnas={this.obtenerColumnas()}
            // seleccionMultiple={false}
            // seleccionarEntidad={this.cargarDatos}
            ref={ref => this.consultaGenerica = ref}
            interfazGestion={RUTAS_VISTA.GESTION_PAGO_PREPAGO.url}
            rutaConsulta={RUTAS_API.PARAMETRIZACION.GESTION_PAGO_PREPAGO.CONSULTAR_RECAUDOS}
          />
        </div>
      </div>

    );
  };

}

ConsultaPrepago.propTypes = {
  history: PropTypes.object,
  esModal: PropTypes.bool,
  seleccionMultiple: PropTypes.bool,
  seleccionarEntidad: PropTypes.func,
  seleccionarEntidades: PropTypes.func,
  entidadesSeleccionadas: PropTypes.array
};

ConsultaPrepago.defaultProps = {
  esModal: false,
  seleccionMultiple: false,
  entidadesSeleccionadas: []
};

const mapStateToProps = state => {
  return {};
};

const mapDispatchToProps = dispatch => {
  return bindActionCreators({}, dispatch);
};

const VistaRedux = connect(mapStateToProps, mapDispatchToProps)(ConsultaPrepago);

export { VistaRedux as RConsultaPrepago };
