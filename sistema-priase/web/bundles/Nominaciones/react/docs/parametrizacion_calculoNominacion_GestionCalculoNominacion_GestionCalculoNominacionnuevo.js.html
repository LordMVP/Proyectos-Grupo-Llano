  obtenerIdEstado = (codigoEstado) => {
    const { listaEstados } = this.state;
    if (!Util.validarArreglo(listaEstados)) {
      return 0;

    const estado = listaEstados.filter(e => (JSON.parse(e.uniPropiedad).estado == codigoEstado));
