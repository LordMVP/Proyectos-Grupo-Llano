import React, { Component, Fragment } from 'react';
import connect from 'react-redux/es/connect/connect';
import { bindActionCreators } from 'redux';
import PropTypes from 'prop-types';
import { Input, Botonera, Combo, VentanaModal, Util, Fecha, TextoNumerico } from 'appfuture-react';
import axios from 'axios';
import RUTAS_API from '../../../../global/rutas_api';
import { formatearArray } from '../../../../global/util_nominaciones';
import { mostrarAlerta } from '../../../../store/actions/AplicacionAcciones';
import './GestionCuentaBalance.scss';

class GestionCuentaBalance extends Component {

  state = {
    // Datos de la entidad
    listaTramos: [],
    listaContratos: [],
    listaPuntosPositivos: [],
    listaConsulta: [],
    tramo: '',
    anio: '',
    mes: '',
  };

  /**
   * Método encargado de comprobar si el formulario ya cargo
   */
  componentDidMount() {
    const { state } = this.props.history && this.props.history.location;
    if (state && state.entidadEditar) {
      this.cargarDatos(state.entidadEditar);
    }
    this.obtenerAnios();
    const peticiones = [
      axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CUENTA_BALANCE.CONSULTAR_TRAMOS),
      axios.post(RUTAS_API.CONTRATOS.CONSULTAR_CONTRATOS, { criterio: '' })
    ];

    axios.all(peticiones)
      .then(axios.spread((tramos, contratos) => {
        const datosAplicacion = {
          listaTramos: [],
          listaContratos: [],
        };

        if (tramos.data.codigo > 0) {
          datosAplicacion.listaTramos = tramos.data.datos;
        }

        if (contratos.data.codigo > 0) {
          datosAplicacion.listaContratos = contratos.data.datos;
        }
        this.setState({ ...datosAplicacion });
      }));

  };

  /**
   * Método encargado de limpiar los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  limpiarFormulario = (evento) => {
    this.setState({
      tramo: '-1',
      anio: '',
      mes: '',
      listaConsulta: [],
      listaPuntosPositivos: [],
    });
  };

  /**
   * Método encargado de llenar el combo de años
   */
  obtenerAnios = () => {
    const fechaActual = new Date();
    const anioActual = fechaActual.getFullYear();
    const anioInicial = anioActual - 100;
    for (let i = anioActual; i > anioInicial; i--) {
      listaAnios.push({ texto: i, valor: i });
    }
  };

  /**
   * Método encargado de limpiar el formulario al momento de cambiar de interfaz
   */
  componentWillUnmount() {
    this.limpiarFormulario();
  };

  /**
   * Método encargado de generar los botones del formulario
   * @returns {Object}
   */
  obtenerFunciones = () => {
    return [
      { texto: 'Agregar Tramo', callback: this.agregarTramo },
      { texto: 'Guardar', callback: this.guardarEntidad },
      { texto: 'Generar Documento', callback: this.generarDocumento },
      { texto: 'Limpiar', callback: this.limpiarFormulario }
    ];
  };

  /**
   * Método encargado de validar los campos de la tabla de negativos
   * @returns {Object}
   */
  validarTablaNegativos = () => {
    const { listaPuntosPositivos } = this.state;
    for (let index = 0; index < listaPuntosPositivos.length; index++) {
      const puntoValidar = listaPuntosPositivos[index];
      if ((!puntoValidar.puntoNegativo) || (!puntoValidar.puntoNegativo.numeroContrato || puntoValidar.puntoNegativo.numeroContrato <= 0)) {
        return { respuesta: false, mensaje: { titulo: 'Datos Incompletos', mensaje: `Debe seleccionar un contrato para el punto negativo: ${puntoValidar.puntoNegativo.ptsaNombre}` } };
      }

      if (!puntoValidar.puntoNegativo.descripcion || puntoValidar.puntoNegativo.descripcion.trim() === '') {
        return { respuesta: false, mensaje: { titulo: 'Datos Incompletos', mensaje: `Debe seleccionar una descripción para el punto negativo: ${puntoValidar.puntoNegativo.ptsaNombre}` } };
      }
    };
    return { respuesta: true }
  };

  /**
   * Método encargado de validar las variables necesarias para consultar los datos del cruce en el tramo seleccionado
   * @returns {Object}
   */
  validarFormularioAgregarTramo = () => {
    const { mes, anio, tramo } = this.state;
    if (tramo <= 0) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el tramo' } };
    }

    if (anio <= 0) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el año' } };
    }

    if (mes <= 0) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el mes' } };
    }

    return { respuesta: true };
  };

  /**
   * Método encargado de validar las variables necesarias para guardar
   */
  validarFormulario = () => {
    const { listaPuntosPositivos } = this.state;
    const listaFiltrada = listaPuntosPositivos.filter(p => p.puntoNegativo != null && p.puntosPositivos != null);
    if (!Util.validarArreglo(listaPuntosPositivos)) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe agregar al menos 1 tramo' } };
    }

    if (!Util.validarArreglo(listaFiltrada)) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe agregar al menos 1 tramo que tenga cruce' } };
    }
    return { respuesta: true };
  };

  /**
   * Método encargado de validar las variables del formulario para generar el documento de cruces
   * @returns {Object}
   */
  validarFormularioDocumento = () => {
    const { mes, anio } = this.state;
    // Validaciones
    if (anio <= 0) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el año' } };
    }

    if (mes <= 0) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el mes' } };
    }

    return { respuesta: true };
  };

  /**
   * Método encargado de generar el documento del cruce
   * @returns {bool}
   */
  generarDocumento = () => {
    const validarDocumento = this.validarFormularioDocumento();
    if (!validarDocumento.respuesta) {
      this.props.mostrarAlerta(validarDocumento.mensaje.titulo, validarDocumento.mensaje.mensaje);
      return false;
    }
    const { mes, anio } = this.state;
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CUENTA_BALANCE.GENERAR_DOCUMENTO, { mes: mes, anio: anio })
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          let a = document.createElement('a');
          a.href = 'data:' + { type: "Content-Type: application/vnd.ms-excel" } + ';base64,' + respuesta.data.datos;
          a.download = "Reporte Cruce Cuenta Balance.xls";
          a.target = '_blank';
          a.click();
        }
      });
  };

  /**
   * Método encargado de generar un objeto para guardar el cruce o cruces
   * @returns {Object}
   */
  obtenerObjetoGuardar = () => {
    const { listaPuntosPositivos, mes, anio } = this.state;
    const listaFiltrada = listaPuntosPositivos.filter(p => p.puntoNegativo != null && p.puntosPositivos != null);
    const lista = listaFiltrada.map((dato, index) => {
      return {
        'cntIdecontrato': { 'cntIderegistro': dato.puntoNegativo.numeroContrato },
        'ptsaIdepositivo': { 'ptsaIderegistro': dato.puntosPositivos.ptsaIderegistro },
        'ptsaIdenegativo': { 'ptsaIderegistro': dato.puntoNegativo.ptsaIderegistro },
        'uniIdemedida': { 'uniIderegistro': dato.unidadMedida.uniIderegistro },
        'ctbSaldopositivo': dato.ctbSaldopositivo,
        'ctbSaldonegativo': dato.ctbSaldonegativo,
        'ctbNuevosaldo': dato.ctbNuevosaldo,
        'ctbFechacruce': `${anio}-${mes}`,
        'ctbDescripcion': dato.puntoNegativo.descripcion,
      }
    });
    const objetoEnviar = {
      cruceCuentaLista: lista,
    };
    return objetoEnviar;
  };

  /**
   * Método encargado de guardar los datos de la entidad
   * @returns {bool}
   */
  guardarEntidad = () => {
    const validacion = this.validarFormulario();
    const validacionTabla = this.validarTablaNegativos();
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return false;
    }

    if (!validacionTabla.respuesta) {
      this.props.mostrarAlerta(validacionTabla.mensaje.titulo, validacionTabla.mensaje.mensaje);
      return false;
    }

    const entidadGuardar = this.obtenerObjetoGuardar();
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CUENTA_BALANCE.GUARDAR, entidadGuardar)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.limpiarFormulario();
        }
      });
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    this.setState(change);
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   * @param {number} index Posición que se esta modificando
   */
  controlarCambioNumeroContrato = (evento, index) => {
    let puntos = [...this.state.listaPuntosPositivos];
    puntos[index].puntoNegativo.numeroContrato = evento.target.value;
    this.setState({ listaPuntosPositivos: puntos });
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   * @param {number} index Posición que se esta modificando
   */
  controlarCambioDescripcion = (evento, index) => {
    let puntos = [...this.state.listaPuntosPositivos];
    puntos[index].puntoNegativo.descripcion = evento.target.value;
    this.setState({ listaPuntosPositivos: puntos });
  };

  /**
   * Obtiene la lista de contratos que tienen fuente.
   * @return {array}
   */
  obtenerContratosConFuente = () => {
    return this.state.listaContratos.filter(contrato => {
      if (contrato.uniIdefuente && contrato.uniIdefuente.uniIderegistro) {
        return contrato;
      }
    }).map(contrato => {
      contrato.nombre = contrato.cntNumero + ' - ' + contrato.terIdeagente.terNomcompleto;
      return contrato;
    });
  };

  /**
   * Método encargado de mostrar la tabla con los tramos y puntos negativos agregados
   * @returns {Object}
   */
  renderTablaNegativos = () => {
    const { listaPuntosPositivos } = this.state
    let lista = listaPuntosPositivos.filter(p => p.puntoNegativo != null);
    return (
      <table className='table table-striped mt28'>
        <thead>
          <tr>
            <th>Lista Puntos Negativos</th>
          </tr>
          <tr>
            <th>Tramo</th>
            <th>Punto de Salida</th>
            <th>Saldo</th>
            <th>Saldo Cruzado</th>
            <th>Nuevo Saldo</th>
            <th>U/M</th>
            <th>Número de contrato</th>
            <th>Detalle</th>
          </tr>
        </thead>
        <tbody>
          {
            lista.map((dato, index) => {
              return (
                <tr key={`"punto_"${dato.puntoNegativo.ptsaNombre}`}>
                  <td>{dato.tramo.trmNombre}</td>
                  <td>{dato.puntoNegativo.ptsaNombre}</td>
                  <td>{dato.ctbSaldonegativo}</td>
                  <td>{dato.ctbNuevosaldo}</td>
                  <td>{dato.ctbNuevosaldo}</td>
                  <td>{dato.unidadMedida.uniNombre1}</td>
                  <td>
                    <Combo
                      opciones={this.obtenerContratosConFuente()}
                      propTexto='nombre'
                      propValor='cntIderegistro'
                      name='contrato'
                      cols={12}
                      value={dato.puntoNegativo.numeroContrato}
                      onChange={(evento) => {
                        this.controlarCambioNumeroContrato(evento, index)
                      }}
                    />
                  </td>
                  <td>
                    <Input
                      value={dato.puntoNegativo.descripcion}
                      cols={12}
                      onChange={(evento) => {
                        this.controlarCambioDescripcion(evento, index)
                      }}
                      name='descripcion'
                    />
                  </td>
                </tr>
              );
            })
          }
        </tbody>
      </table>
    );
  };

  /**
   * Método encargado de mostrar la tabla con los puntos positivos agregados
   * @returns {Object}
   */
  renderTablaPositivos = () => {
    const { listaPuntosPositivos } = this.state
    let lista = listaPuntosPositivos.filter(p => p.puntosPositivos != null);
    return (
      <table className='table table-striped'>
        <thead>
          <tr>
            <th>Lista Puntos Positivos</th>
          </tr>
          <tr>
            <th>Tramo</th>
            <th>Punto de Salida</th>
            <th>Desbalance positivo</th>
            <th>U/M</th>
            <th>Nuevo Saldo</th>
            <th>Estado</th>
          </tr>
        </thead>
        <tbody>
          {
            lista.map((dato, index) => {
              return (
                <tr key={dato.puntosPositivos.ptsaIderegistro}>
                  <td>{dato.tramo.trmNombre}</td>
                  <td>{dato.puntosPositivos.ptsaNombre}</td>
                  <td>{dato.ctbSaldopositivo}</td>
                  <td>{dato.unidadMedida.uniNombre1}</td>
                  <td>{dato.ctbNuevosaldo}</td>
                  <td>Pendiente</td>
                </tr>
              );
            })
          }
        </tbody>
      </table>
    );
  };

  /**
   * Método encargado de validar que no se seleccione dos veces el mismo tramo
   * @param {number} idTramo Identificador del tramo seleccionado
   * @returns {number}
   */
  validarRepetido = (idTramo) => {
    const lista = [...this.state.listaConsulta];
    const index = lista.findIndex(p => p.trmIderegistro.trim() == idTramo.trim());
    return index >= 0
  };

  /**
   * Método engargado de generar un objeto con los datos para consultar el cruce
   * @returns {Object}
   */
  obtenerDatosGenerar = () => {
    const { tramo, mes, anio } = this.state;
    let { listaConsulta } = this.state;
    let objetoDevolver = {
      mes: parseInt(mes),
      anio: parseInt(anio),
    };
    listaConsulta.push({ trmIderegistro: tramo });
    const lista = listaConsulta.map(dato => {
      return {
        trmIderegistro: dato.trmIderegistro
      }
    });
    objetoDevolver.listaTramo = lista;
    return objetoDevolver;
  };

  /**
   * Metodo encargado de llenar la lista con el punto negativo o positivo consultado
   * @param {Object} datos Datos de los puntos consultados
   */
  pushDatosSinCruce = (datos) => {
    const { tramo, listaTramos } = this.state;
    const { listaConsulta } = this.state;
    const index = listaConsulta.findIndex(p => p.trmtrmIderegistro == tramo);
    listaConsulta.splice(index, 1);
    let tramoSeleccionado = listaTramos.find(p => p.trmIderegistro == tramo);
    const lista = [...this.state.listaPuntosPositivos];
    for (let index = 0; index < datos.length; index++) {
      const dato = datos[index];
      const punto = lista.find(p => p.puntosPositivos && p.puntosPositivos.ptsaIderegistro == dato.puntoSalida);
      if (dato.desbalance > 0) {
        if (punto) {
          this.props.mostrarAlerta('Ateción', `El punto ${dato.nombrePuntoSalida} ya se encuentra en la lista de puntos positivos`);
          return;
        } else {
          lista.push({
            puntosPositivos: {
              ptsaIderegistro: dato.puntoSalida,
              ptsaNombre: dato.nombrePuntoSalida,
            },
            ctbSaldopositivo: dato.desbalance,
            ctbSaldonegativo: 0,
            ctbNuevosaldo: 0,
            unidadMedida: {
              uniNombre1: 'MBTU',
            },
            tramo: tramoSeleccionado,
            puntoNegativo: null,
          });
        }
      } else {
        lista.push({
          puntosPositivos: null,
          ctbSaldopositivo: 0,
          ctbSaldonegativo: dato.desbalance,
          ctbNuevosaldo: 0,
          unidadMedida: {
            uniNombre1: 'MBTU',
          },
          tramo: tramoSeleccionado,
          puntoNegativo: {
            ptsaIderegistro: dato.puntoSalida,
            ptsaNombre: dato.nombrePuntoSalida,
          },
        });
      }
    }
    this.setState({ listaPuntosPositivos: lista, listaConsulta: listaConsulta });
  };

  /**
   * Método encargado de llenar la lista con los puntos positivos en el cruce
   * @param {Object} datos Datos del cruce
   */
  pushDatosPuntosPositivos = (datos) => {
    const { tramo, listaTramos } = this.state;
    let tramoSeleccionado = listaTramos.find(p => p.trmIderegistro == tramo);
    const lista = [...this.state.listaPuntosPositivos];
    for (let index = 0; index < datos.length; index++) {
      const dato = datos[index];
      const punto = lista.find(p => p.puntosPositivos.ptsaIderegistro == dato.cruceCuenta.ptsaIdepositivo.ptsaIderegistro);
      const posicion = lista.findIndex(p => p.puntosPositivos.ptsaIderegistro == dato.cruceCuenta.ptsaIdepositivo.ptsaIderegistro);
      if (lista.length > 0) {
        if (punto) {
          if (punto.puntoNegativo == null) {
            lista.splice(posicion, 1);
            lista.push({
              puntosPositivos: dato.cruceCuenta.ptsaIdepositivo,
              ctbSaldopositivo: dato.cruceCuenta.ctbSaldopositivo,
              ctbSaldonegativo: dato.cruceCuenta.ctbSaldonegativo,
              ctbNuevosaldo: dato.cruceCuenta.ctbNuevosaldo,
              unidadMedida: dato.cruceCuenta.uniIdemedida,
              tramo: tramoSeleccionado,
              puntoNegativo: dato.cruceCuenta.ptsaIdenegativo,
            });
          }
          continue;
        } else {
          lista.push({
            puntosPositivos: dato.cruceCuenta.ptsaIdepositivo,
            ctbSaldopositivo: dato.cruceCuenta.ctbSaldopositivo,
            ctbSaldonegativo: dato.cruceCuenta.ctbSaldonegativo,
            ctbNuevosaldo: dato.cruceCuenta.ctbNuevosaldo,
            unidadMedida: dato.cruceCuenta.uniIdemedida,
            tramo: tramoSeleccionado,
            puntoNegativo: dato.cruceCuenta.ptsaIdenegativo,
          });
        }
      } else {
        lista.push({
          puntosPositivos: dato.cruceCuenta.ptsaIdepositivo,
          ctbSaldopositivo: dato.cruceCuenta.ctbSaldopositivo,
          ctbSaldonegativo: dato.cruceCuenta.ctbSaldonegativo,
          ctbNuevosaldo: dato.cruceCuenta.ctbNuevosaldo,
          unidadMedida: dato.cruceCuenta.uniIdemedida,
          tramo: tramoSeleccionado,
          puntoNegativo: dato.cruceCuenta.ptsaIdenegativo,
        });
      }
    }
    this.setState({ listaPuntosPositivos: lista });
  };

  /**
   * Método encargado de consultar los puntos positivos y negativos del tramo seleccionado
   * @returns {bool}
   */
  agregarTramo = () => {
    const { tramo } = this.state;
    if (this.validarRepetido(tramo)) {
      this.props.mostrarAlerta('Atención', 'Este tramo ya se encuentra agregado');
      return false;
    }
    const validar = this.validarFormularioAgregarTramo();
    if (!validar.respuesta) {
      this.props.mostrarAlerta(validar.mensaje.titulo, validar.mensaje.mensaje);
      return;
    }

    const entidadGenerar = this.obtenerDatosGenerar();
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CUENTA_BALANCE.CONSULTAR_CRUCE, entidadGenerar)
      .then(respuesta => {
        if (respuesta.data.codigo <= 0) {
          this.setState({ listaConsulta: [] });
          return;
        }
        if (respuesta.data.datos[0].cruceCuenta) {
          this.pushDatosPuntosPositivos(respuesta.data.datos);
          return;
        }
        this.pushDatosSinCruce(respuesta.data.datos);
      });
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <Fragment>
        <Botonera funciones={this.obtenerFunciones()} />
        <div className='conf-general row mt-5'>
          <Combo
            opciones={this.state.listaTramos}
            propTexto='trmNombre'
            propValor='trmIderegistro'
            label='Tramos:'
            name='tramo'
            value={this.state.tramo}
            onChange={this.controlarCambio}
          />
          <Combo
            opciones={listaAnios}
            propTexto='texto'
            propValor='valor'
            label='Años:'
            name='anio'
            value={this.state.anio}
            onChange={this.controlarCambio}
          />
          <Combo
            opciones={listaMes}
            propTexto='texto'
            propValor='valor'
            label='Mes:'
            name='mes'
            value={this.state.mes}
            onChange={this.controlarCambio}
          />
          {this.state.listaPuntosPositivos.length > 0 &&
            this.renderTablaPositivos()
          }
          {
            this.state.listaPuntosPositivos.length > 0 &&
            this.renderTablaNegativos()
          }
        </div>
      </Fragment>
    );
  };
}
