  consultarListas = () => {
    const peticiones = [
      axios.post(RUTAS_API.PARAMETRIZACION.GESTION_PUNTOS_CONSUMO.CONSULTAR_PUNTOS_CONSUMO, { criterio: '' }),
      axios.post(RUTAS_API.CONFIGURACION.CONSULTAR_UNIDAD, { criterio: '', idClase: CLASES_UNIDADES.ESTADOS_CONTRATO }),
      axios.post(RUTAS_API.PARAMETRIZACION.GESTION_DESVIOS.CONSULTAR_DESVIOS, { criterio: '' })
    ];
    axios.all(peticiones)
      .then(axios.spread((puntosConsumo, estadosContrato, listaDesvios) => {
        const datosAplicacion = {
          listaPuntos: [],
          listaEstados: [],
        };
        if (puntosConsumo.data.codigo > 0) {
          datosAplicacion.listaPuntos = puntosConsumo.data.datos;
        }
        if (estadosContrato.data.codigo > 0) {
          datosAplicacion.listaEstados = estadosContrato.data.datos;
        }
        if (listaDesvios.data.codigo > 0) {
          datosAplicacion.listaDesvios = listaDesvios.data.datos;
        }
        // datosAplicacion.listaCalculo = this.getObject();
        this.setState({ ...datosAplicacion });
      }));
  };

  /**
   * Método encargado de obtener el objeto para guardar la nominación
   * @returns {Object}
   */
  obtenerObjetoGuardar = () => {
    const { listaCalculo } = this.state;
    const lista = listaCalculo.map(punto => {
      const medidores = getProp(punto.nominacionDetalleDTO, 'medidores', null);
      const fuentes = getProp(punto.nominacionDetalleDTO, 'fuenteSuministro', null);
      const tramos = getProp(punto.nominacionDetalleDTO, 'tramos', null);
      if (medidores != null && Object.keys(medidores).length == 0) {
        punto.nominacionDetalleDTO.medidores = null;
      }
      if (fuentes != null && Object.keys(fuentes).length == 0) {
        punto.nominacionDetalleDTO.fuenteSuministro = null;
      }
      if (tramos != null && Object.keys(tramos).length == 0) {
        punto.nominacionDetalleDTO.tramos = null;
      }
      return punto;
    });
    return lista;
  };

  /**
   * Método encargado de consultar el horario para nominar dependiendo del tipo de contrato.
   * @param {string} tipoContrato Tipo de contrato seleccionado
   */
  consultarHorario = (tipoContrato) => {
    let { visibilidadBoton } = this.state;
    if (tipoContrato != '-1') {
      axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_NOMINACION.CONSULTAR_HORARIO_NOMINACION, { tipoContrato: tipoContrato })
        .then(respuesta => {
          if (respuesta.data.codigo > 0) {
            const hora = respuesta.data.datos[0];
            let tiempo = (parseFloat(respuesta.data.datos[1]) * -1);
            tiempo = tiempo.toFixed(2);
            if (tiempo > 0) {
              visibilidadBoton = true;
            }
            const tRestante = `${tiempo} horas`;
            this.setState({
              tipoContrato: tipoContrato,
              horaLimite: hora,
              tiempoRestante: tRestante,
              mostrarAlertaActividad: true,
              visibilidadBoton: visibilidadBoton
            });
          }
        });
      return;
    }
    this.setState({
      tipoContrato: tipoContrato,
      mostrarAlertaActividad: false,
      visibilidadBoton: false,
    });
  };

  /**
   * Método encargado de mostrar el nombre del punto de consumo.

  /**
   * Método encargado de convertir la fecha del desvio a formato DD/MM/YYYY
   * @param {number} fechaNumero Fecha parseada a número
   * @returns {string}
   */
  obtenerFecha = (fechaNumero) => {
    let fecha = new Date(fechaNumero);
    const anio = fecha.getFullYear();
    const dia = fecha.getDate();
    const mes = fecha.getMonth();
    fecha = `${anio}/${(mes < 10) ? '0' : ''}${mes + 1}/${(dia < 10) ? '0' : ''}${dia}`;
    return fecha;
  };

  /**

   * @param {number} posicionC Posición en la lista.

    const { listaCalculo } = this.state;
    let total = 0;

  /**
   * Método encargado de procesar el total para agregarlo a la lista
   */
  procesarTotal = (total) => {
    const objeto = {};
    let division = total.split(',');
    const ciudad = division[0].slice(1);
    const totalFinal = division[1].substr(0, (division[1].length - 1));
    objeto.ciudad = ciudad;
    objeto.totalFinal = totalFinal;
    return objeto;

   * @param {Object} datosPuntoConsumo Objecto con los datos del punto de consumo seleccionado

  consultarNominacion = (datosPuntoConsumo) => {
    const { listaPuntosSeleccionados, listaTotales } = this.state;
    let listaT = [];
    let objeto = {};
    listaT.push(datosPuntoConsumo);
    objeto.listaPuntoConsumo = listaT.map((dato, index) => {
      return {
        ptcIderegistro: dato.idPuntoConsumo.ptcIderegistro
      }
    })
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_CALCULO_NOMINACION.CONSULTAR_NOMINACION_PUNTO, objeto)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          const datos = respuesta.data.datos[0];
          datosPuntoConsumo.cantidadNominacion = datos.nominacion
          const total = this.procesarTotal(datos.total);
          listaTotales.push(total);
          listaPuntosSeleccionados.push(datosPuntoConsumo);
          this.setState({
            listaPuntosSeleccionados: listaPuntosSeleccionados,

        const p = { ...punto };
        if (index == indexPunto) {
          p.distribucion = p.distribucion.map((dis, indexDist) => {

            if (indexDist == indexDistribucion) {
              distribucion.nominacion = control.value;

            return distribucion;
