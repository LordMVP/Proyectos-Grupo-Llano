const ESTADO_FINALIZADO = 'F';
const ESTADO_LIQUIDADO = 'L';
const ESTADO_ACTIVO = 'A';
const SUMINISTRO = 'S';

const listaTipoRenumeracion = [
  { texto: 'Plana', id: 'PL' },
  { texto: 'Ponderada', id: 'PN' },
];

const listaTiposMercado = [
  { texto: 'Regulado', id: 'REG' },
  { texto: 'No Regulado', id: 'NREG' },
];

  { texto: 'SI', id: 'S' },
  { texto: 'NO', id: 'N' },

class GestionLiquidacionSuministro extends Component {
  constructor(props) {
    super(props);
    this.referencia = React.createRef();
  }

  state = {
    // Datos de la entidad
    periodoInicial: '',
    periodoFinal: '',
    tipoRenumeracion: '',
    regulado: '',
    liquidacionCertificada: '',
    noRegulado: '',
    periodoLiquidar: '',
    tipoMercado: '',
    contratoDetalle: '',
    //Listas de la entidad
    listaTiposMercado: [],
    listaUnidadMedida: [],
    listaContratos: [],
    listaEstados: [],
    listaTipoContrato: [],
    listaContratosSinDetalle: [],
    listaContratosLiquidados: [],
    listaConsulta: [],
    objetoCalculo: {},
    //Estados de la entidad
    consultado: false,
    mostrarModalConsulta: false
  };

  /**
   * Método encargado de comprobar si el formulario ya cargo
   */
  componentDidMount() {
    const { state } = this.props.history && this.props.history.location;
    if (state && state.listaContratos) {
      this.setState({
        listaContratosSinDetalle: state.listaContratos
      });
    }
    const peticiones = [
      axios.post(RUTAS_API.CONFIGURACION.CONSULTAR_UNIDAD, { criterio: '', idClase: CLASES_UNIDADES.ESTADOS_CONTRATO }),
      axios.post(RUTAS_API.CONFIGURACION.CONSULTAR_UNIDAD, { criterio: '', idClase: CLASES_UNIDADES.TIPO_CONTRATO }),
      axios.post(RUTAS_API.PARAMETRIZACION.UNIDADES_MEDIDA.CONSULTAR_POR_ESTRUCTURA, { criterio: '', categoria: TIPOS_UNIDADES_MEDIDA.MONEDA })

    ];
    axios.all(peticiones)
      .then(axios.spread((estados, tipos, unidades) => {
        const appData = {
          listaEstados: [],
          listaTipoContrato: [],
          listaUnidadMedida: []
        };
        if (estados.data.codigo > 0) {
          appData.listaEstados = estados.data.datos;
        }
        if (tipos.data.codigo > 0) {
          appData.listaTipoContrato = tipos.data.datos;
        }
        if (unidades.data.codigo > 0) {
          appData.listaTipoContrato = tipos.data.datos;
        }
        this.setState({ ...appData });
      }));
  };

  /**
   * Método encargado de validar los datos para consultar el detalle.
   * @returns {Object}
   */
  validarDetalle = () => {
    const { periodoInicial, periodoFinal, listaContratosSinDetalle } = this.state;
    if (!periodoInicial || periodoInicial == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar periodo inicial.' } };
    }
    if (!periodoFinal || periodoFinal == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el periodo final.' } };
    }
    if (!Util.validarArreglo(listaContratosSinDetalle)) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar al menos un contrato.' } };
    }

    return { respuesta: true };
  };

  /**
   * Método encargado de consultar los datos necesarios del contrato.
   * @returns {boolean}
   */
  consultarDetalleContrato = async () => {
    const validacion = this.validarDetalle();
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return false;
    }
    const { periodoInicial, periodoFinal } = this.state;
    const contratosSeleccionados = [...this.state.listaContratosSinDetalle];
    const idsContrato = contratosSeleccionados.map(dato => dato.cntIderegistro);
    const parametros = {
      fechaInicio: periodoInicial,
      fechaFinal: periodoFinal,
      listaContrato: idsContrato
    };
    const respuesta = await axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_SUMINISTRO_NEGOCIACION_DIRECTA.CONSULTAR_DETALLE_CONTRATO, parametros);
    if (respuesta.data.codigo > 0) {
      let data = JSON.parse(respuesta.data.datos);
      if (data == null) {
        toast.info('No se encontrarón datos del tercero de los contratos seleccionados');
        data = [];
      }
      this.setState({
        listaContratos: data
      });
    }
  };

  /**
   * Método encargado de procesar el detalle para agregar el precio del contrato
   * @param {Array} datosLiquidacion Datos del objeto liquidación
   * @returns {Array}
   */
  procesarData = (datosLiquidacion) => {
    const { listaContratosSinDetalle } = this.state;
    for (let index = 0; index < listaContratosSinDetalle.length; index++) {
      const contratoSinDetalle = listaContratosSinDetalle[index];
      datosLiquidacion.forEach(c => {
        if (c.contrato.cntIderegistro == contratoSinDetalle.cntIderegistro) {
          c.contrato.cntPrecio = contratoSinDetalle.cntPrecio;
        }
      });
    }
    return datosLiquidacion;
  };

  /**
   * Método encargado ejecutar una acción cuando se elimina el componente
   */
  componentWillUnmount() {
    this.props.history.replace({ entidadEditar: null });
  };

  /**
   * Método encargado de limpiar los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  limpiarFormulario = (evento) => {
    this.setState({
      mostrarModalConsulta: false,
      periodoInicial: '',
      periodoFinal: '',
      tipoRenumeracion: '',
      tipoMercado: '',
      regulado: '',
      noRegulado: '',
      periodoLiquidar: '',
      listaContratos: [],
      listaContratosSinDetalle: [],
      listaContratosLiquidados: [],
      listaConsulta: [],
      objetoCalculo: {},
      contratoDetalle: '',
      liquidacionCertificada: '',
      consultado: false,
      mostrarModalConsulta: false
    });
    limpiarHistorico(this.props);
  };

  /**
   * Método encargado de limpiar el formulario al momento de salir
   */
  componentWillUnmount() {
    this.limpiarFormulario();
  };

  /**
   * Método encargado de generar los botones del formulario
   * @returns {Object}
   */
  obtenerFunciones = () => {
    let funciones = [{ texto: 'Limpiar', callback: this.limpiarFormulario }];
    funciones.push({ texto: 'Consultar', callback: this.consultarEntidad });
    if (!this.state.consultado) {
      funciones.push(
        { texto: 'Guardar', callback: this.guardarEntidad },
        { texto: 'Consultar Detalle', callback: this.consultarDetalleContrato }
      );
    }
    return funciones;
  };

  /**
   * Método encargado de validar las variables del formulario
   * @returns {Object}
   */
  validarFormulario = () => {
    const { objetoCalculo, listaContratosLiquidados, liquidacionCertificada } = this.state;
    // Ejemplo Validacion
    if (!Util.validarArreglo(listaContratosLiquidados)) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe liquidar los contratos' } };
    }

    if (objetoCalculo == null || Object.keys(objetoCalculo).length === 0) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe realizar el calculo' } };
    }

    if (!liquidacionCertificada || liquidacionCertificada == 'N') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe certificar la liquidación' } };
    }


    return { respuesta: true };
  };

  /**
   * Método encargado de extraer el id estado de los tipos de contrato FINALIZADO, ACTIVO y LIQUIDADO
   * @returns {Array}
   */
  obtenerEstados = () => {
    const { listaEstados } = this.state;
    if (!Util.validarArreglo(listaEstados)) {
      return 0;
    }
    const estados = listaEstados.filter(e =>
      (JSON.parse(e.uniPropiedad).estado == ESTADO_ACTIVO) ||
      (JSON.parse(e.uniPropiedad).estado == ESTADO_FINALIZADO) ||
      (JSON.parse(e.uniPropiedad).estado == ESTADO_LIQUIDADO));
    if (!Util.validarArreglo(estados)) {
      return 0;
    }
    return estados.map(dato => (
      dato.uniIderegistro
    ));
  };

  /**
   * Método encargado de abrir la itnerfaz de consulta de contratos
   * @returns {boolean}
   */
  consultarContrato = () => {
    const estados = this.obtenerEstados();
    const tipos = this.state.listaTipoContrato.filter(p =>
      (JSON.parse(p.uniPropiedad).tipocontrato == SUMINISTRO)).map(dato => JSON.parse(dato.uniPropiedad).tipocontrato);
    if (estados === 0) {
      this.props.mostrarAlerta('Error de Configuración', 'Debe configurar los estados del programa para poder ejecutar la petición solicitada');
      return;
    }
    this.props.history.push({
      pathname: RUTAS_VISTA.CONSULTA_CONTRATOS.url,
      state: {
        interfazGestion: RUTAS_VISTA.GESTION_LIQUIDACION_FACTURAS_SUMINISTRO_NEGOCIACION_DIRECTA.url,
        seleccionMultiple: true,
        estadosContrato: estados,
        tiposContrato: tipos,
        ocultarFiltros: true,
        autoconsultar: true,
      }
    });
  };

  /**
   * Método encargado de validar los valores de la tabla calculo diario
   * @returns {Boolean}
   */
  validarTablaCalculo = () => {
    const { objetoCalculo } = this.state;
    for (let index = 0; index < objetoCalculo.listaValores.length; index++) {
      const calculo = objetoCalculo.listaValores[index];
      if (!calculo.precioDelDia || calculo.precioDelDia == '') {
        toast.error('El precio del contrato número : ' + calculo.contrato.cntNumero + " no puede ser vacío");
        return false
      }
      if (!calculo.cantidadContratada || calculo.cantidadContratada == '') {
        toast.error('La cantidad contratada del contrato número : ' + calculo.contrato.cntNumero + " no puede ser vacía");
        return false
      }
    }
    return true;
  }

  /**
   * Método encargado de obtener el objeto guardar
   * @returns {Array}
   */
  obtenerObjetoGuardar = () => {
    const { periodoFinal, periodoInicial, listaContratosLiquidados,
      objetoCalculo, periodoLiquidar, cruceSaldo,
      liquidacionCertificada, tipoMercado, tipoRenumeracion,
      regulado, noRegulado } = this.state;
    const partesPeriodo = periodoLiquidar.split('-');
    let listaFinal = [];
    for (let index = 0; index < listaContratosLiquidados.length; index++) {
      const contratoLiquidado = listaContratosLiquidados[index];
      const contratoCalculado = objetoCalculo.listaValores.find(c => c.contrato.cntIderegistro == contratoLiquidado.contrato.cntIderegistro);
      let objeto = {
        rangoFechas: {
          fechaInicio: periodoInicial,
          fechaFinal: periodoFinal
        },
        gestorMercado: {
          contrato: {
            cntIderegistro: contratoLiquidado.contrato.cntIderegistro,
            cntNumero: contratoLiquidado.contrato.cntNumero,
            cntPrecio: contratoLiquidado.contrato.cntPrecio,
            terIdeagente: {
              terNomcompleto: contratoLiquidado.contrato.terIdeagente.terNomcompleto,
              terIderegistro: contratoLiquidado.contrato.terIdeagente.terIderegistro,
              terCorreo: contratoLiquidado.contrato.terIdeagente.terCorreo,
            }
          },
          anio: partesPeriodo[0],
          mes: partesPeriodo[1],
          fecha: `${this.state.periodoLiquidar}-01`,
          cantidadContratadaRegulado: contratoLiquidado.cantidadContratadaRegulado,
          cantidadContratadaNoRegulado: contratoLiquidado.cantidadContratadaNoRegulado,
          planaTarifaMbtu: contratoLiquidado.planaTarifaMbtu,
          renumeracionCalculadaContrato: contratoLiquidado.renumeracionCalculadaContrato,
          cruzarSaldos: cruceSaldo,
          liquidacionCertificada: liquidacionCertificada,
          tipoRenumeracion: tipoRenumeracion,
          tipoMercado: (tipoRenumeracion == 'PN') ? null : tipoMercado,
          valorRegulado: (!regulado || regulado == '') ? null : regulado,
          valorNoRegulado: (!noRegulado || noRegulado == '') ? null : noRegulado
        },
        valorDiario: {
          contrato: {
            cntIderegistro: contratoCalculado.contrato.cntIderegistro,
            cntNumero: contratoCalculado.contrato.cntNumero,
            cntPrecio: contratoCalculado.contrato.cntPrecio,
            terIdeagente: {
              terNomcompleto: contratoCalculado.contrato.terIdeagente.terNomcompleto,
              terIderegistro: contratoCalculado.contrato.terIdeagente.terIderegistro,
            }
          },
          claseContrato: contratoCalculado.claseContrato,
          cantidadContratada: contratoCalculado.cantidadContratada,
          precioDelDia: contratoCalculado.precioDelDia,
          valorDiario: contratoCalculado.valorDiario,
          subtotalFirme: objetoCalculo.subtotalFirme,
          totalPrepagos: objetoCalculo.totalPrepagos,
          saldosAFavor: objetoCalculo.saldosAFavor,
          totalFactura: objetoCalculo.totalFactura,
          valorTRM: contratoCalculado.valorTRM,
        }
      };
      listaFinal.push(objeto);
    }
    return listaFinal;
  };

  /**
   * Método encargado de guardar los datos de la entidad
   * @returns {bool}
   */
  guardarEntidad = () => {
    const validacion = this.validarFormulario();
    const validacionTabla = this.validarTablaCalculo();
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return false;
    }
    if (!validacionTabla) {
      return false;
    }
    const entidadGuardar = this.obtenerObjetoGuardar();
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_SUMINISTRO_NEGOCIACION_DIRECTA.GUARDAR, entidadGuardar)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.limpiarFormulario();
        }
      });
  };

  /**
   * Método encargado de abrir la ventana modal del boton consulta
   */
  consultarEntidad = () => {
    this.setState({ mostrarModalConsulta: true });
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    this.setState(change);
  };

   * Método encargado de eliminar el contrato seleccionado

  eliminarContrato = (posicion) => {
    const lista = [...this.state.listaContratos];
    lista.splice(posicion, 1);
    this.setState({ listaContratos: lista });
  };

  /**
   * Método encargado de cerrar la ventana modal del boton consulta

  abrirCerrarModal = () => {
    this.setState({
      mostrarModalConsulta: false
    });

  /**
  * Método encargado de convertir la fecha a formato YYYY/MM/DD
  * @returns {String}
  */
  formatearFecha = (timestamp) => {
    const fecha = new Date(timestamp);
    const anio = fecha.getFullYear();
    let mes = (1 + fecha.getMonth()).toString();
    mes = mes.length > 1 ? mes : '0' + mes;
    let dia = fecha.getDate().toString();
    dia = dia.length > 1 ? dia : '0' + dia;
    return anio + '-' + mes + '-' + dia;
  };

  /**
   * Método encargado de mostrar la tabla de los contratos seleccionados para calcular el valor diario
   * @returns {Object}
   */
  renderTablaValorDiario = () => {
    const objeto = { ...this.state.objetoCalculo };
    if (objeto == null || Object.keys(objeto).length === 0) {
      return;
    }
    return (
      <div className='row mt-5'>
        <div className='col-12'>
          <table className='table table-striped table-condensed table-bordered table-striped'>

              <tr>
                <th>Contrato</th>
                <th>Fecha Final</th>
                <th>Clase de contrato</th>
                <th>Cantidad Contratada</th>
                <th>Precio Día</th>
                <th>Valor Diario</th>
                <th>Subtotal Firme</th>
                <th>Total Prepagos</th>
                <th>Saldos a Favor</th>
                <th>Total de la Factura</th>
              </tr>
            </thead>
            <tbody>
              {

                      <td>
                        <TextoNumerico

                          aceptaNegativos={false}
                          cols={12}
                          value={dato.precioDelDia}

                          name='precioDelDia'
                        />
                      </td>
                      <td>
                        <TextoNumerico
                          aceptaDecimales={false}
                          aceptaNegativos={false}

                          name='valorDiario'
                        />
                      </td>
                      <td>

                          aceptaDecimales={false}
                          aceptaNegativos={false}
                          cols={12}
                          value={this.obtenerValores(objeto)}

                          extra={{ disabled: true }}
                          name='subtotalFirme'
                        />
                      </td>
                      <td>{objeto.totalPrepagos}</td>
                      <td>{objeto.saldosAFavor}</td>
                      <td>
                        {this.obtenerValores(objeto, 'TOTAl')}
                      </td>

                  );
                })
              }
            </tbody>
          </table>
        </div>
        {this.renderAprobacion()}

    );

  /**
   * Método encargado de controlar el cambio en la tabla de valores diarios
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambioTabla = (evento) => {
    let objeto = { ...this.state.objetoCalculo };
    const control = evento.target;
    const { name, value } = control;
    const index = control.attributes['data-index'].value;
    objeto.listaValores[index][name] = value;
    objeto.listaValores[index].valorDiario = (objeto.listaValores[index].precioDelDia * objeto.listaValores[index].valorTRM * objeto.listaValores[index].cantidadContratada);
    this.setState({ objetoCalculo: objeto });
  }

  /**
   * Método encargado de calcular el total si se cambian los valores
   * @param {Object} datos Datos de la tabla calculo diario
   * @param {String} tipo Variable a devolver
   */
  obtenerValores = (datos, tipo = '', index = null) => {
    let remuneración = 0.0;
    this.state.listaContratosLiquidados.forEach(v => {
      remuneración += v.renumeracionCalculadaContrato;
    });
    if (tipo == 'VALORDIARIO') {
      return (datos.listaValores[index].precioDelDia * datos.listaValores[index].valorTRM * datos.listaValores[index].cantidadContratada);
    }
    let subtotalFirme = 0.0;
    datos.listaValores.forEach(v => {
      subtotalFirme += v.valorDiario;
    });
    if (tipo == '') {
      return subtotalFirme;
    }
    if (datos.saldosAFavor != null) {
      return subtotalFirme + remuneración - datos.saldosAFavor;
    }
    return subtotalFirme + remuneración;
  };

  /**
   * Método encargado de validar el formulario para liquidar
   * @returns {Object}
   */
  validarLiquidar = () => {
    const { tipoRenumeracion, tipoMercado, regulado, noRegulado, periodoLiquidar } = this.state;
    if (!tipoRenumeracion || tipoRenumeracion == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el tipo de renumeración.' } };
    }
    if (tipoRenumeracion == 'PL') {
      if (!tipoMercado || tipoMercado == '') {
        return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el tipo de mercado.' } };
      }
      if (tipoMercado == 'REG') {
        if (!regulado || regulado == '') {
          return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el valor del mercado regulado.' } };
        }
      }
      if (tipoMercado == 'NREG') {
        if (!noRegulado || noRegulado == '') {
          return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el valor del mercado no regulado.' } };
        }
      }
    }
    if (tipoRenumeracion == 'PN') {
      if (!regulado || regulado == '') {
        return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el valor del mercado regulado.' } };
      }
      if (!noRegulado || noRegulado == '') {
        return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el valor del mercado no regulado.' } };
      }
    }
    if (!periodoLiquidar || periodoLiquidar == '') {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar el periodo a liquidar.' } };
    }
    return { respuesta: true }
  };

  /**
   * Método encargado de liquidar los contratos seleccionados
   * @returns {Boolean}
   */
  liquidar = () => {
    const { listaContratos, tipoRenumeracion, tipoMercado, regulado, noRegulado } = this.state;
    const validacionLiquidar = this.validarLiquidar();
    if (!validacionLiquidar.respuesta) {
      toast.error(validacionLiquidar.mensaje.mensaje);
      return;
    }
    let contratos = listaContratos.filter(c => c.seleccionado).map(c => {
      return {
        "cntIderegistro": c.contrato.cnt_ideregistro,
        "cntNumero": c.contrato.cnt_numero,
        "terIdeagente": {
          "terNomcompleto": c.ter_tercero.ter_nomcompleto,
          "terIderegistro": c.ter_tercero.ter_ideregistro,
          "terCorreo": c.ter_tercero.ter_correo,
        }
      };
    });
    const partesPeriodo = this.state.periodoLiquidar.split('-');
    const obj = {
      listaContrato: contratos,
      tipoRenumeracion: tipoRenumeracion,
      tipoMercado: tipoMercado,
      valorRegulado: regulado,
      valorNoRegulado: noRegulado,
      anio: partesPeriodo[0],
      mes: partesPeriodo[1],
      fecha: `${this.state.periodoLiquidar}-01`
    };
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_SUMINISTRO_NEGOCIACION_DIRECTA.LIQUIDAR, obj)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          const obj = respuesta.data.datos;
          this.setState({ listaContratosLiquidados: this.procesarData(obj) });
        }
      });
  };

  /**
   * Método encargado de mostrar el formulario para la remuneración
   * @returns {Object}
   */
  renderRemuneracion = () => {
    return (
      <div className='conf-general row mt-5'>
        <Combo
          opciones={listaTipoRenumeracion}
          propTexto='texto'
          propValor='id'
          label='Tipo de Renumeración:'
          name='tipoRenumeracion'
          value={this.state.tipoRenumeracion}
          onChange={this.controlarCambio}
        />
        {this.state.tipoRenumeracion === 'PN' &&
          <div className='row'>
            <TextoNumerico
              aceptaDecimales={false}
              aceptaNegativos={false}
              label='Regulado:'
              value={this.state.regulado}
              onChange={this.controlarCambio}
              name='regulado'
            />
            <TextoNumerico
              aceptaDecimales={false}
              aceptaNegativos={false}
              label='No Regulado:'
              value={this.state.noRegulado}
              onChange={this.controlarCambio}
              name='noRegulado'
            />
            {/* <Combo
              opciones={this.state.listaUnidadMedida}
              propTexto='uniNombre1'
              propValor='uniIderegistro'
              label='Unidad de Medida:'
              name='unidadMedida'
              value={this.state.unidadMedida}
              onChange={this.controlarCambio}
            /> */}
          </div>
        }
        {this.state.tipoRenumeracion === 'PL' &&
          <Combo
            propTexto='texto'
            propValor='id'
            opciones={listaTiposMercado}
            label='Tipo de Mercado:'
            name='tipoMercado'
            value={this.state.tipoMercado}
            onChange={this.controlarCambio}
          />
        }
        {
          this.state.tipoMercado === 'REG' &&
          <TextoNumerico
            aceptaDecimales={false}
            aceptaNegativos={false}
            label='Regulado:'
            value={this.state.regulado}
            onChange={this.controlarCambio}
            name='regulado'
          />
        }
        {
          this.state.tipoMercado === 'NREG' &&
          <TextoNumerico
            aceptaDecimales={false}
            aceptaNegativos={false}
            label='No Regulado:'
            value={this.state.noRegulado}
            onChange={this.controlarCambio}
            name='noRegulado'
          />
        }
        <Fecha
          label='Periodo a Liquidar'
          name='periodoLiquidar'
          sinDia={true}
          fecha={this.state.periodoLiquidar}
          onChange={this.controlarCambio}
        />
        <div className='form-group m-t-24'>
          <button className='btn btn-primary' onClick={this.liquidar}><i className='fa fa-fw fa-check'></i> Liquidar</button>
        </div>
      </div>
    );
  };

  /**
   * Controla el evento change de los checks de la tabla de contratos.
   * @returns {boolean}
   */
  controlarCambioCheckTablaContratos = (evento) => {
    const contratos = [...this.state.listaContratos];
    if (evento.target.name == 'checkSeleccionarTodos') {
      contratos.forEach(contrato => {
        contrato.seleccionado = evento.target.checked;
      });
      this.setState({ listaContratos: contratos });
      return;
    }
    const value = evento.target.value;
    const index = contratos.findIndex(c => c.contrato.cnt_ideregistro == value);
    const contratoActual = { ...contratos[index] };
    contratoActual.seleccionado = evento.target.checked;
    contratos[index] = contratoActual;
    this.setState({ listaContratos: contratos });
  };

  /**
   * Método encargado de para obtener la fecha y retornar en formato String.
   * @param {String} fecha fecha a convertir.
   * @returns {String}
   */
  obtenerFecha = (fecha) => {
    if (!fecha || fecha == null) {
      return '';
    }
    fecha = new Date(fecha);
    fecha = (fecha.getFullYear() + "-" + (fecha.getMonth() + 1) + "-" + fecha.getDate());
    return fecha;
  };

        </tbody>
      </table>

  /**
   * Método encargado de hacer scroll a la pantalla al momento
   */
  scroll = () => {
    if (this.referencia.current) {
      this.referencia.current.scrollIntoView({
        behavior: "smooth",
        block: "nearest"

  }

  /**
   * Método encargado de controlar el cambio de la cantidad contratada en el detalle
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambioCantidadContratada = (evento) => {
    const { contratoDetalle, listaContratos } = this.state;
    const indexLista = listaContratos.findIndex(c => c.contrato.cnt_ideregistro == contratoDetalle);
    const control = evento.target;
    const index = control.attributes['data-index'].value;
    listaContratos[indexLista].listaDetalle[index].cantidadContratada = control.value;
    this.setState({ ...listaContratos });
  };

  /**

   * @returns {JSX}

  renderListaDetalle = () => {
    const { contratoDetalle, listaContratos } = this.state;
    const detalleRender = listaContratos.find(c => c.contrato.cnt_ideregistro == contratoDetalle);
    return (
      <div className='table table-responsive' ref={this.referencia}>

            <tr className='bg-dark text-white'>
              <th>Fecha</th>
              <th>Cantidad Contratada</th>
              <th>Unidad Medida</th>
            </tr>
          </thead>
          <tbody>
            {
              detalleRender.listaDetalle.map((dato, index) => {
                return (

                    <td>{dato.fecha}</td>
                    <td>
                      <TextoNumerico
                        aceptaDecimales={false}
                        aceptaNegativos={false}
                        cols={12}
                        value={dato.cantidadContratada}
                        onChange={this.controlarCambioCantidadContratada}
                        name='cantidadContratada'
                        extra={{ 'data-index': index, disabled: true }}
                      />
                    </td>
                    <td>{dato.unidad.uniNombre1}</td>
                  </tr>
                )
              })
            }
          </tbody>
        </table>
        <button className='btn btn-primary' onClick={() => { this.setState({ contratoDetalle: '' }) }}>Ocultar Detalle</button>
      </div>
    )
  };

  /**
   * Método encargado de consultar el detalle día a día del contrato seleccionado
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  verDetalle = (evento) => {
    const { periodoFinal, periodoInicial, listaContratos } = this.state;
    const idContrato = evento.target.attributes['data-id'].value;
    const obj = {
      fechaInicial: periodoInicial,
      fechaFinal: periodoFinal,
      idContrato: idContrato
    };
    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_SUMINISTRO_NEGOCIACION_DIRECTA.CONSULTAR_DIA_DIA, obj)
      .then(respuesta => {
        const index = listaContratos.findIndex(c => c.contrato.cnt_ideregistro == idContrato);
        const data = respuesta.data.datos;
        listaContratos[index].listaDetalle = data;
        this.setState({
          contratoDetalle: idContrato,
          listaContratos: listaContratos
        }, this.scroll);
      });
  };

  /**
   * Método encargado de realizar el calculo del valor diario
   * @returns {Boolean}
   */
  calcular = async () => {
    const { cruceSaldo, listaContratosLiquidados, periodoLiquidar } = this.state;
    const partesPeriodo = periodoLiquidar.split('-');
    if (!cruceSaldo || cruceSaldo == '') {
      toast.error('Debe seleccionar si se cruzaran saldos');
      return;
    }
    let objeto = listaContratosLiquidados.map(c => {
      return {
        "contrato": {
          "cntIderegistro": c.contrato.cntIderegistro,
          "cntNumero": c.contrato.cntNumero,
          "cntPrecio": c.contrato.cntPrecio,
          "terIdeagente": {
            "terNomcompleto": c.contrato.terIdeagente.terNomcompleto,
            "terIderegistro": c.contrato.terIdeagente.terIderegistro,
            "terCorreo": c.contrato.terIdeagente.terCorreo,
          }
        },
        "anio": partesPeriodo[0],
        "mes": partesPeriodo[1],
        "fecha": `${this.state.periodoLiquidar}-01`,
        "planaTarifaMbtu": c.planaTarifaMbtu,
        "cantidadContratadaRegulado": c.cantidadContratadaRegulado,
        "cantidadContratadaNoRegulado": c.cantidadContratadaNoRegulado,
        "renumeracionCalculadaContrato": c.renumeracionCalculadaContrato,
        "cruzarSaldos": cruceSaldo
      };
    });
    const respuesta = await axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_FACTURAS_SUMINISTRO_NEGOCIACION_DIRECTA.CALCULAR_VALOR_DIARIO, objeto);
    if (respuesta.data.codigo > 0) {
      this.setState({ objetoCalculo: respuesta.data.datos });
    }
  };

  renderFormularioCruce = () => {

  renderAprobacion = () => {

  /**
   * Método encargado de procesar los datos de la liquidacion seleccionada
   * @param {Object} entidad Datos de la entidad seleccionada
   */
  seleccionarEntidad = (entidad) => {
    let { listaConsulta } = this.state;
    listaConsulta = [];
    listaConsulta.push(entidad);
    this.setState({
      listaConsulta: listaConsulta,
      consultado: true,
      mostrarModalConsulta: false,
    });
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <Fragment>
        <Botonera funciones={this.obtenerFunciones()} />
        <div className='conf-general row mt-5'>
          {!this.state.consultado &&
            <div className='form-group col-4'>
              <label>Contrato:</label>
              <div className='input-group'>
                <input type='text' disabled={true} className='form-control' placeholder='Buscar contrato' value={(Util.validarArreglo(this.state.listaContratosSinDetalle)) ? this.state.listaContratosSinDetalle.length + ' contratos seleccionados' : ''} />
                <button className='btn btn-primary' onClick={this.consultarContrato}><i className='fa fa-fw fa-search'></i></button>
              </div>
            </div>
          }
          {!this.state.consultado &&
            <Fecha
              label='Fecha Inicial'
              name='periodoInicial'
              fecha={this.state.periodoInicial}
              onChange={this.controlarCambio}
            />
          }
          {!this.state.consultado &&
            <Fecha
              label='Fecha Final'
              name='periodoFinal'
              fecha={this.state.periodoFinal}
              onChange={this.controlarCambio}
            />
          }
          {Util.validarArreglo(this.state.listaConsulta) &&
            this.renderTablaConsulta()
          }
        </div>

        {this.state.listaContratos.length > 0 &&
          this.renderTablaContratosSeleccionados()
        }
        {this.state.listaContratos.length > 0 &&
          this.renderRemuneracion()
        }
        {this.state.contratoDetalle != '' &&
          this.renderListaDetalle()
        }
        {Util.validarArreglo(this.state.listaContratosLiquidados) && this.renderContratosLiquidados()}
        {this.state.objetoCalculo != null && this.renderTablaValorDiario()}

        <VentanaModal
          mostrar={this.state.mostrarModalConsulta}
          titulo='Consultar Liquidación'
          cerrarModal={this.abrirCerrarModal}>
          <RConsultaLiquidacionSuministro
            esModal
            seleccionarEntidad={this.seleccionarEntidad}
          />
        </VentanaModal>
      </Fragment>
    );
  };
}
