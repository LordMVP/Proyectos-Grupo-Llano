  /**
   * Método encargado de obtener el precio del contrato dependiendo del rango
   * @param {Object} rango Datos del rango
   */
  obtenerPrecioContrato = (rango) => {
    let valores = {
      precioContrato: 0,
      valorCargos: {
        variable: 0,
        fijo: 0,
        aom: 0
      },
      valorPorcentaje: 0,
    };
    if (!Array.isArray(rango.tramos) || rango.tramos.length == 0) {
      return valores;
    }
    //Recorremos los tramos y sumamos los valores.
    const tramos = rango.tramos;
    for (let i = 0; i < tramos.length; i++) {
      const tramo = tramos[i];
      const valorCargoVariable = this.obtenerValorCargo(tramo, 'V');
      const valorCargoFijo = this.obtenerValorCargo(tramo, 'F');
      const valorCargoAOM = this.obtenerValorCargo(tramo, 'A');
      valores.valorCargos.variable += valorCargoVariable;
      valores.valorCargos.fijo += valorCargoFijo;
      valores.valorCargos.aom += valorCargoAOM;
      valores.valorPorcentaje += parseFloat(tramo.valorPorcentaje);
      valores.precioContrato += (valorCargoVariable + valorCargoFijo + valorCargoAOM);
    }
    return valores;
  };

  /**
   * Método encargado de controlar el cambio en los rangos de la ruta
   * @param {Event} evento Evento ejecutado en el control de usuario
   */
  controlarCambioRangoRuta = (evento) => {
    // Controlar cambio
    const control = evento.target;
    const valor = control.value;
    const nombrePropiedad = control.name;
    const idRango = control.attributes['data-rango'].value;
    const idSubRango = control.attributes['data-subrango'].value;
    const idRuta = control.attributes['data-ruta'].value;
    const canastaTransporte = this.props[this.props.nombrePropiedadCanasta];
    const nuevosRangos = canastaTransporte.canastasConRutas.map((r) => {
      const rango = { ...r };
      //Buscamos el rango...
      if (rango.idRangoConRuta === idRango) {
        //Buscamos el subrango.
        rango.subRangos = rango.subRangos.map((sr) => {
          const subRango = { ...sr };
          if (subRango.idSubRango == idSubRango) {
            //Buscamos la ruta...
            subRango.rutas = subRango.rutas.map((rt) => {
              const ruta = { ...rt };
              if (ruta.idRuta == idRuta) {
                ruta[nombrePropiedad] = valor;
              }
              return ruta;
            });
          }
          return subRango;
        });
      }
      return rango;
    });

    // if (nombrePropiedad === 'valorPorcentaje') {
    const rango = { ...nuevosRangos[0] };
    const valores = this.obtenerPrecioContratoPorRutas(rango);
    // const valorPorcentaje = valores.valorPorcentaje;
    this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, precioCanasta: valores.precioContrato, porcentajeComercializacion: valores.valorPorcentaje });
    // }
    this.actualizarEstadoRedux({ canastasConRutas: nuevosRangos });
  };

  /**
   * Método encargado de obtener el precio del contrato por las rutas
   * @param {Object} rango Datos del rango
   */
  obtenerPrecioContratoPorRutas = (rango) => {
    let valores = {
      precioContrato: 0,
      valorPorcentaje: 0,
    };
    if (Util.validarArreglo(rango.subRangos) && !Util.validarArreglo(rango.subRangos[0].rutas)) {
      //Obtenemos los valores de los valores de los subrangos...
      const subRangos = rango.subRangos;
      if (!Util.validarArreglo(subRangos)) {
        return valores;
      }
      valores.precioContrato = subRangos[0].valor;
      return valores;
    }
    const subRango = rango.subRangos[0];
    valores.precioContrato = subRango.rutas[0].valor;
    valores.valorPorcentaje = subRango.rutas[0].valorPorcentaje;
    return valores;
  };

    const control = evento.target;
    const valor = control.value;
    const nombrePropiedad = control.name;
    const idRango = control.attributes['data-rango'].value;
    const idSubRango = control.attributes['data-subrango'].value;
    const idTramo = control.attributes['data-tramo'].value;
    const canastaTransporte = this.props[this.props.nombrePropiedadCanasta];
    const nuevosRangos = canastaTransporte.canastasSinRutas.map((r) => {

        //Buscamos el subrango.
        rango.subRangos.map((subRango) => {
          if (subRango.idSubRango == idSubRango) {
            //Buscamos los tramos del rango.
            subRango.tramos = subRango.tramos.map((t) => {
              const tramo = { ...t };

    const valores = this.obtenerPrecioContrato(rango);
    const valorPorcentaje = valores.valorPorcentaje;
    this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, precioCanasta: valores.precioContrato, valorCargos: valores.valorCargos, porcentajeComercializacion: valorPorcentaje });

  /**
   * Obtiene el valor cargo de un tramo.
   * @param {Object} tramo Datos del tramo
   * @param {String} tipo Tipo de cargo
   * @return {Number}
   */
  obtenerValorCargo = (tramo, tipo) => {
    //Verificamos el cargo seleccionado...
    if (!Util.validarArreglo(tramo.listaCargos)) {
      return -1;
    }
    var cargoSeleccionado = tramo.listaCargos.find(cargo => cargo.seleccionado);
    switch (tipo) {
      case 'V':
        return cargoSeleccionado.trcaCargovariable;
      case 'F':
        return cargoSeleccionado.trcaCargofijo;
      case 'A':
        return cargoSeleccionado.trcaCargoaoym;
    }
  };

  /**
   * Obtiene el valor de la unidad de medida del cargo
   * @param {Object} tramo Datos del tramo
   * @param {String} tipo Tipo de cargo
   * @return {String}
   */
  obtenerUnidadMedidaCargo = (tramo, tipo) => {
    //Verificamos el cargo seleccionado...
    if (!Util.validarArreglo(tramo.listaCargos)) {
      return -1;
    }
    var cargoSeleccionado = tramo.listaCargos.find(cargo => cargo.seleccionado);

      case 'V':

      case 'F':
        return cargoSeleccionado.uniIdemedidafijo.uniIderegistro;
      case 'A':
        return cargoSeleccionado.uniIdemedidaoym.uniIderegistro;
    }
  };

                              onChange={this.controlarCambioRangoSinRutasSinTramos}
                              extra={{ 'data-rango': objSelected.idRangoConRuta, 'data-subrango': subRango.idSubRango }}
                            />
                          </td>
                          <td className='input' style={style}>
                            <TextoNumerico
                              aceptaDecimales={false}
                              aceptaNegativos={false}
                              label='Valor'
                              cols={12}
                              value={subRango.valorPorcentaje}
                              onChange={this.controlarCambioRangoSinRutasSinTramos}
                              name='valorPorcentaje'
                              extra={{ 'data-rango': objSelected.idRangoConRuta, 'data-subrango': subRango.idSubRango }}
                            />
                          </td>
                        </Fragment>
                      )
                    })()
                  }
                </tr>
              </Fragment>
            );
          })
        }
      </tbody>
    );
  };

  /**
   * Método encargado de calcular el precio del contrato dado un rango
   * @param {Object} rango Datos del rango
   */
  calcularPrecioContrato = (rango, usaTramo = false) => {
    const valores = usaTramo ? this.obtenerPrecioContrato(rango) : this.obtenerPrecioContratoPorRutas(rango);
    this.props.actualizarCabeceraRedux({ precioContrato: valores.precioContrato, precioCanasta: valores.precioContrato, porcentajeComercializacion: valores.valorPorcentaje });
  };

  /**
   * Método encargado de mostrar la tabla de la canasta duplex con rutas y sin tramos
   * @returns {JSX}
   */
  renderTBodyDuplexConRutasSinTramos = () => {
    const listaCanasta = getProp(this.props, this.props.nombrePropiedadCanasta, { canastasConRutas: [] }).canastasConRutas;
    var rangos = listaCanasta.length;
    var rangosHorarios = listaCanasta[0].subRangos.length;
    var rutas = listaCanasta[0].subRangos[0].rutas.length;
    //filas a generar
    var rowsNumber = ((rangos === 0) ? 1 : rangos)
      * ((rangosHorarios === 0) ? 1 : rangosHorarios)
      * ((rutas === 0) ? 1 : rutas);

    var rowsRangos = ((rangosHorarios === 0) ? 1 : rangosHorarios)
      * ((rutas === 0) ? 1 : rutas);

    var rowsRangosHorarios = ((rutas === 0) ? 1 : rutas);
    var objSelected = listaCanasta[0];
    var numRango = -1;
    var numRangoHorario = -1;
    var numRuta = -1;

    let array = [];
    for (var index = 0; index < rowsNumber; index++) {
      array.push(index);
    }

    let inputs = 0;

    return (
      <tbody>
        {
          array.map((dato, index) => {
            return (
              <Fragment>
                <tr key={index}>
                  {
                    (index % rowsRangos === 0) && (() => {
                      numRango++;
                      numRangoHorario = -1;
                      numRuta = -1;
                      objSelected = listaCanasta[numRango];
                      if (numRango === 0) {
                        this.calcularPrecioContrato(objSelected);
                      }
                      return <td rowSpan={rowsRangos}>{`${objSelected.rango.horaInicio} - ${objSelected.rango.horaFin}`}</td>
                    })()
                  }
                  {
                    (index % rowsRangosHorarios === 0) && (() => {
                      numRangoHorario++;
                      numRuta = -1;
                      const subRango = objSelected.subRangos[numRangoHorario];
                      return <td rowSpan={rowsRangosHorarios}>{`${subRango.horaInicio} - ${subRango.horaFin}`}</td>
                    })()
                  }
                  {
                    (() => {
                      numRuta++;
                      const subRango = objSelected.subRangos[numRangoHorario];
                      const ruta = subRango.rutas[numRuta];
                      inputs = 4;
                      const style = { width: 'calc(60%/' + inputs + ')' };
                      return (
                        <Fragment>
                          <td className='input' style={style}>{ruta.uniNombre1}</td>
                          <td className='input' style={style}>
                            <Input
                              label='Valor'
                              cols={12}
                              value={ruta.valor}
                              onChange={this.controlarCambioRangoRuta}
                              name='valor'
                              extra={{ 'data-rango': objSelected.idRangoConRuta, 'data-ruta': ruta.idRuta, 'data-subrango': subRango.idSubRango }}
                            />
                          </td>
                          <td className='input' style={style}>
                            <Combo
                              opciones={this.props.listaUnidadesMedida}
                              propTexto='uniNombre1'
                              propValor='uniIderegistro'
                              name='unidadMedida'
                              cols={12}
                              value={ruta.unidadMedida}
                              onChange={this.controlarCambioRangoRuta}
                              extra={{ 'data-rango': objSelected.idRangoConRuta, 'data-ruta': ruta.idRuta, 'data-subrango': subRango.idSubRango }}
                            />
                          </td>
                          <td className='input' style={style}>
                            <TextoNumerico
                              aceptaDecimales={false}
                              aceptaNegativos={false}
                              label='Valor'
                              cols={12}
                              value={ruta.valorPorcentaje}
                              onChange={this.controlarCambioRangoRuta}
                              name='valorPorcentaje'
                              extra={{ 'data-rango': objSelected.idRangoConRuta, 'data-ruta': ruta.idRuta, 'data-subrango': subRango.idSubRango }}
                            />
                          </td>
                        </Fragment>
                      );
                    })()
                  }
                </tr>
              </Fragment>
            );
          })
        }
      </tbody>
    );
  };

  /**
   * Método encargado de mostrar el body de la tabla
   * @returns {JSX}
   */
  renderTBody = () => {
    //Verificamos como vamos a procesar los datos.
    if (this.props.usaRangoIntervalos && this.props.usaRangoHorario && this.props.usaTramos && !this.props.usaRutas) {
      return this.renderTBodyDuplexSinRutas();
    } else if (this.props.usaRangoIntervalos && this.props.usaRangoHorario && this.props.usaTramos && this.props.usaRutas) {
      return this.renderTBodyDuplexConRutas();
    } else if (!this.props.usaTramos && !this.props.usaRutas && this.props.usaRangoIntervalos && this.props.usaRangoHorario) {
      return this.renderTBodyDuplexSinRutasSinTramos();
    } else if (!this.props.usaTramos && this.props.usaRangoIntervalos && this.props.usaRangoHorario && this.props.usaRutas) {
      return this.renderTBodyDuplexConRutasSinTramos();
    } else if (!this.props.usaTramos && this.props.usaRangoIntervalos && this.props.usaRangoHorario && this.props.usaRutas) {

    }
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <table className='table table-condensed table-striped label-nodisplay'>
        <thead className='bg-dark color-white'>
          <tr>
            {(this.props.usaRangoIntervalos) && (<th>Rango</th>)}
            {(this.props.usaRangoHorario) && (<th>Rango Horario</th>)}
            {(this.props.usaRutas) && (<th>Ruta</th>)}
            {(this.props.usaTramos) && (<th>Tramo</th>)}
            {(this.props.usaTramos) && (
              <Fragment>
                <th>C.Variable</th>
                <th>U.Medida</th>
                <th>C.Fijo</th>
                <th>C.AO&M</th>
                <th>%</th>
              </Fragment>
            )}
            {(!this.props.usaTramos) && (
              <Fragment>
                <th>Valor</th>
                <th>U.Medida</th>
                <th>%</th>
              </Fragment>
            )}
          </tr>
        </thead>
        {this.renderTBody()}
      </table>
    );
  }
}
