import React, { Component, Fragment } from 'react';
import connect from 'react-redux/es/connect/connect';
import { bindActionCreators } from 'redux';
import PropTypes from 'prop-types';
import { Input, Botonera, Combo, Tabla, VentanaModal, Util, Fecha } from 'appfuture-react';
import axios from 'axios';
import { get as getProp } from 'object-path';

import RUTAS_API from '../../../../global/rutas_api';
import { mostrarAlerta } from '../../../../store/actions/AplicacionAcciones';
import { limpiarDatosHistorico } from '../../../../global/util_nominaciones';

import './GestionLiquidacionCuentaBalance.scss';
import RUTAS_VISTA from '../../../../global/rutas_vista';

class GestionLiquidacionCuentaBalance extends Component {

  state = {
    // Datos de la entidad
    numeroContrato: '0 Seleccionados',
    periodo: '',
    listaContratos: [],
    listaContratosLiquidacion: [],
    // Estado de la aplicacion
    mostrarModalConsulta: false,

  };

  /**
   * Método encargado de comprobar si el formulario ya cargo
   */
  componentDidMount() {
    const { state } = this.props.history && this.props.history.location;
    if (state && state.listaContratos) {
      this.cargarDatos(state.listaContratos);
    }
  };

  /**
   * Método encargado ejecutar una acción cuando se elimina el componente
   */
  componentWillUnmount() {
    this.props.history.replace({ entidadEditar: null });
  };

  /**
   * Método encargado de limpiar los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  limpiarFormulario = (evento) => {
    this.setState({
      // Datos de la entidad
      periodo: '',
      numeroContrato: '0 Seleccionados',
      listaContratos: [],
      listaContratosLiquidacion: [],
      // Estado de la aplicacion
      mostrarModalConsulta: false,
    });
    limpiarDatosHistorico(RUTAS_VISTA.GESTION_LIQUIDACION_CUENTA_BALANCE.url, this.props);
  };

  /**
   * Método encargado de limpiar el formulario al momento de salir
   */
  componentWillUnmount() {
    this.limpiarFormulario();
  };

  /**
   * Método encargado de generar los botones del formulario
   * @returns {Object}
   */
  obtenerFunciones = () => {
    return [
      { texto: 'Guardar', callback: this.guardarEntidad },
      { texto: 'Limpiar', callback: this.limpiarFormulario }
    ];
  };

  /**
   * Método encargado de validar las variables del formulario
   * @returns {Object}
   */
  validarFormulario = () => {
    // Ejemplo Validacion
    if (false) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar al menos un cargo de tipo AO&M para poder continuar.' } };
    }

    const { periodo, listaContratosLiquidacion } = this.state;
    const contratos = listaContratosLiquidacion.filter(contrato => contrato.seleccionado)
      .map(contrato => {
        return contrato.cntIderegistro;
      });

    if (!periodo || (typeof periodo == 'string' && periodo.trim() == '')) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar la fecha del periodo.' } };
    }

    if (!Util.validarArreglo(contratos)) {
      return { respuesta: false, mensaje: { titulo: 'Datos incompletos', mensaje: 'Debe seleccionar mínimo un contrato.' } };
    }

    return { respuesta: true };
  };

  /**
   * Método encargado de guardar los datos de la entidad
   * @returns {bool}
   */
  guardarEntidad = () => {
    const validacion = this.validarFormulario();
    if (!validacion.respuesta) {
      this.props.mostrarAlerta(validacion.mensaje.titulo, validacion.mensaje.mensaje);
      return false;
    }

    const { periodo, listaContratosLiquidacion } = this.state;
    const contratos = listaContratosLiquidacion.filter(contrato => contrato.seleccionado)
      .map(contrato => {
        return contrato.cntIderegistro;
      });

    const entidadGuardar = {
      periodo: periodo,
      idsContratos: JSON.stringify(contratos),
    };

    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_CUENTA_BALANCE.GUARDAR, entidadGuardar)
      .then(respuesta => {
        if (respuesta.data.codigo > 0) {
          this.limpiarFormulario();
        }
      });
  };

  /**
   * Método encargado de abrir la ventana modal del boton consulta
   */
  consultarEntidad = () => {
    this.setState({ mostrarModalConsulta: true });
  };

  /**
   * Método encargado de controlar el cambio del valor de los campos del formulario
   * @param {Event} evento El evento que se ejecuta en el control de usuario
   */
  controlarCambio = (evento) => {
    let change = {};
    change[evento.target.name] = evento.target.value;
    this.setState(change);
  };

  /**
   * Marca como seleccionado un item de la lista.
   */
  seleccionarItemContrato = (evento) => {
    const { listaContratosLiquidacion } = this.state;
    const idContrato = evento.target.value;
    const indexContrato = listaContratosLiquidacion.findIndex(item => item.cntIderegistro == idContrato);
    if (indexContrato >= 0) {
      listaContratosLiquidacion[indexContrato].seleccionado = evento.target.checked;
    }
    this.setState({ listaContratosLiquidacion: listaContratosLiquidacion });
  };

  /**
   * Método encargado de cerrar la ventana modal del boton consulta
   */
  abrirCerrarModal = () => {
    this.setState({
      mostrarModalConsulta: false
    });
  };

  /**
   * Método encargado de cargar los datos de la ventana modal de consulta
   * @param {Array} listaContratos Datos seleccionados de la consulta
   */
  cargarDatos = (listaContratos) => {
    this.setState({
      mostrarModalConsulta: false,
      numeroContrato: (Util.validarArreglo(listaContratos) ? listaContratos.length : 0) + ' Seleccionados',
      listaContratosLiquidacion: (Util.validarArreglo(listaContratos)) ? listaContratos : [],
    });
  };

  /**
   * Método encargado de eliminar el contrato seleccionado
   * @param {number} posicion Posicion de la lista que se desea eliminar
   */
  eliminarContrato = (posicion) => {
    const lista = [...this.state.listaContratos];
    lista.splice(posicion, 1);
    this.setState({ listaContratos: lista });
  };

  /**
   * Método encargado de eliminar el contrato seleccionado
   * @param {number} posicion Posicion de la lista que se desea eliminar
   */
  eliminarContratoLiquidacion = (posicion) => {
    const lista = [...this.state.listaContratosLiquidacion];
    lista.splice(posicion, 1);
    this.setState({ listaContratos: lista });
  };

  /**
   * Método encargado de agregar el contrato seleccionado a la lista de liquidación
   */
  agregarLiquidacion = (posicion) => {

  };

  /**
   * Obtiene en un string los nombres de tipos de contrato del registro actual.
   * @return {string}
   */
  obtenerTiposContrato = (dato) => {
    return dato.listaTipos.map(tipo => {
      return tipo.uniIdetipocontrato.uniNombre1;
    }).join(',');
  };

  /**
   * Método encargado de mostrar la tabla con los contratos seleccionados
   * @returns {Array}
   */
  renderTablaContratosLiquidacion = () => {
    return (
      <table className='table nolabel table-striped table-bordered table-condensed table-hover'>
        <thead>
          <tr>
            <th>Seleccionar</th>
            <th># Contrato</th>
            <th>Tipo de Contrato</th>
            <th>Tercero</th>
            <th>Tipo de Tercero</th>
            <th>Fecha Inicio</th>
            <th>Fecha Final</th>
          </tr>
        </thead>
        <tbody>
          {
            this.state.listaContratosLiquidacion.map((dato, index) => {
              return (
                <tr key={"contrato_" + dato.cntIderegistro}>
                  <td>
                    <label className='visible'>
                      <input type="checkbox" value={dato.cntIderegistro} onChange={this.seleccionarItemContrato} checked={dato.seleccionado} /> Seleccionado
                    </label>
                  </td>
                  <td>{dato.cntNumero}</td>
                  <td>{this.obtenerTiposContrato(dato)}</td>
                  <td>{dato.terIdeagente.terNomcompleto}</td>
                  <td>{'No definido'}</td>
                  <td>{dato.cntFechainicio}</td>
                  <td>{dato.cntFechafin}</td>
                </tr>
              )
            })
          }
        </tbody>
      </table>
    );
  };

  /**
   * Método encargado de mostrar la tabla con los contratos a liquidar seleccionados
   * @returns {Array}
   */
  renderTablaLiquidacion = () => {
    if (!Util.validarArreglo(this.state.datosLiquidacion)) {
      return null;
    }
    return (
      <table className='table nolabel table-striped table-bordered table-hover table-condensed'>
        <thead>
          <tr>
            <th>Contrato</th>
            <th>Periodo</th>
            <th>Cuenta Balance</th>
            <th>Total Medición MBTU</th>
            <th>Desbalance</th>
            <th>% Desbalance</th>
            <th>Compensación M3</th>
            <th>Compensación</th>
            <th>Tarifa Parqueo</th>
            <th>% Cobro</th>
          </tr>
        </thead>
        <tbody>
          {
            this.state.datosLiquidacion.map((dato, index) => {
              return (
                <tr key={"contrato_" + dato.ctnIdregistro}>
                  <td>{getProp(dato, 'cntIdecontrato.cntIderegistro', 0)}</td>
                  <td>{getProp(dato, 'lcbPeriodo', 'Indefinido')}</td>
                  <td>{getProp(dato, 'lcbCuentabalance', 0)}</td>
                  <td>{getProp(dato, 'lcbTotalmedicion', 0)}</td>
                  <td>{getProp(dato, 'lcbDesbalance', 0)}</td>
                  <td>{getProp(dato, 'lcbPorcentajedesbalance', 0)}</td>
                  <td>{getProp(dato, 'lcbCompensacion', 0)}</td>
                  <td>{getProp(dato, 'lcbValorcompensacion', 0)}</td>
                  <td>{getProp(dato, 'lcbTarifaparqueo', 0)}</td>
                  <td>{getProp(dato, 'lcbPorcentajecobro', 0)}</td>
                </tr>
              )
            })
          }
        </tbody>
      </table>
    );
  };

  /**
   * Método encargado de redireccionar a la interfaz de busqueda de contratos
   */
  buscarContrato = () => {
    this.props.history.push({
      pathname: RUTAS_VISTA.CONSULTA_CONTRATOS.url, //Cambiar por la ruta real.
      state: {
        interfazGestion: RUTAS_VISTA.GESTION_LIQUIDACION_CUENTA_BALANCE.url,
        seleccionMultiple: true,
      }
    });
  };

  /**
   * Método encargado de liquidar los contratos seleccionados
   * @returns {Boolean}
   */
  liquidar = () => {
    const { listaContratosLiquidacion, periodo } = this.state;
    const contratos = listaContratosLiquidacion.filter(contrato => contrato.seleccionado)
      .map(contrato => {
        return contrato.cntIderegistro;
      });

    if (!Util.validarArreglo(contratos)) {
      this.props('Error', 'Debe seleccionar como mínimo un contrato para liquidar');
      return;
    }

    if (!periodo || (typeof periodo === 'string' && periodo.trim() == '')) {
      this.props.mostrarAlerta('Error', 'Seleccione el periodo.');
      return;
    }

    axios.post(RUTAS_API.PARAMETRIZACION.GESTION_LIQUIDACION_CUENTA_BALANCE.LIQUIDAR, {
      periodo: periodo,
      idsContratos: JSON.stringify(contratos)
    }).then(respuesta => {
      if (respuesta.data.codigo > 0) {
        this.setState({ datosLiquidacion: respuesta.data.datos });
      }
    });
  };

  /**
   * Método encargado de mostrar el formulario
   * @returns {Object}
   */
  render() {
    return (
      <Fragment>
        <Botonera funciones={this.obtenerFunciones()} />
        <div className='conf-general row mt-5'>
          <div className='form-group col-4'>
            <label>Contrato:</label>
            <div className='input-group'>
              <input type="text" className='form-control' disabled={true} value={this.state.numeroContrato} />
              <div className='input-group-btn'>
                <button className='btn btn-primary' onClick={this.buscarContrato} title="Buscar Contrato"><i className='fa fa-fw fa-check-square-o'></i> Seleccionar contratos</button>
              </div>
            </div>
          </div>
          <Fecha
            label='Periodo a Liquidar'
            name='periodo'
            fecha={this.state.periodo}
            sinDia={true}
            onChange={this.controlarCambio}
          />
        </div>
        {this.state.listaContratosLiquidacion.length > 0 &&
          <div className='pt-3 row'>
            <div className='col-12'>
              {this.renderTablaContratosLiquidacion()}
            </div>
            <div className='col-12'>
              <button className='btn btn-primary' onClick={this.liquidar}><i className='fa fa-fw fa-calculator'></i> Liquidar</button>
            </div>
            <div className='col-12'>
              <hr />
              {this.renderTablaLiquidacion()}
            </div>
          </div>
        }
        {/* <VentanaModal
          mostrar={this.state.mostrarModalConsulta}
          titulo=''
          cerrarModal={this.abrirCerrarModal}>
          <RConsultaLiquidacionCuentaBalance esModal seleccionarEntidad={this.cargarDatos} />
        </VentanaModal> */}
      </Fragment>
    );
  };
}

GestionLiquidacionCuentaBalance.propTypes = {
  history: PropTypes.object,
  mostrarAlerta: PropTypes.func
};

const mapStateToProps = state => {
  return {};
};

const mapDispatchToProps = dispatch => {
  return bindActionCreators({
    mostrarAlerta,
  }, dispatch);
};

const VistaRedux = connect(mapStateToProps, mapDispatchToProps)(GestionLiquidacionCuentaBalance);

export { VistaRedux as RGestionLiquidacionCuentaBalance };
