import React, { Component } from 'react'
import { Botonera, Captcha, Combo, Input, Tabla } from 'appfuture-react'
import connect from 'react-redux/es/connect/connect'
import { bindActionCreators } from 'redux'
import axios from 'axios'
//Rutas entre la vista Solicitud Agendamiento y Symfony-SolicitudAgendamientoController
import URL from '../../global/rutas_api'
//LLamado de la petición por POST o GET empleando AXIOS
import Peticion from '../Assets/util/peticion'
//Carga funciones
import { Util } from '../Assets/util/Util'
//Usar componente modal
import Modal from '../Assets/componentes/Modal'

/**
 *
 *
 * @class SolicitudAgendamiento
 * @extends {Component}
 */
class SolicitudAgendamiento extends Component {
    //inicialización de variables

    /**
     *Define estados iniciales
     * @memberof SolicitudAgendamiento
     */
    state = {
        timer: false,
        actividadJson: [],
        actividad: '-1',
        listaDisponibilidad: [],
        referencia: '',
        observacion: '',
        correo: '',
        numero: '',
    }

    //Arreglo con los id y nombre de columnas para el componente Tabla

    columnas = [
        {
            Header: 'Solicitud agendamiento',

            columns: [
                { Header: 'Fecha', accessor: 'fechadisponbie' },
                { Header: 'Horario', accessor: 'horadisponible' },

                {
                    Header: 'Acción',
                    Cell: (props) => (
                        <input
                            type="radio"
                            name="disponibilidad"
                            onChange={(e) => this.asignar(props.index)}
                        />
                    ),
                },
            ],
        },
    ]

    /**
     *Asigna la fila seleccionada para procesarla más adelante
     *@method
     *@param {int} index - número de la fila
     */
    asignar = (index) => {
        this.setState({ listaDisponibilidad: this.state.lista[index] })
    }

    peticion = new Peticion(this)

    /**
     * Cambia el valor del estado asociado a cada componente
     * @method
     * @async
     * @param {int} id al nombre del estado que se desea modificar
     * @param {(int|string)} value del componente correspondiente al dato
     * que se visualizará en el componente
     */

    change = async ({ target: { id, value } }) => {
        this.setState({ [id]: value })

        if (id === 'actividad') {
            let busquedaListaActividad = this.state.listaActividad.filter(
                (fila) => fila.idprocesoactividad == Util.obtenerId(value)
            )
            busquedaListaActividad = busquedaListaActividad[0]

            await this.peticion
                .post({
                    url: URL.SOLICITUD_AGENDAMIENTO.CONSULTAR_DISPONIBILIDAD,
                    parametros: {
                        suscripcion: this.state.referencia,
                        idActividad: Util.obtenerId(value),
                        idMunicipio: busquedaListaActividad.codigodane,
                        descripcionActividad:
                            busquedaListaActividad.descripcionactividad,
                    },
                })
                .then((data) => {
                    const { disponibilidad } = data[0]
                    if (disponibilidad != null) {
                        this.setState({ lista: disponibilidad })
                    } else {
                        //Modal
                        this.setState({
                            titulo: '¡SIN DISPONIBILIDAD!',
                            texto: `Por el momento no hay disponibilidad`,
                        })
                        //despliega modal
                        this.setState({ mostrar: true })
                    }
                })
        }
    }

    /**
     * Despliega el modal para realizar la consulta
     * @method
     */

    handleConsulta = () =>
        this.setState({ consultaModal: !this.state.consultaModal })

    /**
     * Se ejecuta al momento de pulsar sobre el botón Guardar
     * @method
     * @async
     */

    guardar = async () => {
        //cambiar el estado del modal
        if (
            this.state.listaDisponibilidad.length <= 0 ||
            this.state.referencia === '' ||
            this.state.actividad === '-1' ||
            this.state.numero === '' ||
            this.state.correo === ''
        ) {
            await this.setState({ mensaje: [] })
            this.state.listaDisponibilidad <= 0
                ? this.state.mensaje.push('Solicitud Agendamiento')
                : ''
            this.state.referencia === ''
                ? this.state.mensaje.push('Referencia Cliente')
                : ''
            this.state.actividad === '-1'
                ? this.state.mensaje.push('Actividades')
                : ''
            this.state.numero === ''
                ? this.state.mensaje.push('Número telefónico')
                : ''
            this.state.correo === '' ? this.state.mensaje.push('Correo') : ''

            //Modal
            this.setState({
                titulo: '¡FALTA INFORMACIÓN!',
                texto: `Falta escoger:\n ${this.state.mensaje.join(', ')}`,
            })
            //despliega modal
            this.setState({ mostrar: true })
        } else if (
            Util.validarCorreo(this.state.correo) ||
            Util.validarTelefono(this.state.numero)
        ) {
            await this.setState({ mensaje: [] })
            Util.validarTelefono(this.state.numero)
                ? this.state.mensaje.push(
                      'Número telefónico inválido. Se permite de 7 a 10 dígitos númericos'
                  )
                : ''
            Util.validarCorreo(this.state.correo)
                ? this.state.mensaje.push(
                      'Correo electrónico inválido. Ejemplo: ejemplo@ejemplo.com'
                  )
                : ''

            //Modal
            this.setState({
                titulo: '¡INFORMACIÓN INVÁLIDA!',
                texto: `${this.state.mensaje.join(', ')}`,
            })
            //despliega modal
            this.setState({ mostrar: true })
        } else {
            let busquedaListaActividad = this.state.listaActividad.filter(
                (fila) =>
                    fila.idprocesoactividad ==
                    Util.obtenerId(this.state.actividad)
            )

            const {
                suscripcion,
                idsuscripcion,
                idproceso,
                descripcionactividad,
                idactividad,
                idprocesoactividad,
                camposreferencia,
                tablareferencia,
                duracion,
            } = busquedaListaActividad[0]
            const {
                unidadresponsable,
                fechadisponbie,
                hora,
                jornada,
            } = this.state.listaDisponibilidad

            let campos = {
                sigue_ide: camposreferencia.sigueIde,
                sigue_codemp: camposreferencia.sigueCodemp,
            }

            const datoFinal = {
                idProceso: Util.obtenerId(idproceso),
                suscripcion: suscripcion,
                idsuscripcion: idsuscripcion,
                unidadResponsable: unidadresponsable,
                fechaProgramacion: fechadisponbie,
                sigueIde: camposreferencia.sigueIde,
                duracion: duracion,
                idprocesoActividad: idprocesoactividad,
                tipo: 'D',
                correoCliente: this.state.correo,
                direccionCliente: '',
                telefonoCliente: this.state.numero,
                nombreCliente: '',
                actividad: '',
                observaciones: this.state.observacion,
            }

            this.peticion.post({
                url: URL.SOLICITUD_AGENDAMIENTO.EDITAR_ACTIVIDAD,
                parametros: [datoFinal],
            })

            this.setState({
                actividad: '-1',
                listaDisponibilidad: [],
                listaActividad: [],
                lista: [],
                actividadJson:[],
                referencia: '',
                observacion: '',
                correo: '',
                numero: '',
            })
        }
    }

    /**
     * Restablece los valores a las condiciones iniciales
     * @method
     */

    nuevo = () => {
        this.setState({
            actividad: '-1',
            actividadJson:[],
            listaDisponibilidad: [],
            listaActividad: [],
            referencia: '',
            observacion: '',
            correo: '',
            numero: '',
        })
    }

    //Arreglo para emplearlo en el componente botonera, con sus respectivas funciones

    funciones = [
        { texto: 'guardar', callback: this.guardar },
        { texto: 'nuevo', callback: this.nuevo },
    ]

    /**
     *Detecta el cambio de estado, al digitar en el campo de texto
     *@method
     *@param {Object} e - Adjudicado a un campo de texto
     */
    onBlur = (e) => {
        var currentTarget = e.currentTarget
        setTimeout(() => {
            if (!currentTarget.contains(document.activeElement)) {
                this.peticion
                    .post({
                        url: URL.SOLICITUD_AGENDAMIENTO.LISTAR_ACTIVIDADES,
                        parametros: { idSuscripcion: this.state.referencia },
                        config: ['idactividad', 'descripcionactividad'],
                        json: 'actividadJson',
                        value: 'actividad',
                    })
                    .then((data) => {
                        this.setState({ listaActividad: data })
                    })
            }
        }, 0)
    }

    /**
     *Renderiza la vista 
     * @return {JSX} componente - returna vista jsx 
     */
    render() {
        return (
            <React.Fragment>
                <Modal
                    titulo={this.state.titulo}
                    texto={this.state.texto}
                    mostrar={this.state.mostrar}
                    ocultarAlerta={this.change}
                    botones={this.botones}
                />

                <h1>Empresa - Solicitud agendamiento por demanda</h1>

                <Botonera funciones={this.funciones} />

                <div className="contenedor caja formulario">
                    <label className="tag">Información Ruta</label>
                    <div tabIndex="1" onBlur={this.onBlur}>
                        <Input
                            id="referencia"
                            label="Referencia Cliente:"
                            value={this.state.referencia}
                            onChange={this.change}
                        />
                    </div>
                    <Combo
                        propTexto="texto"
                        propValor="id"
                        id="actividad"
                        label="Actividades:"
                        value={this.state.actividad}
                        opciones={this.state.actividadJson}
                        onChange={this.change}
                    />
                    <div className="contenedor">
                        <Tabla
                            datos={this.state.lista}
                            columnas={this.columnas}
                        />
                    </div>
                    <div className="contenedor caja formulario">
                        <label className="tag">Info contacto</label>
                        <Input
                            id="numero"
                            label="Número telefónico cliente:"
                            value={this.state.numero}
                            onChange={this.change}
                        />
                        <Input
                            id="correo"
                            label="Correo cliente:"
                            value={this.state.correo}
                            onChange={this.change}
                        />
                        <Input
                            id="observacion"
                            label="Observaciones:"
                            value={this.state.observacion}
                            onChange={this.change}
                        />
                    </div>
                </div>

                {/* <Captcha /> */}
            </React.Fragment>
        )
    }
}

SolicitudAgendamiento.propTypes = {}
